local espEnabled = { tracers = false, highlight = false, box = false, health = false }
local espObjects  = {} -- [player] = { highlight, box, tracer, healthBillboard, nameLabel }
 
local function removeESP(p)
    local obj = espObjects[p]
    if not obj then return end
    if obj.highlight       then obj.highlight:Destroy() end
    if obj.box             then pcall(function() obj.box:Remove() end) end
    if obj.tracer          then pcall(function() obj.tracer:Remove() end) end
    if obj.healthBillboard then obj.healthBillboard:Destroy() end
    if obj.nameBillboard   then obj.nameBillboard:Destroy() end
    espObjects[p] = nil
end
 
local function newDrawing(type_, props)
    local ok, obj = pcall(Drawing.new, type_)
    if not ok then return nil end
    for k, v in pairs(props) do pcall(function() obj[k] = v end) end
    return obj
end
 
-- Highlight: vÅ¾dy znovu vytvoÅ s aktuÃ¡lnÃ­m charakterem, zabraÅuje blikÃ¡nÃ­
local function rebuildHighlight(p)
    local obj = espObjects[p]
    if not obj then return end
    if obj.highlight then obj.highlight:Destroy(); obj.highlight = nil end
    if not espEnabled.highlight then return end
    if not p.Character then return end
    local hl = Instance.new("Highlight")
    hl.FillColor          = Color3.fromRGB(220, 50, 50)
    hl.OutlineColor       = Color3.fromRGB(255, 80, 80)
    hl.FillTransparency   = 0.65   -- mÃ©nÄ agresivnÃ­ fill = Å¾Ã¡dnÃ¡ ÄervenÃ¡ mapa
    hl.OutlineTransparency= 0
    hl.DepthMode          = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Adornee            = p.Character
    hl.Parent             = player.PlayerGui -- v PlayerGui neovlivnÃ­ workspace geometrii
    obj.highlight         = hl
end
 
local function setupESP(p)
    if p == player then return end
    if not espObjects[p] then espObjects[p] = {} end
    local obj = espObjects[p]
 
    -- Highlight
    if espEnabled.highlight then
        rebuildHighlight(p)
    elseif obj.highlight then
        obj.highlight:Destroy(); obj.highlight = nil
    end
 
    -- Box (Drawing)
    if espEnabled.box and not obj.box then
        obj.box = newDrawing("Square", {
            Color     = Color3.fromRGB(255, 60, 60),
            Thickness = 1.5,
            Filled    = false,
            Visible   = false,
        })
    elseif not espEnabled.box and obj.box then
        pcall(function() obj.box:Remove() end); obj.box = nil
    end
 
    -- Tracer (Drawing)
    if espEnabled.tracers and not obj.tracer then
        obj.tracer = newDrawing("Line", {
            Color     = Color3.fromRGB(255, 200, 50),
            Thickness = 1,
            Visible   = false,
        })
    elseif not espEnabled.tracers and obj.tracer then
        pcall(function() obj.tracer:Remove() end); obj.tracer = nil
    end
 
    -- Health BillboardGui
    if espEnabled.health and not obj.healthBillboard then
        local bb = Instance.new("BillboardGui")
        bb.Size          = UDim2.new(0, 70, 0, 14)
        bb.StudsOffset   = Vector3.new(0, 3.2, 0)
        bb.AlwaysOnTop   = true
        bb.ResetOnSpawn  = false
        bb.Name          = "DzindraHealth"
        -- background
        local bg = Instance.new("Frame", bb)
        bg.Size               = UDim2.new(1, 0, 1, 0)
        bg.BackgroundColor3   = Color3.fromRGB(20, 20, 20)
        bg.BorderSizePixel    = 0
        bg.Name               = "BG"
        Instance.new("UICorner", bg).CornerRadius = UDim.new(1, 0)
        -- fill bar
        local bar = Instance.new("Frame", bg)
        bar.Size             = UDim2.new(1, 0, 1, 0)
        bar.BackgroundColor3 = Color3.fromRGB(30, 200, 60)
        bar.BorderSizePixel  = 0
        bar.Name             = "HealthBar"
        Instance.new("UICorner", bar).CornerRadius = UDim.new(1, 0)
        -- text
        local lbl = Instance.new("TextLabel", bb)
        lbl.Size                 = UDim2.new(1, 0, 1, 0)
        lbl.BackgroundTransparency = 1
        lbl.TextColor3           = Color3.fromRGB(255, 255, 255)
        lbl.TextStrokeColor3     = Color3.fromRGB(0, 0, 0)
        lbl.TextStrokeTransparency = 0.3
        lbl.TextSize             = 9
        lbl.Font                 = Enum.Font.GothamBold
        lbl.Text                 = "100%"
        lbl.Name                 = "HealthLabel"
        if p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp then bb.Adornee = hrp end
        end
        bb.Parent          = player.PlayerGui
        obj.healthBillboard = bb
    elseif not espEnabled.health and obj.healthBillboard then
        obj.healthBillboard:Destroy(); obj.healthBillboard = nil
    end
end
 
local function refreshAllESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player then setupESP(p) end
    end
end
 
-- RenderStepped update loop
RunService.RenderStepped:Connect(function()
    local anyActive = espEnabled.tracers or espEnabled.box or espEnabled.highlight or espEnabled.health
    if not anyActive then return end
 
    local vp           = Camera.ViewportSize
    local screenCenter = Vector2.new(vp.X / 2, vp.Y)
 
    for _, p in ipairs(Players:GetPlayers()) do
        if p == player then continue end
        local obj = espObjects[p]
        if not obj then continue end
 
        local char = p.Character
        if not char then continue end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
 
        -- Highlight: oprav adornee pokud se charakter zmÄnil
        if obj.highlight then
            if obj.highlight.Adornee ~= char then
                obj.highlight.Adornee = char
            end
        end
 
        -- Health bar update
        if obj.healthBillboard and hrp then
            obj.healthBillboard.Adornee = hrp
            if hum then
                local pct = math.clamp(hum.Health / math.max(hum.MaxHealth, 1), 0, 1)
                local bar = obj.healthBillboard:FindFirstChild("BG") and obj.healthBillboard.BG:FindFirstChild("HealthBar")
                local lbl = obj.healthBillboard:FindFirstChild("HealthLabel")
                if bar then
                    bar.Size             = UDim2.new(pct, 0, 1, 0)
                    bar.BackgroundColor3 = Color3.fromRGB(
                        math.floor((1 - pct) * 220),
                        math.floor(pct * 200),
                        40
                    )
                end
                if lbl then lbl.Text = math.floor(pct * 100) .. "%" end
            end
        end
 
        -- Box + Tracer
        if (obj.box or obj.tracer) and hrp then
            local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
            if onScreen and pos.Z > 0 then
                if obj.box then
                    local scale = 1 / pos.Z
                    local w = 100 * scale * 35
                    local h = w * 1.8
                    obj.box.Size     = Vector2.new(w, h)
                    obj.box.Position = Vector2.new(pos.X - w / 2, pos.Y - h / 2)
                    obj.box.Visible  = true
                end
                if obj.tracer then
                    obj.tracer.From    = screenCenter
                    obj.tracer.To      = Vector2.new(pos.X, pos.Y)
                    obj.tracer.Visible = true
                end
            else
                if obj.box    then obj.box.Visible    = false end
                if obj.tracer then obj.tracer.Visible = false end
            end
        end
    end
end)

local function rejoin()
    local ok, err = pcall(function() TeleportService:Teleport(game.PlaceId, player) end)
    if not ok then warn("[DZINDRAS] Rejoin failed: " .. tostring(err)) end
end
 
local function joinSmallServer()
    local placeId = game.PlaceId
    local cursor, smallestServer, smallestCount = "", nil, math.huge
    repeat
        local ok, result = pcall(function()
            return HttpService:JSONDecode(game:HttpGet(
                "https://games.roblox.com/v1/games/"..placeId.."/servers/Public?sortOrder=Asc&limit=100&cursor="..cursor))
        end)
        if not ok then break end
        for _, s in ipairs(result.data or {}) do
            if s.playing < smallestCount and s.id ~= game.JobId then
                smallestCount = s.playing; smallestServer = s.id
            end
        end
        cursor = result.nextPageCursor or ""
    until cursor == "" or cursor == nil
    if smallestServer then TeleportService:TeleportToPlaceInstance(placeId, smallestServer, player)
    else warn("[DZINDRAS] No small server found.") end
end
 
local function joinBigServer()
    local placeId = game.PlaceId
    local biggestServer, biggestCount = nil, 0
    local ok, result = pcall(function()
        return HttpService:JSONDecode(game:HttpGet(
            "https://games.roblox.com/v1/games/"..placeId.."/servers/Public?sortOrder=Desc&limit=100"))
    end)
    if ok then
        for _, s in ipairs(result.data or {}) do
            if s.playing > biggestCount and s.id ~= game.JobId then
                biggestCount = s.playing; biggestServer = s.id
            end
        end
    end
    if biggestServer then TeleportService:TeleportToPlaceInstance(placeId, biggestServer, player)
    else warn("[DZINDRAS] No big server found.") end
end

local function teleportToName(name)
    local target = workspace:FindFirstChild(name, true)
    if not target then return false, "'"..name.."' not found!" end
    local pos
    if target:IsA("BasePart") or target:IsA("MeshPart") then
        pos = target.Position + Vector3.new(0, 4, 0)
    elseif target:IsA("Model") then
        pos = target:GetModelCFrame().Position + Vector3.new(0, 4, 0)
    elseif target.Parent and target.Parent:IsA("BasePart") then
        pos = target.Parent.Position + Vector3.new(0, 4, 0)
    end
    if pos then rootPart.CFrame = CFrame.new(pos); return true, "Teleported!" end
    return false, "Can't get position."
end
 
local function teleportToAny(names)
    for _, name in ipairs(names) do
        local ok, msg = teleportToName(name)
        if ok then return true, msg end
    end
    return false, "Location not found!"
end

local dmgIndicatorEnabled = false
local dmgConnections = {}
 
local function spawnDmgLabel(targetChar, amount)
    if not targetChar then return end
    local hrp = targetChar:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local bb = Instance.new("BillboardGui")
    bb.Size = UDim2.new(0, 70, 0, 30)
    bb.StudsOffset = Vector3.new(math.random(-10, 10) * 0.1, 3.5 + math.random(0, 15) * 0.1, 0)
    bb.AlwaysOnTop = true; bb.ResetOnSpawn = false; bb.Adornee = hrp
    bb.Parent = player.PlayerGui
    local lbl = Instance.new("TextLabel", bb)
    lbl.Size = UDim2.new(1, 0, 1, 0); lbl.BackgroundTransparency = 1
    lbl.Text = "-" .. tostring(math.floor(amount))
    lbl.TextColor3 = Color3.fromRGB(255, 60, 60)
    lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    lbl.TextStrokeTransparency = 0.3
    lbl.TextSize = 17; lbl.Font = Enum.Font.GothamBold
    local startTime = tick()
    local conn
    conn = RunService.RenderStepped:Connect(function()
        local elapsed = tick() - startTime
        if elapsed > 1.2 then conn:Disconnect(); bb:Destroy(); return end
        local alpha = elapsed / 1.2
        bb.StudsOffset = bb.StudsOffset + Vector3.new(0, 0.045, 0)
        lbl.TextTransparency = alpha
        lbl.TextStrokeTransparency = 0.3 + alpha * 0.7
    end)
end
 
local function hookDmgIndicator()
    for _, c in ipairs(dmgConnections) do c:Disconnect() end
    dmgConnections = {}
    local function watchChar(char)
        if not char then return end
        local hum = char:WaitForChild("Humanoid", 5)
        if not hum then return end
        local lastHP = hum.Health
        local c = hum:GetPropertyChangedSignal("Health"):Connect(function()
            if not dmgIndicatorEnabled then return end
            local diff = lastHP - hum.Health
            if diff > 0 then spawnDmgLabel(char, diff) end
            lastHP = hum.Health
        end)
        table.insert(dmgConnections, c)
    end
    for _, p in ipairs(Players:GetPlayers()) do
        if p == player then continue end
        if p.Character then watchChar(p.Character) end
        local c2 = p.CharacterAdded:Connect(watchChar)
        table.insert(dmgConnections, c2)
    end
    local c3 = Players.PlayerAdded:Connect(function(p)
        if p == player then return end
        if p.Character then watchChar(p.Character) end
        p.CharacterAdded:Connect(watchChar)
    end)
    table.insert(dmgConnections, c3)
end
 
makeToggle("ð¥  Damage Numbers", dmgOrder + 1,
    function() dmgIndicatorEnabled = true; hookDmgIndicator() end,
    function()
        dmgIndicatorEnabled = false
        for _, c in ipairs(dmgConnections) do c:Disconnect() end
        dmgConnections = {}
    end
)
