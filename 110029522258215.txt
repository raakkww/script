local function spawnLightning()
    local segments = math.random(8,11)
    local x = math.random(15,85)/100
    local y = -0.25
    local lastX = x

    for _ = 1, segments do
        local segHeight = math.random(60,90)
        local seg = Instance.new("Frame", lightningLayer)
        seg.AnchorPoint = Vector2.new(0.5,0)
        seg.Position = UDim2.new(lastX,0,y,0)
        seg.Size = UDim2.new(0,4,0,segHeight)
        seg.Rotation = math.random(-20,20)
        seg.BackgroundColor3 = Color3.fromRGB(120,240,255) -- celeste sólido
        seg.BorderSizePixel = 0
        seg.ZIndex = 1

        local glow = Instance.new("UIStroke", seg)
        glow.Thickness = 3
        glow.Color = Color3.fromRGB(120,255,255)
        glow.Transparency = 0

        local offset = math.random(-25,25)/1000
        lastX += offset
        y += segHeight / math.max(1, (game:GetService("Workspace").CurrentCamera and game:GetService("Workspace").CurrentCamera.ViewportSize.Y) or 720)

        -- Desvanece después
        TweenService:Create(seg, TweenInfo.new(0.25), {BackgroundTransparency = 1}):Play()
        task.delay(0.3, function() if seg and seg.Parent then seg:Destroy() end end)
    end
end

local function closeSplash()

    closed = true

    TweenService:Create(blur, TweenInfo.new(0.35), {Size = 0}):Play()
    -- fade gui elements (rápido)
    for _, v in pairs(bg:GetDescendants()) do

        end

    task.wait(0.36)

end

local function RefreshMobList()
    local newList = {}
    local seen = {}
    for _, island in ipairs(MobsFolder:GetChildren()) do
        for _, mob in ipairs(island:GetChildren()) do
            if mob:IsA("Model") and mob:FindFirstChildOfClass("Humanoid") then
                if not seen[mob.Name] then
                    table.insert(newList, mob.Name)
                    seen[mob.Name] = true
                end
            end
        end
    end
    table.sort(newList)
    availableMobs = newList
end

local function createDrop(parent, text, getItemsFunc, callback)
    local b = Instance.new("TextButton", parent)
    b.Size = UDim2.new(0.95,0,0,40)
    b.Text = text
    b.BackgroundColor3 = Color3.fromRGB(30,30,35)

    b.Font = Enum.Font.GothamBold

    Instance.new("UICorner", b)

    local s = Instance.new("ScrollingFrame", parent)
    s.Size = UDim2.new(0.95,0,0,150)
    s.Visible = false
    s.BackgroundColor3 = Color3.fromRGB(20,20,25)
    s.BackgroundTransparency = 0.3
    s.ZIndex = 5
    Instance.new("UIListLayout", s)
    s.ScrollBarThickness = 4

    b.MouseButton1Click:Connect(function()
        s.Visible = not s.Visible
        if s.Visible then
            for _, v in ipairs(s:GetChildren()) do if v:IsA("TextButton") then v:Destroy() end end
            for _, val in ipairs(getItemsFunc()) do
                local btn = Instance.new("TextButton", s)
                btn.Size = UDim2.new(1,0,0,35)
                btn.Text = val
                btn.BackgroundColor3 = Color3.fromRGB(35,35,40)

                btn.ZIndex = 6

                btn.MouseButton1Click:Connect(function() 
                    b.Text = btn.Text
                    s.Visible = false
                    callback(val) 
                end)
            end
        end
    end)
    return b
end

RunService.Stepped:Connect(function()
    if not is_script_active or not farming_enabled then return end
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
   
    for _, part in pairs(char:GetChildren()) do if part:IsA("BasePart") then part.CanCollide = false end end
    if not target_part or not target_part.Parent or not target_part.Parent:FindFirstChildOfClass("Humanoid") or target_part.Parent:FindFirstChildOfClass("Humanoid").Health <= 0 then
        target_part = nil
        local dist = math.huge
        for _, island in ipairs(MobsFolder:GetChildren()) do
            local mobModel = island:FindFirstChild(selected_mob)
            if mobModel and mobModel:IsA("Model") then
                local mHrp = mobModel:FindFirstChild("HumanoidRootPart") or mobModel:FindFirstChildWhichIsA("BasePart")
                local hum = mobModel:FindFirstChildOfClass("Humanoid")
                if mHrp and hum and hum.Health > 0 then
                    local d = (hrp.Position - mHrp.Position).Magnitude
                    if d < dist then dist = d; target_part = mHrp end
                end
            end
        end
    end
    if target_part then
        local offset = Vector3.new(0, farm_distance, 0)
        if farm_mode == "Below" then offset = Vector3.new(0, -farm_distance, 0)
        elseif farm_mode == "Behind" then offset = target_part.CFrame.LookVector * -farm_distance end
       
        hrp.CFrame = CFrame.lookAt(target_part.Position + offset, target_part.Position)
        hrp.Velocity = Vector3.new(0,0,0)
    end
end)

    if not is_script_active then return end
    if is_binding then 
        ui_toggle_key = i.KeyCode
        keyBtn.Text = "Toggle Menu: "..i.KeyCode.Name
        is_binding = false
        return 
    end
    if not p then
        if i.KeyCode == key_farm then farming_enabled = not farming_enabled
        elseif i.KeyCode == key_click then autoclick_enabled = not autoclick_enabled
        elseif i.KeyCode == ui_toggle_key then main.Visible = not main.Visible end
       
        tFarm.Text = farming_enabled and "AUTO FARM: ON [K]" or "AUTO FARM: OFF [K]"
