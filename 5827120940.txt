-- Logic and Functionality

local strengthToggled = false
local rebirthToggled = false
local messageDebounce = false
local isMinimized = false

local replicatedStorage = game:GetService("ReplicatedStorage")
local trainEvent = replicatedStorage:WaitForChild("Events"):WaitForChild("Game"):WaitForChild("Re_TrainPower")
local rebirthEvent = replicatedStorage:WaitForChild("Events"):WaitForChild("Stats"):WaitForChild("Re_Rebirth")

-- Rainbow Background Loop
task.spawn(function()
    local hue = 0
    while true do
        hue = hue + 0.005
        if hue > 1 then hue = 0 end
        pcall(function()
            MainFrame.BackgroundColor3 = Color3.fromHSV(hue, 0.6, 0.5) 
        end)
        task.wait(0.03) 
    end
end)

-- Background auto-farm loops
task.spawn(function()
    while true do
        if strengthToggled then pcall(function() trainEvent:FireServer() end) end
        task.wait(0.1) 
    end
end)

task.spawn(function()
    while true do
        if rebirthToggled then pcall(function() rebirthEvent:FireServer() end) end
        task.wait(0.1)
    end
end)

-- Minimize Button Functionality
MinimizeButton.MouseButton1Click:Connect(function()
    isMinimized = not isMinimized
    if isMinimized then
        MainFrame:TweenSize(UDim2.new(0, 280, 0, 40), Enum.EasingDirection.Out, Enum.EasingStyle.Quart, 0.3, true)
        MinimizeButton.Text = "+"
    else
        MainFrame:TweenSize(UDim2.new(0, 280, 0, 320), Enum.EasingDirection.Out, Enum.EasingStyle.Quart, 0.3, true)
        MinimizeButton.Text = "-"
    end
end)

-- Strength Toggle Event
ToggleButton.MouseButton1Click:Connect(function()
    strengthToggled = not strengthToggled
    if strengthToggled then
        ToggleButton.Text = "Auto Gain Strength: ON"
        ToggleButton.BackgroundColor3 = Color3.fromRGB(50, 200, 50) 
    else
        ToggleButton.Text = "Auto Gain Strength: OFF"
        ToggleButton.BackgroundColor3 = Color3.fromRGB(200, 50, 50) 
    end
end)

-- Rebirth Toggle Event
RebirthButton.MouseButton1Click:Connect(function()
    rebirthToggled = not rebirthToggled
    if rebirthToggled then
        RebirthButton.Text = "Auto Rebirth: ON"
        RebirthButton.BackgroundColor3 = Color3.fromRGB(50, 200, 50) 
    else
        RebirthButton.Text = "Auto Rebirth: OFF"
        RebirthButton.BackgroundColor3 = Color3.fromRGB(200, 50, 50) 
    end
end)

-- AutoWin Patched Message Event
WinButton.MouseButton1Click:Connect(function()
    if messageDebounce or isMinimized then return end
    messageDebounce = true
    
    for i = 1, 0, -0.1 do
        MessageLabel.TextTransparency = i
        task.wait(0.02)
    end
    MessageLabel.TextTransparency = 0
    
    task.wait(5)
    
    for i = 0, 1, 0.1 do
        MessageLabel.TextTransparency = i
        task.wait(0.02)
    end
    MessageLabel.TextTransparency = 1
    
    messageDebounce = false
end)
