local function fire(mod, act, a1, a2)
    if LocalData and LocalData.DataReplica then
        pcall(function() LocalData.DataReplica:FireServer(mod, act, a1, a2) end)
    elseif ReplicaSignal and ReplicaID then
        local args = {ReplicaID, mod, act}
        if a1 ~= nil then table.insert(args, tostring(a1)) end
        if a2 ~= nil then table.insert(args, tostring(a2)) end
        pcall(function() ReplicaSignal:FireServer(table.unpack(args)) end)
    end
end

local function magnetPickups()
    if not flags.pickup then return end
    local char = player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local parts = {}
    local pickupsFolder = Workspace:FindFirstChild("Pickups")
    if pickupsFolder then
        for _, obj in ipairs(pickupsFolder:GetDescendants()) do
            if obj:IsA("BasePart") and obj.Name ~= "Collect" and obj.Name ~= "Flare" and obj.Name ~= "Sparkle" then
                table.insert(parts, obj)
            end
        end
    end
    local energyFolder = Workspace:FindFirstChild("EnergyPickups")
    if energyFolder then
        for _, obj in ipairs(energyFolder:GetDescendants()) do
            if obj:IsA("BasePart") then table.insert(parts, obj) end
        end
    end
    if #parts == 0 then return end

    local originalCF = hrp.CFrame
    for _, part in ipairs(parts) do
        if part and part.Parent then
            pcall(function() hrp.CFrame = CFrame.new(part.Position) end)
            pickupClaimed = pickupClaimed + 1
            task.wait()
        end
    end
    pcall(function() hrp.CFrame = originalCF end)
end

local goClickStats = 0

local function fireButton(btn)
    local getconn = getconnections or get_signal_cons
    if not getconn then return false end
    local fired = false
    for _, sigName in ipairs({"MouseButton1Click", "Activated", "MouseButton1Down", "MouseButton1Up"}) do
        local ok, sig = pcall(function() return btn[sigName] end)
        if ok and sig then
            pcall(function()
                for _, conn in ipairs(getconn(sig)) do
                    pcall(function() conn:Fire() end)
                    pcall(function() conn:Fire(btn) end)
                    fired = true
                end
            end)
        end
    end
    return fired
end

local function hasGoText(obj)
    if (obj:IsA("TextButton") or obj:IsA("TextLabel")) and obj.Text then
        if obj.Text:upper():gsub("%s", "") == "GO" then return true end
    end
    for _, child in ipairs(obj:GetChildren()) do
        if (child:IsA("TextLabel") or child:IsA("TextButton")) and child.Text then
            if child.Text:upper():gsub("%s", "") == "GO" then return true end
        end
    end
    return false
end

local function getBuildProgress(name)
    if LocalData and LocalData.DataReplica and LocalData.DataReplica.Data 
       and LocalData.DataReplica.Data.Plot and LocalData.DataReplica.Data.Plot.Buildable then
        return LocalData.DataReplica.Data.Plot.Buildable[name] or 0

local function getBuildCost(name)
    if BuildConfig and BuildConfig.Pieces and BuildConfig.Pieces[name] then
        return BuildConfig.Pieces[name].Cost
    end
    return nil
end

local function isBuildComplete(name)
    local cost = getBuildCost(name)
    if not cost then return false end
    return getBuildProgress(name) >= cost
end

local function getNextBuildTarget()
    local plot = getMyPlot()
    if not plot then return nil end
    local buildableFolder = plot:FindFirstChild("Buildable")
    if not buildableFolder then return nil end
    for _, obj in ipairs(buildableFolder:GetChildren()) do
        if obj:IsA("Model") or obj:IsA("BasePart") then
            local name = obj.Name
            local cfg = BuildConfig and BuildConfig.Pieces and BuildConfig.Pieces[name]
            if cfg and not cfg.Disabled and not isBuildComplete(name) then
                return name
            end
        end
    end
    return nil
end

local function releaseBuild()
    if currentBuildTarget and BuildHold then
        pcall(function() BuildHold:FireServer(currentBuildTarget, false) end)
        currentBuildTarget = nil
    end
end

local function startHold(name)
    if not BuildHold then return end
    currentBuildTarget = name
    pcall(function() BuildHold:FireServer(name, true) end)
end

local function buildTick()
    if not flags.build then
        if currentBuildTarget then releaseBuild() end
        buildDisplayName = "OFF"
        return
    end
    local target = getNextBuildTarget()
    if not target then
        if currentBuildTarget then releaseBuild() end
        buildDisplayName = "Done"
        return
    end
    if currentBuildTarget ~= target then
        if currentBuildTarget then
            releaseBuild()
            task.wait(0.1)
        end
        startHold(target)
    else
        pcall(function() BuildHold:FireServer(target, true) end)
    end
    local progress = getBuildProgress(target)
    local cost = getBuildCost(target) or 0
    buildDisplayName = string.format("%s %d/%d", target, progress, cost)
end

--// ============ RESERVATION + ROUND-ROBIN ============

local function computeTotalDemand()
    local demand = {}
    local orders = getOrders()
    for _, order in ipairs(orders) do
        for _, req in ipairs(order.Items) do
            demand[req.Id] = (demand[req.Id] or 0) + req.Count
        end
    end
    return demand, #orders
end

local function computeReservations(slots, demand)
    local reservedSlots = {}
    local reservedCount = 0
    local missing = {}

    local ids = {}
    for id in pairs(demand) do table.insert(ids, id) end
    table.sort(ids)

    for _, id in ipairs(ids) do
        local needed = demand[id]
        local available = {}
        for slotStr, data in pairs(slots) do
            if data and data.Id == id and not data.Hidden and not data.Locked then
                table.insert(available, slotStr)
            end
        end
        table.sort(available, function(a,b) return tonumber(a)<tonumber(b) end)

        local toReserve = math.min(needed, #available)
        for i = 1, toReserve do
            reservedSlots[available[i]] = true
            reservedCount = reservedCount + 1
        end

        local stillMissing = needed - toReserve
        if stillMissing > 0 then
            missing[id] = stillMissing
        end
    end

    return reservedSlots, reservedCount, missing
end

local function buildReverseMap()
    local rm = {}
    for id, def in pairs(Items) do
        if def.MergesInto then
            if not rm[def.MergesInto] then rm[def.MergesInto] = {} end
            table.insert(rm[def.MergesInto], id)
        end
    end
    return rm
end

local function expandChain(targetId, reverseMap, outSet)
    if outSet[targetId] then return end
    outSet[targetId] = true
    local sources = reverseMap[targetId]
    if sources then
        for _, src in ipairs(sources) do
            expandChain(src, reverseMap, outSet)
        end
    end
end

local rrCycle = 0
local function getRotatedMissingIds(missing)
    local ids = {}
    for id in pairs(missing) do table.insert(ids, id) end
    table.sort(ids)
    if #ids == 0 then return ids end

    rrCycle = (rrCycle + 1) % #ids
    local rotated = {}
    for i = 1, #ids do
        local idx = ((rrCycle + i - 1) % #ids) + 1
        table.insert(rotated, ids[idx])
    end
    return rotated
end

local function computeUsefulForId(targetId, reverseMap)
    local s = {}
    expandChain(targetId, reverseMap, s)
    return s
end

--// ============ MERGE / TAP ============
local function countEmpty(slots)
    local c = 0
    for i = 1, GridSize do
        if not slots[tostring(i)] then c = c + 1 end
    end
    return c
end

local function findMergesForTarget(slots, reservedSlots, usefulSetTarget)
    local groups = {}
    for str, data in pairs(slots) do
        if data and not data.Hidden and not reservedSlots[str] and usefulSetTarget[data.Id] then
            local id = data.Id
            if not groups[id] then groups[id] = {locked={}, unlocked={}} end
            if data.Locked then
                table.insert(groups[id].locked, str)
            else
                table.insert(groups[id].unlocked, str)
            end
        end
    end

    local pairs_list = {}
    for id, g in pairs(groups) do
        local def = getDef(id)
        if def and def.MergesInto then
            table.sort(g.unlocked, function(a,b) return tonumber(a)<tonumber(b) end)
            table.sort(g.locked, function(a,b) return tonumber(a)<tonumber(b) end)
            while #g.unlocked >= 1 and #g.locked >= 1 do
                table.insert(pairs_list, {g.unlocked[1], g.locked[1]})
                table.remove(g.unlocked, 1); table.remove(g.locked, 1)
            end
            for i = 1, #g.unlocked - 1, 2 do
                table.insert(pairs_list, {g.unlocked[i], g.unlocked[i+1]})
            end
        end
    end
    return pairs_list
end

local function findTapsForTarget(slots, queues, cooldowns, uses, reservedSlots, usefulSetTarget)
    local targets = {}
    local energy = getEnergy()
    local now = os.time()
    local empty = countEmpty(slots)

    for slotStr, data in pairs(slots) do
        if data and not data.Hidden and not data.Locked and not reservedSlots[slotStr] then
            local def = getDef(data.Id)
            if def then
                local isUseful = usefulSetTarget[data.Id]
                if not isUseful and queues and queues[slotStr] and queues[slotStr][1] then
                    local nextItem = queues[slotStr][1]
                    local nextId = type(nextItem) == "table" and (nextItem.Id or nextItem[1]) or nextItem
                    if nextId and usefulSetTarget[nextId] then isUseful = true end
                end

                if isUseful then
                    local ready, kind = false, nil
                    if def.OnTap then
                        local ok = true
                        if def.OnTap.EnergyCost and energy < def.OnTap.EnergyCost then ok = false end
                        if cooldowns and cooldowns[slotStr] and now < cooldowns[slotStr] then ok = false end
                        if def.OnTap.MaxUses and uses and uses[slotStr] ~= nil and uses[slotStr] <= 0 then ok = false end
                        if def.OnTap.Type == "spawn" then
                            if not queues or not queues[slotStr] or #queues[slotStr] == 0 then ok = false end
                            if empty <= 0 then ok = false end
                        end
                        if ok then ready = true; kind = "OnTap" end
                    end
                    if not ready and def.AutoGen then
                        local ok = true
                        if not queues or not queues[slotStr] or #queues[slotStr] == 0 then ok = false end
                        if cooldowns and cooldowns[slotStr] and now < cooldowns[slotStr] then ok = false end
                        if empty <= 0 then ok = false end
                        if ok then ready = true; kind = "AutoGen" end
                    end
                    if ready then
                        table.insert(targets, {slot = slotStr, kind = kind})
                    end
                end
            end
        end
    end
    return targets
end

local function findNormalTaps(slots, queues, cooldowns, uses, reservedSlots)
    local targets = {}
    local energy = getEnergy()
    local now = os.time()
    local empty = countEmpty(slots)

    for slotStr, data in pairs(slots) do
        if data and not data.Hidden and not data.Locked and not reservedSlots[slotStr] then
            local def = getDef(data.Id)
            if def then
                local ready, kind = false, nil
                if def.OnTap then
                    local ok = true
                    if def.OnTap.EnergyCost and energy < def.OnTap.EnergyCost then ok = false end
                    if cooldowns and cooldowns[slotStr] and now < cooldowns[slotStr] then ok = false end
                    if def.OnTap.MaxUses and uses and uses[slotStr] ~= nil and uses[slotStr] <= 0 then ok = false end
                    if def.OnTap.Type == "spawn" then
                        if not queues or not queues[slotStr] or #queues[slotStr] == 0 then ok = false end
                        if empty <= 0 then ok = false end
                    end
                    if ok then ready = true; kind = "OnTap" end
                end
                if not ready and def.AutoGen then
                    local ok = true
                    if not queues or not queues[slotStr] or #queues[slotStr] == 0 then ok = false end
                    if cooldowns and cooldowns[slotStr] and now < cooldowns[slotStr] then ok = false end
                    if empty <= 0 then ok = false end
                    if ok then ready = true; kind = "AutoGen" end
                end
                if ready then
                    table.insert(targets, {slot = slotStr, kind = kind})
                end
            end
        end
    end
    return targets
end

local function findNormalMerges(slots, reservedSlots)
    local groups = {}
    for str, data in pairs(slots) do
        if data and not data.Hidden and not reservedSlots[str] then
            local id = data.Id
            if not groups[id] then groups[id] = {locked={}, unlocked={}} end
            if data.Locked then
                table.insert(groups[id].locked, str)
            else
                table.insert(groups[id].unlocked, str)
            end
        end
    end

    local pairs_list = {}
    for id, g in pairs(groups) do
        local def = getDef(id)
        if def and def.MergesInto then
            table.sort(g.unlocked, function(a,b) return tonumber(a)<tonumber(b) end)
            table.sort(g.locked, function(a,b) return tonumber(a)<tonumber(b) end)
            while #g.unlocked >= 1 and #g.locked >= 1 do
                table.insert(pairs_list, {g.unlocked[1], g.locked[1]})
                table.remove(g.unlocked, 1); table.remove(g.locked, 1)
            end
            for i = 1, #g.unlocked - 1, 2 do
                table.insert(pairs_list, {g.unlocked[i], g.unlocked[i+1]})
            end
        end
    end
    return pairs_list
end

local function findCollectables(slots, reservedSlots)
    local list = {}
    for slotStr, data in pairs(slots) do
        if data and not data.Hidden and not data.Locked and not reservedSlots[slotStr] then
            local def = getDef(data.Id)
            if def and def.OnCollect then table.insert(list, slotStr) end
        end
    end
    return list
end

--// ============ LOOPS ============
local running = false
local actionsThisCycle = 0
local ordersServed = 0
local activeOrdersCount = 0
local statusMode = "Idle"
local targetDisplay = "—"
local reservedDisplay = "—"

local function mergeLoop()
    local reverseMap = buildReverseMap()

    while running do
        if flags.merge then
            local grid = getGrid()
            if grid then
                local slots = grid.Slots or {}
                local queues = grid.Queues or {}
                local cooldowns = grid.Cooldowns or {}
                local uses = grid.Uses or {}

                local demand, oc = computeTotalDemand()
                activeOrdersCount = oc
                local hasOrders = oc > 0

                local reservedSlots, reservedCount, missing = computeReservations(slots, demand)

                local totalDemand = 0
                for _, c in pairs(demand) do totalDemand = totalDemand + c end
                reservedDisplay = string.format("%d / %d", reservedCount, totalDemand)

                local smart = flags.smart
                local collectables = findCollectables(slots, reservedSlots)

                local rotatedMissing = getRotatedMissingIds(missing)
                local hasMissing = #rotatedMissing > 0

                local actionsOrder = 0
                local actionsStock = 0

                for _, slotStr in ipairs(collectables) do
                    if not running then break end
                    fire("MergeGrid", "Collect", slotStr)
                    task.wait(0.05)
                end
                actionsOrder = actionsOrder + #collectables

                if smart and hasMissing then
                    targetDisplay = rotatedMissing[1] .. " +" .. (#rotatedMissing - 1)

                    for _, targetId in ipairs(rotatedMissing) do
                        if not running then break end
                        local usefulForThis = computeUsefulForId(targetId, reverseMap)

                        local tapsT = findTapsForTarget(slots, queues, cooldowns, uses, reservedSlots, usefulForThis)
                        local mergesT = findMergesForTarget(slots, reservedSlots, usefulForThis)

                        local maxPerTarget = 2
                        for i = 1, math.min(maxPerTarget, #tapsT) do
                            if not running then break end
                            local t = tapsT[i]
                            fire("MergeGrid", t.kind == "AutoGen" and "TapAutoGen" or "Tap", t.slot)
                            actionsOrder = actionsOrder + 1
                            task.wait(0.05)
                        end
                        for i = 1, math.min(maxPerTarget, #mergesT) do
                            if not running then break end
                            local p = mergesT[i]
                            fire("MergeGrid", "Merge", p[1], p[2])
                            actionsOrder = actionsOrder + 1
                            task.wait(0.07)
                        end
                    end
                else
                    if hasOrders then
                        targetDisplay = "all ready"
                    else
                        targetDisplay = "no orders"
                    end
                    local taps = findNormalTaps(slots, queues, cooldowns, uses, reservedSlots)
                    local merges = findNormalMerges(slots, reservedSlots)
                    for _, t in ipairs(taps) do
                        if not running then break end
                        fire("MergeGrid", t.kind == "AutoGen" and "TapAutoGen" or "Tap", t.slot)
                        task.wait(0.05)
                    end
                    for _, p in ipairs(merges) do
                        if not running then break end
                        fire("MergeGrid", "Merge", p[1], p[2])
                        task.wait(0.07)
                    end
                    actionsStock = #taps + #merges
                end

                if smart and hasMissing and actionsOrder <= #collectables then
                    local taps = findNormalTaps(slots, queues, cooldowns, uses, reservedSlots)
                    local merges = findNormalMerges(slots, reservedSlots)
                    for _, t in ipairs(taps) do
                        if not running then break end
                        fire("MergeGrid", t.kind == "AutoGen" and "TapAutoGen" or "Tap", t.slot)
                        task.wait(0.05)
                    end
                    for _, p in ipairs(merges) do
                        if not running then break end
                        fire("MergeGrid", "Merge", p[1], p[2])
                        task.wait(0.07)
                    end
                    actionsStock = #taps + #merges
                end

                actionsThisCycle = actionsOrder + actionsStock

                if actionsOrder > #collectables and hasOrders then
                    statusMode = "Order"
                elseif actionsThisCycle > 0 then
                    statusMode = "Stocking"
                else
                    statusMode = "Waiting"
                end
            end
        end
        task.wait(actionsThisCycle > 0 and 0.12 or 0.4)
    end
end

local function pickupLoop()
    while running do
        if flags.pickup then magnetPickups() end
        task.wait(0.3)
    end
end

local function ordersLoop()
    while running do
        if flags.orders then
            local before = goClickStats
            clickAllGo()
            if goClickStats > before then
                ordersServed = ordersServed + (goClickStats - before)
            end
        end
        task.wait(0.25)
    end
end

local function buildLoop()
    while running do
        buildTick()
        task.wait(0.5)
    end
    if currentBuildTarget then releaseBuild() end
end

local function speedLoop()
    while running do
        applyWalkSpeed()
        task.wait(1)
    end
end

local function uiLoop()
    while running do
        if actionsThisCycle > 0 then
            infoStatus.Text = statusMode .. " " .. actionsThisCycle
            if statusMode == "Order" then
                infoStatus.TextColor3 = Color3.fromRGB(130,220,150)
            elseif statusMode == "Stocking" then
                infoStatus.TextColor3 = Color3.fromRGB(220,200,130)
            else
                infoStatus.TextColor3 = Color3.fromRGB(180,180,195)
            end
        else
            infoStatus.Text = "Waiting"
            infoStatus.TextColor3 = Color3.fromRGB(150,150,165)
        end
        infoOrders.Text = string.format("%d / %s", activeOrdersCount, tostring(getEnergy()))
        infoReserved.Text = reservedDisplay
        infoTarget.Text = targetDisplay
        infoBuild.Text = buildDisplayName
        task.wait(0.3)
    end
end

        task.spawn(buildLoop)

        if currentBuildTarget then releaseBuild() end
        buildDisplayName = "—"
        targetDisplay = "—"
        reservedDisplay = "—"
    end
end)
