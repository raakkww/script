local function getItemMainPart(item)
    if item.PrimaryPart then return item.PrimaryPart end
    for _, child in ipairs(item:GetChildren()) do
        if child:IsA("BasePart") then
            return child
        end
    end
    return nil
end

local function removeMobESP(char)
    local esp = mobESPInstances[char]
    if esp then
        if esp.Highlight then esp.Highlight:Destroy() end
        if esp.Billboard then esp.Billboard:Destroy() end
        if esp.DistanceConnection then
            esp.DistanceConnection:Disconnect()
        end
        mobESPInstances[char] = nil
    end
end

local function removeItemESP(item)
    local esp = itemESPInstances[item]
    if esp then
        if esp.Highlight then esp.Highlight:Destroy() end
        if esp.Billboard then esp.Billboard:Destroy() end
        if esp.DistanceConnection then
            esp.DistanceConnection:Disconnect()
        end
        itemESPInstances[item] = nil
    end
end

local function refreshMobESP()
    for char, _ in pairs(mobESPInstances) do
        removeMobESP(char)
    end
    if not mobOptions.ESP then return end
    if not charactersFolder then
        Library:Notify({ Title = "Mob ESP", Description = "Characters folder not found", Time = 3 })
        return
    end
    for _, child in ipairs(charactersFolder:GetChildren()) do
        if table.find(mobNames, child.Name) then
            createMobESP(child)
        end
    end
end

local function refreshItemESP()
    for item, _ in pairs(itemESPInstances) do
        removeItemESP(item)
    end
    if not itemOptions.ESP then return end
    if not droppedItemsFolder then
        Library:Notify({ Title = "Item ESP", Description = "DroppedItems folder not found", Time = 3 })
        return
    end
    for _, child in ipairs(droppedItemsFolder:GetChildren()) do
        if table.find(itemNames, child.Name) then
            createItemESP(child)
        end
    end
end

-- Anti-Slow function (without auto-jump)
local function startAntiSlow()
    if antiSlowActive then return end
    antiSlowActive = true
    
    speedCheckConnection = RunService.Heartbeat:Connect(function()
        if not Toggles.SpeedHack.Value then return end
        
        local char = LocalPlayer.Character
        if char then
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            if humanoid then
                -- Check if speed was reduced (anti-slow)
                if humanoid.WalkSpeed < currentSpeedValue and humanoid.WalkSpeed ~= 0 then
                    -- Restore speed hack speed without affecting jump
                    humanoid.WalkSpeed = currentSpeedValue
                end
                
                -- Prevent auto-jump by ensuring jump power isn't modified
                if humanoid.JumpPower ~= (originalValues.jumpPower or 50) then
                    if not Toggles.InfJump.Value then
                        humanoid.JumpPower = originalValues.jumpPower or 50
                    end
                end
            end
        end
    end)
    
    table.insert(connections, speedCheckConnection)
end

local function stopAntiSlow()
    if speedCheckConnection then
        speedCheckConnection:Disconnect()
        speedCheckConnection = nil
    end
    antiSlowActive = false
end

-- Apply speed hack function (without affecting jump)
local function applySpeedHack(state, speedValue)
    local char = LocalPlayer.Character
    if char then
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        if humanoid then
            if state then
                originalValues.walkSpeed = humanoid.WalkSpeed
                originalValues.jumpPower = humanoid.JumpPower
                humanoid.WalkSpeed = speedValue
                currentSpeedValue = speedValue
                Library:Notify({ Title = "Speed Hack", Description = "Speed set to " .. speedValue, Time = 2 })
            else
                humanoid.WalkSpeed = originalValues.walkSpeed or 16
                if originalValues.jumpPower then
                    humanoid.JumpPower = originalValues.jumpPower
                end
                Library:Notify({ Title = "Speed Hack", Description = "Speed restored to " .. (originalValues.walkSpeed or 16), Time = 2 })
            end
        end
    end
end

-- ========== UPDATED AUTO PICKUP FUNCTIONS (MULTIPLE ITEMS SELECTION) ==========
local function isItemAllowed(itemName)
    -- Check if "Pick Up All Items" is enabled
    if Toggles.AllItems.Value then
        return true
    end
    
    -- Check if multiple items mode is enabled (replaces single item mode)
    if Toggles.MultipleItemsMode.Value then
        -- Check if the item is in the selected multiple items list
        for _, selectedItem in pairs(selectedMultipleItems) do
            if itemName == selectedItem then
                return true
            end
        end
        return false
    end
    
    -- Check whitelist (multi-select)
    local whitelist = Options.ItemWhitelist.Value
    if whitelist and type(whitelist) == "table" then
        for _, v in pairs(whitelist) do
            if v == itemName then
                return true
            end
        end
    end
    
    return false
end

local function startAutoPickup()
    if autoPickupConnection then return end
    
    autoPickupConnection = RunService.Heartbeat:Connect(function()
        if not Toggles.AutoPickup.Value then return end

        local char = LocalPlayer.Character
        if not char then return end
        local rootPart = char:FindFirstChild("HumanoidRootPart")
        if not rootPart then return end

        local radius = Options.PickupRadius.Value
        local pos = rootPart.Position

        if not droppedItemsFolder then return end

        -- Get the list of items to pick up
        local itemsToPickup = {}
        
        for _, item in ipairs(droppedItemsFolder:GetChildren()) do
            if isItemAllowed(item.Name) then
                table.insert(itemsToPickup, item)
            end
        end
        
        -- Sort by distance (optional but more efficient)
        table.sort(itemsToPickup, function(a, b)
            local aPart = a.PrimaryPart or getItemMainPart(a)
            local bPart = b.PrimaryPart or getItemMainPart(b)
            if not aPart or not bPart then return false end
            local distA = (aPart.Position - pos).Magnitude
            local distB = (bPart.Position - pos).Magnitude
            return distA < distB
        end)
        
        for _, item in ipairs(itemsToPickup) do
            if not Toggles.AutoPickup.Value then break end
            
            local mainPart = item.PrimaryPart or getItemMainPart(item)
            if mainPart then
                local dist = (mainPart.Position - pos).Magnitude
                if dist <= radius then
                    -- Try to pick up the item
                    local success = false
                    
                    if adjustBackpackRemote then
                        pcall(function()
                            adjustBackpackRemote:FireServer(item)
                            success = true
                        end)
                    end
                    
                    if not success and pickUpItemRemote then
                        pcall(function()
                            pickUpItemRemote:FireServer(item)
                            success = true
                        end)
                    end
                    
                    if not success then
                        local fireProximity = item:FindFirstChild("FireProximity")
                        if fireProximity then
                            pcall(function()
                                fireProximity:FireServer()
                                success = true
                            end)
                        end
                    end
                    
                    -- Small delay to prevent overwhelming the server
                    task.wait(0.05)
                end
            end
        end
    end)
end

local function stopAutoPickup()
    if autoPickupConnection then
        autoPickupConnection:Disconnect()
        autoPickupConnection = nil
    end
end

-- Fixed Jump Connection - Proper infinite jump
local jumpConn = UserInputService.JumpRequest:Connect(function()
    if Toggles.InfJump.Value then
        local char = LocalPlayer.Character
        if char then
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            if humanoid then
                -- Remove the state checks that prevent jumping
                -- Just force the jump state
                humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
                -- Also set jump power to a higher value for better effect
                if originalValues.jumpPower == nil then
                    originalValues.jumpPower = humanoid.JumpPower
                end
                humanoid.JumpPower = 80 -- Higher jump for better infinite jump feel
            end
        end
    end
end)
table.insert(connections, jumpConn)

-- Also add a spacebar listener as backup (sometimes JumpRequest doesn't fire)
local spacebarConn = UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    
    if Toggles.InfJump.Value and input.KeyCode == Enum.KeyCode.Space then
        local char = LocalPlayer.Character
        if char then
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            if humanoid then
                -- Only jump if not already jumping (to prevent double jumps)
                if humanoid:GetState() ~= Enum.HumanoidStateType.Jumping and 
                   humanoid:GetState() ~= Enum.HumanoidStateType.Falling then
                    humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
                end
            end
        end
    end
end)
table.insert(connections, spacebarConn)

-- Unload Cleanup
Library:OnUnload(function()
    for char, _ in pairs(mobESPInstances) do
        removeMobESP(char)
    end
    for item, _ in pairs(itemESPInstances) do
        removeItemESP(item)
    end

    for _, conn in ipairs(connections) do
        if typeof(conn) == "RBXScriptConnection" then
            conn:Disconnect()
        end
    end
    connections = {}

    stopAutoPickup()
    stopAntiSlow()

    if Toggles.SpeedHack.Value then
        local char = LocalPlayer.Character
        if char then
            local humanoid = char:FindFirstChildOfClass("Humanoid")
            if humanoid then
                humanoid.WalkSpeed = originalValues.walkSpeed or 16
                if originalValues.jumpPower then
                    humanoid.JumpPower = originalValues.jumpPower
                end
            end
        end
    end

    Library:Notify({ Title = "", Description = "Bye!", Time = 3 })
    print("")
end)

-- UI Settings Tab
local MenuGroup = Tabs["UI Settings"]:AddLeftGroupbox("Menu", "wrench")

MenuGroup:AddToggle("KeybindMenuOpen", {
    Default = Library.KeybindFrame.Visible,
    Text = "Open Keybind Menu",
    Callback = function(value)
        Library.KeybindFrame.Visible = value
    end,
})

MenuGroup:AddToggle("ShowCustomCursor", {
    Text = "Custom Cursor",
    Default = true,
    Callback = function(Value)
        Library.ShowCustomCursor = Value
    end,
})

MenuGroup:AddDropdown("NotificationSide", {
    Values = { "Left", "Right" },
    Default = "Right",
    Text = "Notification Side",
    Callback = function(Value)
        Library:SetNotifySide(Value)
    end,
})

MenuGroup:AddDropdown("DPIDropdown", {
    Values = { "50%", "75%", "100%", "125%", "150%", "175%", "200%" },
    Default = "100%",
    Text = "DPI Scale",
    Callback = function(Value)
        Value = Value:gsub("%%", "")
        local DPI = tonumber(Value)
        Library:SetDPIScale(DPI)
    end,
})

MenuGroup:AddSlider("UICornerSlider", {
    Text = "Corner Radius",
    Default = Library.CornerRadius,
    Min = 0,
    Max = 20,
    Rounding = 0,
    Callback = function(value)
        Window:SetCornerRadius(value)
    end,
})

MenuGroup:AddDivider()
MenuGroup:AddLabel("Menu bind")
    :AddKeyPicker("MenuKeybind", { Default = "Insert", NoUI = true, Text = "Menu keybind" })

Library.ToggleKeybind = Options.MenuKeybind

SaveManager:SetLibrary(Library)

    Title = "",

    Time = 5,
})
