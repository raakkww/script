local function UpdateMenuVisuals()
    if Flags.CurrentGame == "Misc" then
        SectionTitle.Text = "Miscellaneous Options"
        GamePanel.Visible = false
        MiscPanel.Visible = true
        
        FullbrightBtn.Text = "Global Fullbright: " .. (Flags.Fullbright and "ON" or "OFF")
        FullbrightBtn.BackgroundColor3 = Flags.Fullbright and Color3.fromRGB(0, 140, 60) or Color3.fromRGB(45, 45, 50)
        
        TeammateESPBtn.Text = "Teammate ESP: " .. (Flags.TeammateESP and "ON" or "OFF")
        TeammateESPBtn.BackgroundColor3 = Flags.TeammateESP and Color3.fromRGB(0, 140, 60) or Color3.fromRGB(45, 45, 50)
    else
        SectionTitle.Text = "Active Profile: " .. Flags.CurrentGame
        MiscPanel.Visible = false
        GamePanel.Visible = true
        
        local espState = GameESPStates[Flags.CurrentGame]
        GameESPBtn.Text = Flags.CurrentGame .. " ESP: " .. (espState and "ON" or "OFF")
        GameESPBtn.BackgroundColor3 = espState and Color3.fromRGB(0, 140, 60) or Color3.fromRGB(45, 45, 50)
    end
    
    for gName, btn in pairs(NavButtons) do
        if gName == Flags.CurrentGame then
            btn.BackgroundColor3 = Color3.fromRGB(60, 60, 75)
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            btn.BackgroundColor3 = Color3.fromRGB(40, 40, 45)
            btn.TextColor3 = Color3.fromRGB(170, 170, 170)
        end
    end
end

-- [[ Global Fullbright Engine ]] --
local function UpdateFullbright()
    if Flags.Fullbright then
        Lighting.Ambient = Color3.fromRGB(255, 255, 255)
        Lighting.OutdoorAmbient = Color3.fromRGB(255, 255, 255)
        Lighting.ClockTime = 12
    else
        Lighting.Ambient = OriginalAmbient
        Lighting.OutdoorAmbient = OriginalOutdoorAmbient
        Lighting.ClockTime = OriginalClockTime
    end
end

task.spawn(function()
    while task.wait(1) do
        if Flags.Fullbright then UpdateFullbright() end
    end
end)

-- [[ Core Pipeline Filtering Engine ]] --
local function ClearHighlightFromModel(model)
    if ActiveHighlights[model] then
        ActiveHighlights[model]:Destroy()
        ActiveHighlights[model] = nil
    end
end

local function GetModelCFrame(model)
    if model:IsA("Model") then
        if model.PrimaryPart then return model.PrimaryPart.CFrame end
        local bp = model:FindFirstChildWhichIsA("BasePart")
        if bp then return bp.CFrame end
    elseif model:IsA("BasePart") then
        return model.CFrame
    end
    return nil
end

local function ProcessESPFiltering(model)
    if not model:IsA("Model") and not model:IsA("Folder") and not model:IsA("BasePart") then return end
    
    local matchedColor = nil
    local targetCleanName = model.Name
    
    -- Dynamic Teammate Registry Processor
    if Flags.TeammateESP and model:IsA("Model") then
        local p = Players:GetPlayerFromCharacter(model)
        if p and p ~= Players.LocalPlayer then
            if p.Team == Players.LocalPlayer.Team then
                matchedColor = Color3.fromRGB(255, 255, 255) -- White for Allies
            end
        end
    end
    
    -- Segment 1: FNaF 1 Profiles Processing
    if GameESPStates["FNaF 1"] and model:IsA("Model") and not matchedColor then
        if Fnaf1Configs[targetCleanName] then matchedColor = Fnaf1Configs[targetCleanName] end
    end
    
    -- Segment 2: FNaF 2 Profiles Processing
    if GameESPStates["FNaF 2"] and model:IsA("Model") and not matchedColor then
        if Fnaf2Configs[targetCleanName] then matchedColor = Fnaf2Configs[targetCleanName] end
    end
    
    -- Segment 3: FNaF 3 Real Movement Processing Track
    if GameESPStates["FNaF 3"] and not matchedColor then
        if Fnaf3TrackedThreats[model] and Fnaf3TrackedThreats[model].isReal then
            local nLower = string.lower(targetCleanName)
            matchedColor = string.find(nLower, "spring") and Color3.fromRGB(150, 140, 40) or Color3.fromRGB(0, 255, 100)
        end
    end
    
    -- Segment 4: FNaF 4 Strict NPC Target Extraction
    if GameESPStates["FNaF 4"] and not matchedColor then
        if string.find(targetCleanName, "NPC") then
            if Fnaf4Configs[targetCleanName] then matchedColor = Fnaf4Configs[targetCleanName] end
        end
    end
    
    -- Segment 5: FNaF 5 Pure Keyword Parser Check
    if GameESPStates["FNaF 5"] and not matchedColor and model:IsA("Model") then
        local nameLower = string.lower(targetCleanName)
        if string.find(nameLower, "npc") then
            matchedColor = Color3.fromRGB(255, 255, 255)
            for key, col in pairs(Fnaf5Configs) do
                if string.find(nameLower, key) then matchedColor = col break end
            end
        end
    end
    
    -- Segment 6: FNaF 6 Pizza Sim Registry Core
    if GameESPStates["FNaF 6"] and not matchedColor then
        if string.find(targetCleanName, "NPC") then
            matchedColor = Fnaf6Configs[targetCleanName]
            if not matchedColor then
                local nl = string.lower(targetCleanName)
                if string.find(nl, "baby") then matchedColor = Fnaf6Configs["ScrapBabyNPC"]
                elseif string.find(nl, "molten") or string.find(nl, "molt") then matchedColor = Fnaf6Configs["MoltenFreddyNPC"]
                elseif string.find(nl, "trap") or string.find(nl, "afton") then matchedColor = Fnaf6Configs["ScraptrapNPC"]
                elseif string.find(nl, "lefty") then matchedColor = Fnaf6Configs["LeftyNPC"]
                elseif string.find(nl, "helpy") then matchedColor = Fnaf6Configs["HelpyNPC"]
                elseif string.find(nl, "puppet") then matchedColor = Fnaf6Configs["SecurityPuppetNPC"]
                elseif string.find(nl, "cadet") then matchedColor = Fnaf6Configs["CandyCadetNPC"]
                elseif string.find(nl, "chip") then matchedColor = Fnaf6Configs["ElChipNPC"]
                elseif string.find(nl, "music") then matchedColor = Fnaf6Configs["MusicManNPC"]
                else matchedColor = Color3.fromRGB(255, 255, 255)
                end
            end
        end
    end
    
    -- Execute Tracker Attachment Engine
    if matchedColor then
        if ActiveHighlights[model] then 
            ActiveHighlights[model].FillColor = matchedColor
            return 
        end
        
        local Highlight = Instance.new("Highlight")
        Highlight.Name = "MasterSuiteActiveESP"
        Highlight.FillColor = matchedColor
        Highlight.FillTransparency = 0.55
        Highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
        Highlight.OutlineTransparency = 0
        Highlight.Adornee = model
        Highlight.Parent = model
        
        ActiveHighlights[model] = Highlight
    else
        ClearHighlightFromModel(model)
    end
end

local function RunFullWorkspaceRefresh()
    for _, obj in ipairs(Workspace:GetDescendants()) do
        ProcessESPFiltering(obj)
    end
    -- Append player character safety refresh checks
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then ProcessESPFiltering(p.Character) end
    end
end

-- Dedicated FNaF 3 Position Verification Daemon Threads
task.spawn(function()
    while true do
        task.wait(0.5)
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj:IsA("Model") or obj:IsA("BasePart") then
                if not obj:IsDescendantOf(game.Players) and obj.Name ~= "Terrain" and obj.Name ~= "Workspace" then
                    local cf = GetModelCFrame(obj)
                    if cf then
                        if not Fnaf3TrackedThreats[obj] then
                            Fnaf3TrackedThreats[obj] = { pos = cf.Position, isReal = false }
                        elseif not Fnaf3TrackedThreats[obj].isReal then
                            if (Fnaf3TrackedThreats[obj].pos - cf.Position).Magnitude > 2 then
                                Fnaf3TrackedThreats[obj].isReal = true
                                if GameESPStates["FNaF 3"] then ProcessESPFiltering(obj) end
                            end
                        end
                    end
                end
            end
        end
    end
end)

-- Fast-load runtime injector interface
Workspace.DescendantAdded:Connect(function(descendant)
    task.wait(0.4)
    ProcessESPFiltering(descendant)
end)

-- Runtime listener threads for dynamic team swaps
Players.PlayerAdded:Connect(function(player)
    player.CharacterAdded:Connect(function(char)
        task.wait(0.5)
        if Flags.TeammateESP then ProcessESPFiltering(char) end
    end)
    player:GetPropertyChangedSignal("Team"):Connect(function()
        if Flags.TeammateESP then RunFullWorkspaceRefresh() end
    end)
end)
