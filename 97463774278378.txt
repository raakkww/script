local function createToggle(parent, text, colorOn, stateKey, onToggle)
	local colorOff = Color3.fromRGB(38, 38, 48)

	local row = Instance.new("Frame", parent)
	row.Size                = UDim2.new(0.94, 0, 0, 30)
	row.BackgroundTransparency = 1

	local lbl = Instance.new("TextLabel", row)
	lbl.Size               = UDim2.new(0.75, 0, 1, 0)
	lbl.BackgroundTransparency = 1
	lbl.Text               = text
	lbl.TextColor3         = Color3.fromRGB(220, 225, 240)
	lbl.Font               = Enum.Font.Gotham
	lbl.TextSize           = 11
	lbl.TextXAlignment     = Enum.TextXAlignment.Left

	local track = Instance.new("TextButton", row)
	track.Size             = UDim2.new(0, 34, 0, 18)
	track.Position         = UDim2.new(1, -34, 0.5, -9)
	track.BackgroundColor3 = colorOff
	track.BorderSizePixel  = 0
	track.Text             = ""
	Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)

	local trackStroke = Instance.new("UIStroke", track)
	trackStroke.Color = Color3.fromRGB(60, 60, 75)

	local knob = Instance.new("Frame", track)
	knob.Size              = UDim2.new(0, 12, 0, 12)
	knob.Position         = UDim2.new(0, 3, 0.5, -6)
	knob.BackgroundColor3 = Color3.fromRGB(220, 225, 240)
	knob.BorderSizePixel  = 0
	Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)

	local function updateView(animate)
		local targetBg     = settings[stateKey] and colorOn or colorOff
		local targetPos    = settings[stateKey] and UDim2.new(1, -15, 0.5, -6) or UDim2.new(0, 3, 0.5, -6)
		local targetKnob   = settings[stateKey] and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(180, 185, 200)
		local targetStroke = settings[stateKey] and colorOn or Color3.fromRGB(60, 60, 75)

		if animate then
			TweenService:Create(track, TweenInfo.new(0.15), {BackgroundColor3 = targetBg}):Play()
			TweenService:Create(trackStroke, TweenInfo.new(0.15), {Color = targetStroke}):Play()
			knob:TweenPosition(targetPos, "Out", "Quad", 0.15, true)
			TweenService:Create(knob, TweenInfo.new(0.15), {BackgroundColor3 = targetKnob}):Play()
		else
			track.BackgroundColor3 = targetBg
			trackStroke.Color = targetStroke
			knob.Position = targetPos
			knob.BackgroundColor3 = targetKnob
		end
	end

	local function toggleState()
		settings[stateKey] = not settings[stateKey]
		updateView(true)
		if onToggle then onToggle(settings[stateKey]) end
	end

	track.MouseButton1Click:Connect(toggleState)
	updateView(false)
	return toggleState
end

local function createSlider(parent, text, stateKey, colorFill, maxVal)
	local row = Instance.new("Frame", parent)
	row.Size                = UDim2.new(0.94, 0, 0, 44)
	row.BackgroundTransparency = 1

	local lbl = Instance.new("TextLabel", row)
	lbl.Size               = UDim2.new(0.72, 0, 0, 20)
	lbl.BackgroundTransparency = 1
	lbl.Text               = text
	lbl.TextColor3         = Color3.fromRGB(180, 185, 200)
	lbl.Font               = Enum.Font.Gotham
	lbl.TextSize           = 11
	lbl.TextXAlignment     = Enum.TextXAlignment.Left

	local valLbl = Instance.new("TextLabel", row)
	valLbl.Size               = UDim2.new(1, 0, 0, 20)
	valLbl.BackgroundTransparency = 1

	local function formatVal(pct)
		if maxVal then
			return string.format("%.2f", pct * maxVal)
		else
			return math.floor(pct * 100) .. "%"
		end
	end

	valLbl.Text           = formatVal(settings[stateKey])
	valLbl.TextColor3     = colorFill
	valLbl.Font           = Enum.Font.GothamBold
	valLbl.TextSize       = 11
	valLbl.TextXAlignment = Enum.TextXAlignment.Right

	local track = Instance.new("Frame", row)
	track.Size             = UDim2.new(1, 0, 0, 4)
	track.Position         = UDim2.new(0, 0, 0, 28)
	track.BackgroundColor3 = Color3.fromRGB(38, 38, 48)
	track.BorderSizePixel  = 0
	Instance.new("UICorner", track).CornerRadius = UDim.new(1, 0)

	local fill = Instance.new("Frame", track)
	fill.Size              = UDim2.new(settings[stateKey], 0, 1, 0)
	fill.BackgroundColor3  = colorFill
	fill.BorderSizePixel   = 0
	Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

	local sliderKnob = Instance.new("Frame", track)
	sliderKnob.Size             = UDim2.new(0, 10, 0, 10)
	sliderKnob.AnchorPoint      = Vector2.new(0.5, 0.5)
	sliderKnob.Position         = UDim2.new(settings[stateKey], 0, 0.5, 0)
	sliderKnob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
	sliderKnob.BorderSizePixel  = 0
	sliderKnob.ZIndex           = 2
	Instance.new("UICorner", sliderKnob).CornerRadius = UDim.new(1, 0)

	local dragging = false

	local function update(inputPos)
		local relX            = math.clamp(inputPos.X - track.AbsolutePosition.X, 0, track.AbsoluteSize.X)
		local pct             = relX / track.AbsoluteSize.X
		settings[stateKey]    = pct
		fill.Size             = UDim2.new(pct, 0, 1, 0)
		sliderKnob.Position   = UDim2.new(pct, 0, 0.5, 0)
		valLbl.Text           = formatVal(pct)
	end

	track.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			update(input.Position)
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			update(input.Position)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then dragging = false end
	end)
end

-- ======================== POWER STAMINA SLIDER ========================
-- A special slider for speed/phasing max stamina
-- Range: 0 -> 10 mapped across 0.0 -> ~0.909 of the bar
-- The last ~9% of the bar = "infinite" (math.huge)
-- Returns a getter function so the power systems can read the live value

local INFINITE_THRESHOLD = 0.92 -- if pct >= this, it's infinite

local function createStaminaSlider(parent, text, colorFill, defaultSeconds, onChanged)
	-- defaultSeconds=5, min=0, max=10, above max=infinite
	local DISPLAY_MAX = 10
	local startPct = defaultSeconds / (DISPLAY_MAX / (1 - INFINITE_THRESHOLD)) -- roughly maps 5 to midpoint
	-- Simpler: map 0-10 linearly to 0-INFINITE_THRESHOLD, then the rest is infinite
	startPct = (defaultSeconds / DISPLAY_MAX) * (INFINITE_THRESHOLD)

	local currentPct = startPct

	local row = Instance.new("Frame", parent)
	row.Size                = UDim2.new(0.94, 0, 0, 50)
	row.BackgroundTransparency = 1

	local lbl = Instance.new("TextLabel", row)
	lbl.Size               = UDim2.new(0.65, 0, 0, 20)
	lbl.BackgroundTransparency = 1
	lbl.Text               = text
	lbl.TextColor3         = Color3.fromRGB(180, 185, 200)
	lbl.Font               = Enum.Font.Gotham
	lbl.TextSize           = 11
	lbl.TextXAlignment     = Enum.TextXAlignment.Left

	local valLbl = Instance.new("TextLabel", row)
	valLbl.Size               = UDim2.new(1, 0, 0, 20)
	valLbl.BackgroundTransparency = 1
	valLbl.Font               = Enum.Font.GothamBold
	valLbl.TextSize           = 11
	valLbl.TextXAlignment     = Enum.TextXAlignment.Right

	local function getDisplayStr(pct)
		if pct >= INFINITE_THRESHOLD then
			return "INF"
		else
			local secs = (pct / INFINITE_THRESHOLD) * DISPLAY_MAX
			return string.format("%.1fs", secs)
		end
	end

	local function getRealValue(pct)
		if pct >= INFINITE_THRESHOLD then
			return math.huge
		else
			return (pct / INFINITE_THRESHOLD) * DISPLAY_MAX
		end
	end

	valLbl.Text      = getDisplayStr(currentPct)
	valLbl.TextColor3 = currentPct >= INFINITE_THRESHOLD and Color3.fromRGB(255, 220, 50) or colorFill

	local trackBg = Instance.new("Frame", row)
	trackBg.Size             = UDim2.new(1, 0, 0, 6)
	trackBg.Position         = UDim2.new(0, 0, 0, 32)
	trackBg.BackgroundColor3 = Color3.fromRGB(38, 38, 48)
	trackBg.BorderSizePixel  = 0
	Instance.new("UICorner", trackBg).CornerRadius = UDim.new(1, 0)

	-- Infinite zone overlay
	local infZone = Instance.new("Frame", trackBg)
	infZone.Size             = UDim2.new(1 - INFINITE_THRESHOLD, 0, 1, 0)
	infZone.Position         = UDim2.new(INFINITE_THRESHOLD, 0, 0, 0)
	infZone.BackgroundColor3 = Color3.fromRGB(255, 200, 30)
	infZone.BackgroundTransparency = 0.7
	infZone.BorderSizePixel  = 0
	Instance.new("UICorner", infZone).CornerRadius = UDim.new(1, 0)

	local fill = Instance.new("Frame", trackBg)
	fill.Size              = UDim2.new(currentPct, 0, 1, 0)
	fill.BackgroundColor3  = colorFill
	fill.BorderSizePixel   = 0
	fill.ZIndex            = 2

local function getLeastPopulatedCFrame(targets, offset)
	if #targets == 0 then return nil end
	local best, minCount = nil, math.huge
	for _, target in pairs(targets) do
		local root = target:IsA("Model") and target.PrimaryPart or target:FindFirstChildWhichIsA("BasePart")
		if root then
			local n = 0
			for _, p in pairs(Players:GetPlayers()) do
				if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
					if (p.Character.HumanoidRootPart.Position - root.Position).Magnitude < 30 then n += 1 end
				end
			end
			if n < minCount then minCount = n; best = root end
		end
	end
	return best and best.CFrame * offset or nil
end

local function teleportToSafestCrate()
	local crates = {}
	for _, mapName in pairs({"Map1","Map2"}) do
		local map = workspace:FindFirstChild(mapName)
		if map then
			local spawns = map:FindFirstChild("CrateSpawns")
			if spawns then
				for _, c in pairs(spawns:GetChildren()) do
					if c.Name == "VoughtCrate" then table.insert(crates, c) end
				end
			end
		end
	end
	local cf = getLeastPopulatedCFrame(crates, CFrame.new(0, 3, 0))
	if cf and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
		LocalPlayer.Character.HumanoidRootPart.CFrame = cf
	else
		showNotification("No crate found", true)
	end
end

local function teleportToSafestItem(itemName, cleanName)
	local items = {}
	local loot  = workspace:FindFirstChild("CrateLoot")
	if loot then
		for _, it in pairs(loot:GetChildren()) do
			if it.Name == itemName then table.insert(items, it) end
		end
	end
	local cf = getLeastPopulatedCFrame(items, CFrame.new(0, 0, 0))
	if cf and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
		LocalPlayer.Character.HumanoidRootPart.CFrame = cf
	else
		showNotification(cleanName .. " not found", true)
	end
end

local ijCharacter         = nil
local ijHumanoid          = nil
local ijRootPart          = nil
local ijAnimator          = nil
local ijAirJumpTrack      = nil
local ijTrueJumpPower     = 0
local ijCharacterAddedConn = nil
local ijHeartbeatConn     = nil
local ijJumpRequestConn   = nil
local AIR_JUMP_ANIMATION  = "rbxassetid://100968433956091"

local function ijCleanupCharacter()
	if ijHeartbeatConn then ijHeartbeatConn:Disconnect(); ijHeartbeatConn = nil end
	if ijAirJumpTrack then ijAirJumpTrack:Stop(); ijAirJumpTrack = nil end
end

local function ijSetupCharacter(char)
	ijCleanupCharacter()
	ijCharacter = char
	ijHumanoid  = char:WaitForChild("Humanoid")
	ijRootPart  = char:WaitForChild("HumanoidRootPart")
	ijAnimator  = ijHumanoid:WaitForChild("Animator")

	local anim = Instance.new("Animation")
	anim.AnimationId = AIR_JUMP_ANIMATION
	ijAirJumpTrack = ijAnimator:LoadAnimation(anim)

	if ijHumanoid.UseJumpPower then
		ijTrueJumpPower = ijHumanoid.JumpPower
		ijHumanoid.JumpPower = 0
	else
		ijTrueJumpPower = math.sqrt(2 * Workspace.Gravity * ijHumanoid.JumpHeight)
		ijHumanoid.JumpHeight = 0
	end

	ijHeartbeatConn = RunService.Heartbeat:Connect(function()
		if not ijHumanoid then return end
		if ijHumanoid.UseJumpPower then
			if ijHumanoid.JumpPower ~= 0 then ijHumanoid.JumpPower = 0 end
		else
			if ijHumanoid.JumpHeight ~= 0 then ijHumanoid.JumpHeight = 0 end
		end
	end)
end

local function ijRestoreJump()
	if ijHumanoid then
		if ijHumanoid.UseJumpPower then
			ijHumanoid.JumpPower = ijTrueJumpPower
		else
			ijHumanoid.JumpHeight = (ijTrueJumpPower * ijTrueJumpPower) / (2 * Workspace.Gravity)
		end
	end
end

local function toggleInfiniteJump(state)
	if state then
		if LocalPlayer.Character then ijSetupCharacter(LocalPlayer.Character) end
		ijCharacterAddedConn = LocalPlayer.CharacterAdded:Connect(ijSetupCharacter)
		ijJumpRequestConn = UserInputService.JumpRequest:Connect(function()
			if not ijHumanoid or not ijRootPart then return end
			if ijAirJumpTrack then ijAirJumpTrack:Play() end
			local v = ijRootPart.AssemblyLinearVelocity
			ijRootPart.AssemblyLinearVelocity = Vector3.new(v.X, ijTrueJumpPower, v.Z)
		end)
	else
		if ijJumpRequestConn then ijJumpRequestConn:Disconnect(); ijJumpRequestConn = nil end
		if ijCharacterAddedConn then ijCharacterAddedConn:Disconnect(); ijCharacterAddedConn = nil end
		ijRestoreJump()
		ijCleanupCharacter()
	end
end

local characterPartsCache = {}
local function updateNoclipCache()
	table.clear(characterPartsCache)
	local char = LocalPlayer.Character
	if char then
		for _, part in ipairs(char:GetDescendants()) do
			if part:IsA("BasePart") then
				table.insert(characterPartsCache, part)
			end
		end
	end
end

local function hookNoclipCharacter(char)
	updateNoclipCache()
	table.insert(connections, char.DescendantAdded:Connect(function(d)
		if d:IsA("BasePart") then table.insert(characterPartsCache, d) end
	end))
	table.insert(connections, char.DescendantRemoving:Connect(function(d)
		local idx = table.find(characterPartsCache, d)
		if idx then table.remove(characterPartsCache, idx) end
	end))
end

	end

local function isPlayerVillain(p)
	if p.Name == "Villain" then return true end
	if p.Team and (string.find(string.lower(p.Team.Name), "villain") or string.find(string.lower(p.Team.Name), "killer") or string.find(string.lower(p.Team.Name), "vought")) then return true end
	if p:GetAttribute("Role") == "Villain" or p:GetAttribute("IsVillain") == true then return true end
	local role = p:FindFirstChild("Role") or p:FindFirstChild("role")
	if role and (role:IsA("StringValue") or role:IsA("ValueBase")) and (string.find(string.lower(tostring(role.Value)), "villain") or string.find(string.lower(tostring(role.Value)), "killer")) then return true end
	if p.Character and p.Character:GetAttribute("IsVillain") == true then return true end
	return false
end

local function togglePhasing(state)
	for _, conn in ipairs(phasingConnections) do conn:Disconnect() end
	phasingConnections = {}

	if not state then
		isHoldingPhase = false
		isPhasing = false
		timeHeldPhase = 0
		local joint = getRootJoint(LocalPlayer.Character)
		if joint and rootJointOriginalC0 then joint.C0 = rootJointOriginalC0 end
		resetCollision(LocalPlayer.Character)
		if phasingGui then phasingGui:Destroy() phasingGui = nil end
		return
	end

	createPhasingUI()

	table.insert(phasingConnections, RunService.RenderStepped:Connect(function(dt)
		local character = LocalPlayer.Character
		if not character then return end

		if character ~= currentPhaseCharacter then
			currentPhaseCharacter = character
			rootJointOriginalC0 = nil
			isHoldingPhase = false
			isPhasing = false
			timeHeldPhase = 0
		end

		local humanoid = character:FindFirstChild("Humanoid")
		local rootPart = character:FindFirstChild("HumanoidRootPart")
		if not humanoid or not rootPart then return end

		if humanoid.Health <= 0 then
			isHoldingPhase = false
			isPhasing = false
			timeHeldPhase = 0
		end

		local maxTime = getPhaseMaxTime()

		if currentPhaseTime > maxTime then currentPhaseTime = maxTime end
		if displayPhaseTime > maxTime then displayPhaseTime = maxTime end

		if isHoldingPhase and (currentPhaseTime > 0 or maxTime == math.huge) then
			isPhasing = true
			if maxTime ~= math.huge then
				currentPhaseTime = math.max(0, currentPhaseTime - dt)
			end
			timeHeldPhase = timeHeldPhase + dt
		else
			if maxTime ~= math.huge and currentPhaseTime <= 0 then
				isHoldingPhase = false
			end
			isPhasing = false
			timeHeldPhase = 0
			if maxTime ~= math.huge then
				currentPhaseTime = math.min(maxTime, currentPhaseTime + dt)
			else
				currentPhaseTime = math.huge
			end
		end

		if maxTime == math.huge then
			displayPhaseTime = 1
			if phaseFillBar then phaseFillBar.Size = UDim2.new(1, 0, 1, 0) end
		else
			displayPhaseTime = displayPhaseTime + (currentPhaseTime - displayPhaseTime) * 15 * dt
			if phaseFillBar then phaseFillBar.Size = UDim2.new(displayPhaseTime / maxTime, 0, 1, 0) end
		end

		local joint = getRootJoint(character)
		if not joint then return end
		if not rootJointOriginalC0 then rootJointOriginalC0 = joint.C0 end

		if isPhasing then
			local progress = math.min(timeHeldPhase / 4, 1)
			local currentMaxDist = 0.5 + (progress * 1.0)
			local randomDir = Vector3.new((math.random() - 0.5) * 2, (math.random() - 0.5) * 2, (math.random() - 0.5) * 2)
			if randomDir.Magnitude > 0 then
				randomDir = randomDir.Unit
				local offset = randomDir * (math.random() * currentMaxDist)
				joint.C0 = rootJointOriginalC0 * CFrame.new(offset.X, offset.Y, offset.Z)
			end
		else
			joint.C0 = rootJointOriginalC0
		end
	end))

	table.insert(phasingConnections, RunService.Stepped:Connect(function()
		if isPhasing and LocalPlayer.Character then
			for _, part in ipairs(LocalPlayer.Character:GetDescendants()) do
				if part:IsA("BasePart") then part.CanCollide = false end
			end
		elseif not isPhasing and LocalPlayer.Character then
			resetCollision(LocalPlayer.Character)
		end
	end))
end

local isInvisible = false

local function setInvisTransparency(model)
	for _, part in ipairs(model:GetDescendants()) do
		if part:IsA("BasePart") then
			if part.Name == "HumanoidRootPart" then
				part.Transparency = 1
				part.CanCollide = true
			else
				part.Transparency = 0.7
				part.CanCollide = false
			end
		elseif part:IsA("Decal") or part:IsA("Texture") then
			part.Transparency = 0.7
		end
	end
end

local function syncInvisAnimationTrack(track)
	if not isInvisible or not invisClone then return end
	local cloneHumanoid = invisClone:FindFirstChildOfClass("Humanoid")
	if not cloneHumanoid then return end
	local cloneAnimator = cloneHumanoid:FindFirstChildOfClass("Animator")
	if not cloneAnimator then
		cloneAnimator = Instance.new("Animator")
		cloneAnimator.Parent = cloneHumanoid
	end

	local success, cloneTrack = pcall(function() return cloneAnimator:LoadAnimation(track.Animation) end)
	if success and cloneTrack then
		cloneTrack:Play()
		cloneTrack.TimePosition = track.TimePosition
		cloneTrack:AdjustSpeed(track.Speed)
		cloneTrack:AdjustWeight(track.WeightTarget, 0)
		invisActiveTracks[track] = cloneTrack

		local speedConn = track:GetPropertyChangedSignal("Speed"):Connect(function()
			if invisActiveTracks[track] then invisActiveTracks[track]:AdjustSpeed(track.Speed) end
		end)

		local stopConn
		stopConn = track.Stopped:Connect(function()
			if invisActiveTracks[track] then
				invisActiveTracks[track]:Stop()
				invisActiveTracks[track] = nil
			end
			speedConn:Disconnect()
			stopConn:Disconnect()
		end)
	end
end

local function disableInvisibility()
	isInvisible = false
	if invisRenderConnection then invisRenderConnection:Disconnect() invisRenderConnection = nil end
	if invisAnimConnection then invisAnimConnection:Disconnect() invisAnimConnection = nil end
	for track, cloneTrack in pairs(invisActiveTracks) do pcall(function() cloneTrack:Stop() end) end
	invisActiveTracks = {}

	if invisClone and LocalPlayer.Character then
		local targetPivot = invisClone:GetPivot()
		LocalPlayer.Character:PivotTo(targetPivot)
	end

	local realHumanoid = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid")
	if realHumanoid then Workspace.CurrentCamera.CameraSubject = realHumanoid end

	if invisClone then invisClone:Destroy() invisClone = nil end
	if invisPlatform then invisPlatform:Destroy() invisPlatform = nil end
end

local function triggerInvisibilityToggle()
	if isInvisible then
		disableInvisibility()
	else
		local character = LocalPlayer.Character
		if not character then return end
		local realHumanoid = character:FindFirstChild("Humanoid")
		local realHRP = character:FindFirstChild("HumanoidRootPart")
		if not realHumanoid or realHumanoid.Health <= 0 or not realHRP then return end
		isInvisible = true

		local startPivot = character:GetPivot()

		invisPlatform = Instance.new("Part")
		invisPlatform.Size = Vector3.new(20, 1, 20)
		invisPlatform.Anchored = true
		invisPlatform.Transparency = 1
		invisPlatform.CanCollide = true
		invisPlatform.Position = startPivot.Position + Vector3.new(0, 1000, 0)
		invisPlatform.Parent = workspace

		character:PivotTo(CFrame.new(invisPlatform.Position + Vector3.new(0, 3, 0)) * (startPivot - startPivot.Position))
		character.Archivable = true
		invisClone = character:Clone()
		character.Archivable = false

		for _, item in ipairs(invisClone:GetDescendants()) do
			if item:IsA("LocalScript") or item:IsA("Script") then item:Destroy() end
		end

		setInvisTransparency(invisClone)
		invisClone.Parent = workspace
		invisClone:PivotTo(startPivot)

		local cloneHumanoid = invisClone:FindFirstChildOfClass("Humanoid")
		if cloneHumanoid then
			Workspace.CurrentCamera.CameraSubject = cloneHumanoid
			for _, track in ipairs(realHumanoid:GetPlayingAnimationTracks()) do syncInvisAnimationTrack(track) end
			invisAnimConnection = realHumanoid.AnimationPlayed:Connect(syncInvisAnimationTrack)
		end

		invisRenderConnection = RunService.RenderStepped:Connect(function()
			if not isInvisible or not invisClone or not cloneHumanoid or not realHRP or not invisPlatform then return end

			local currentRotation = realHRP.CFrame - realHRP.CFrame.Position
			realHRP.CFrame = CFrame.new(invisPlatform.Position + Vector3.new(0, 3, 0)) * currentRotation

			cloneHumanoid.WalkSpeed = realHumanoid.WalkSpeed
			cloneHumanoid:Move(realHumanoid.MoveDirection, false)
			if realHumanoid.Jump then cloneHumanoid.Jump = true end
		end)
	end
end

local function handleSlideHalt()

		speedCurrentHumanoid.WalkSpeed = 0
		local hrp = speedCurrentCharacter:FindFirstChild("HumanoidRootPart")

		if hrp and slideTrack then
			slideTrack:Play()
			local duration = slideTrack.Length
			if duration == 0 then duration = 1.2 end
			local startVelocity = hrp.AssemblyLinearVelocity
			local startTime = os.clock()

			if slideHeartbeat then slideHeartbeat:Disconnect() end
			slideHeartbeat = RunService.Heartbeat:Connect(function()
				local elapsed = os.clock() - startTime
				if elapsed >= duration or not speedCurrentHumanoid or not hrp then
					if slideHeartbeat then slideHeartbeat:Disconnect() end
					if slideTrack then slideTrack:Stop() end
					isSliding = false
					if speedCurrentHumanoid then speedCurrentHumanoid.WalkSpeed = NORMAL_SPEED end
				else
					local t = elapsed / duration
					hrp.AssemblyLinearVelocity = startVelocity * (1 - t)
				end
			end)
		else
			isSliding = false
			speedCurrentHumanoid.WalkSpeed = NORMAL_SPEED
		end
	else
		isSliding = false
	end
end

local function setupSpeedCharacter(character)
	speedCurrentCharacter = character
	speedCurrentHumanoid = character:WaitForChild("Humanoid")
	speedCurrentAnimator = speedCurrentHumanoid:WaitForChild("Animator")
	runningTrack = speedCurrentAnimator:LoadAnimation(runAnimation)
	runningTrack.Priority = Enum.AnimationPriority.Movement
	slideTrack = speedCurrentAnimator:LoadAnimation(slideAnimation)
	slideTrack.Priority = Enum.AnimationPriority.Action

	table.insert(speedConnections, speedCurrentHumanoid:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
		if isSpeedSprinting and not isSliding and speedCurrentHumanoid.WalkSpeed ~= SPRINT_SPEED then
			speedCurrentHumanoid.WalkSpeed = SPRINT_SPEED
		end
	end))
	isSliding = false
	stopSprinting()
end

local function toggleSpeed(state)
	for _, conn in ipairs(speedConnections) do conn:Disconnect() end
	speedConnections = {}

	if not state then
		if slideHeartbeat then slideHeartbeat:Disconnect() slideHeartbeat = nil end
		if runningTrack then runningTrack:Destroy() runningTrack = nil end
		if slideTrack then slideTrack:Destroy() slideTrack = nil end
		if speedCurrentHumanoid then speedCurrentHumanoid.WalkSpeed = NORMAL_SPEED end
		if speedGui then speedGui:Destroy() speedGui = nil end
		isHoldingSpeed = false
		isSpeedSprinting = false
		isSliding = false
		return
	end

	local maxStam = getSpeedMaxStamina()
	currentStamina = (maxStam == math.huge) and math.huge or maxStam
	displayStamina = (maxStam == math.huge) and 1 or maxStam

	createSpeedUI()
	if LocalPlayer.Character then setupSpeedCharacter(LocalPlayer.Character) end
	table.insert(speedConnections, LocalPlayer.CharacterAdded:Connect(setupSpeedCharacter))

	table.insert(speedConnections, RunService.RenderStepped:Connect(function(dt)
		if not speedCurrentHumanoid or speedCurrentHumanoid.Health <= 0 then
			stopSprinting()
			isSliding = false
			return
		end

		local maxStam = getSpeedMaxStamina()
		local wantsToSprint = isHoldingSpeed and (currentStamina > 0 or maxStam == math.huge) and not isSliding

		if wantsToSprint then
			if not isSpeedSprinting then
				isSpeedSprinting = true
				speedCurrentHumanoid.WalkSpeed = SPRINT_SPEED
				if runningTrack then
					runningTrack:Play()
					runningTrack:AdjustSpeed(2)
				end
			end
			if speedCurrentHumanoid.WalkSpeed ~= SPRINT_SPEED then speedCurrentHumanoid.WalkSpeed = SPRINT_SPEED end
			speedCurrentHumanoid:Move(Vector3.new(0, 0, -1), true)
			if maxStam ~= math.huge then
				currentStamina = math.max(0, currentStamina - dt)
				if currentStamina <= 0 then handleSlideHalt() end
			end
		else
			if isSpeedSprinting then stopSprinting() end
			if isSliding then
				speedCurrentHumanoid.WalkSpeed = 0
			else
				if maxStam ~= math.huge then
					currentStamina = math.min(maxStam, currentStamina + dt)
				else
					currentStamina = math.huge
				end
			end
		end

		if maxStam == math.huge then
			displayStamina = 1
			if speedFillBar then speedFillBar.Size = UDim2.new(1, 0, 1, 0) end
		else
			displayStamina = displayStamina + (currentStamina - displayStamina) * 15 * dt
			if speedFillBar then speedFillBar.Size = UDim2.new(displayStamina / maxStam, 0, 1, 0) end
		end
	end))
end

-- ---------------- FLIGHT ----------------

local isFlying = false
local isFlightSprinting = false
local isTakingOff = false
local takeoffEndTime = 0
local normalAnim = Instance.new("Animation"); normalAnim.AnimationId = "rbxassetid://101246078472778"
local sprintAnim = Instance.new("Animation"); sprintAnim.AnimationId = "rbxassetid://84554134427206"
local flightNormalTrack = nil
local flightSprintTrack = nil
local flightAttachment = nil
local linearVelocity = nil
local alignOrientation = nil
local flightRenderConn = nil

local function stopFlight()
	if not isFlying then return end
	isFlying = false
	isFlightSprinting = false
	isTakingOff = false
	if flightNormalTrack then flightNormalTrack:Stop() end
	if flightSprintTrack then flightSprintTrack:Stop() end
	if flightAttachment then flightAttachment:Destroy() flightAttachment = nil end
	if linearVelocity then linearVelocity:Destroy() linearVelocity = nil end
	if alignOrientation then alignOrientation:Destroy() alignOrientation = nil end
	if flightRenderConn then flightRenderConn:Disconnect() flightRenderConn = nil end

	local char = LocalPlayer.Character
	if char then
		local hum = char:FindFirstChild("Humanoid")
		if hum then
			hum.PlatformStand = false
			hum:ChangeState(Enum.HumanoidStateType.GettingUp)
		end
	end
end

local function startFlight()
	local char = LocalPlayer.Character
	if not char then return end
	local hum = char:FindFirstChild("Humanoid")
	local hrp = char:FindFirstChild("HumanoidRootPart")
	if not hum or not hrp then return end

	isFlying = true
	isTakingOff = true
	takeoffEndTime = os.clock() + 0.35
	hum.PlatformStand = true

	local animator = hum:FindFirstChild("Animator") or hum
	flightNormalTrack = animator:LoadAnimation(normalAnim)
	flightNormalTrack.Priority = Enum.AnimationPriority.Action4
	flightSprintTrack = animator:LoadAnimation(sprintAnim)
	flightSprintTrack.Priority = Enum.AnimationPriority.Action4

	flightAttachment = Instance.new("Attachment")
	flightAttachment.Parent = hrp

	linearVelocity = Instance.new("LinearVelocity")
	linearVelocity.Attachment0 = flightAttachment
	linearVelocity.MaxForce = math.huge
	linearVelocity.RelativeTo = Enum.ActuatorRelativeTo.World
	linearVelocity.Parent = hrp

	alignOrientation = Instance.new("AlignOrientation")
	alignOrientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
	alignOrientation.Attachment0 = flightAttachment
	alignOrientation.MaxTorque = math.huge
	alignOrientation.Responsiveness = 200
	alignOrientation.Parent = hrp

	flightRenderConn = RunService.RenderStepped:Connect(function(dt)
		if not isFlying or hum.Health <= 0 then
			stopFlight()
			return
		end

		hum.PlatformStand = true

		if isTakingOff and os.clock() >= takeoffEndTime then isTakingOff = false end

		if not isTakingOff then
			local rayOrigin = hrp.Position
			local rayDir = Vector3.new(0, -(hum.HipHeight + (hrp.Size.Y / 2) + 0.5), 0)
			local params = RaycastParams.new()
			params.FilterDescendantsInstances = {char}
			params.FilterType = Enum.RaycastFilterType.Exclude

			local hit = Workspace:Raycast(rayOrigin, rayDir, params)
			if hit and not UserInputService:IsKeyDown(Enum.KeyCode.Space) then
				stopFlight()
				return
			end
		end

		local camLook  = Workspace.CurrentCamera.CFrame.LookVector
		local camRight = Workspace.CurrentCamera.CFrame.RightVector

		if isFlightSprinting then
			if not flightSprintTrack.IsPlaying then flightSprintTrack:Play() end
			if flightNormalTrack.IsPlaying then flightNormalTrack:Stop() end
			alignOrientation.CFrame = Workspace.CurrentCamera.CFrame
			linearVelocity.VectorVelocity = camLook * 150
		else
			if not flightNormalTrack.IsPlaying then flightNormalTrack:Play() end
			if flightSprintTrack.IsPlaying then flightSprintTrack:Stop() end

			local flatLook = Vector3.new(camLook.X, 0, camLook.Z)
			if flatLook.Magnitude > 0.001 then flatLook = flatLook.Unit else flatLook = Vector3.new(0, 0, -1) end
			local flatRight = Vector3.new(camRight.X, 0, camRight.Z)
			if flatRight.Magnitude > 0.001 then flatRight = flatRight.Unit end

			alignOrientation.CFrame = CFrame.lookAt(Vector3.zero, flatLook)
			local moveVector = Vector3.zero
			if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVector = moveVector + flatLook end
			if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVector = moveVector - flatLook end
			if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVector = moveVector + flatRight end
			if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVector = moveVector - flatRight end

			if moveVector.Magnitude > 0.001 then moveVector = moveVector.Unit end

			local yVel = 0
			if isTakingOff then
				yVel = 11.5
			elseif UserInputService:IsKeyDown(Enum.KeyCode.Space) then
				yVel = 30
			elseif UserInputService:IsKeyDown(Enum.KeyCode.C) then
				yVel = -30
			else
				yVel = math.cos(os.clock() * 4) * 2
			end

			linearVelocity.VectorVelocity = (moveVector * 30) + Vector3.new(0, yVel, 0)
		end
	end)
end

local function toggleFlightPower(state)
	if not state and isFlying then stopFlight() end
end

-- ---------------- TELEPORTATION POWER ----------------

local isTeleporting = false
local TRANSITION_TIME = 0.25

local function smoothstep(x)
	return x * x * (3 - 2 * x)
end

local teleportConnections = {}

local function toggleTeleportPower(state)
	for _, conn in ipairs(teleportConnections) do conn:Disconnect() end
	teleportConnections = {}
	if not state then return end

	table.insert(teleportConnections, UserInputService.InputBegan:Connect(function(input, gameProcessed)
		if gameProcessed or isTeleporting then return end
		if not settings.powerTeleport then return end

		if input.KeyCode == keybinds.powerTeleport then
			local character = LocalPlayer.Character
			if not character or not character:FindFirstChild("HumanoidRootPart") then return end

			local mouse    = LocalPlayer:GetMouse()
			local rootPart = character.HumanoidRootPart
			local humanoid = character:FindFirstChildOfClass("Humanoid")
			local targetPos = mouse.Hit.Position

			local startPos = rootPart.Position
			local distance = (targetPos - startPos).Magnitude
			if distance < 10 then return end

			isTeleporting = true

			local bodyParts = {}
			for _, part in ipairs(character:GetDescendants()) do
				if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
					table.insert(bodyParts, part)
				end
			end

			local originalOffsets = {}
			for _, part in ipairs(bodyParts) do
				originalOffsets[part] = rootPart.CFrame:ToObjectSpace(part.CFrame)
			end

			local motors = {}
			for _, motor in ipairs(character:GetDescendants()) do
				if motor:IsA("Motor6D") then
					table.insert(motors, motor)
					motor.Enabled = false
				end
			end

			for _, part in ipairs(bodyParts) do part.Anchored = true end
			rootPart.Anchored = true
			if humanoid then humanoid:ChangeState(Enum.HumanoidStateType.Physics) end

			local startCFs = {}
			for _, part in ipairs(bodyParts) do startCFs[part] = part.CFrame end
			local t0 = tick()
			local centerCF = rootPart.CFrame

			while tick() - t0 < TRANSITION_TIME do
				local alpha = smoothstep(math.clamp((tick() - t0) / TRANSITION_TIME, 0, 1))
				for _, part in ipairs(bodyParts) do
					part.CFrame = startCFs[part]:Lerp(centerCF, alpha)
				end
				RunService.RenderStepped:Wait()
			end

			local finalDir = (targetPos - startPos).Unit
			local flatDir  = Vector3.new(finalDir.X, 0, finalDir.Z)
			if flatDir.Magnitude < 0.001 then flatDir = Vector3.new(0, 0, -1) else flatDir = flatDir.Unit end
			local finalRootCF = CFrame.lookAt(targetPos + Vector3.new(0, 4, 0), targetPos + Vector3.new(0, 4, 0) + flatDir)

			rootPart.CFrame = finalRootCF
			for _, part in ipairs(bodyParts) do part.CFrame = finalRootCF end

			local t1 = tick()
			while tick() - t1 < TRANSITION_TIME do
				local alpha = smoothstep(math.clamp((tick() - t1) / TRANSITION_TIME, 0, 1))
				for _, part in ipairs(bodyParts) do
					local targetCF = finalRootCF * originalOffsets[part]
					part.CFrame = finalRootCF:Lerp(targetCF, alpha)
				end
				RunService.RenderStepped:Wait()
			end

			for _, part in ipairs(bodyParts) do part.CFrame = finalRootCF * originalOffsets[part] end

			for _, motor in ipairs(motors) do motor.Enabled = true end
			for _, part in ipairs(bodyParts) do part.Anchored = false end
			rootPart.Anchored = false

			rootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
			rootPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

			if humanoid then humanoid:ChangeState(Enum.HumanoidStateType.GettingUp) end

			isTeleporting = false
		end
	end))
end

-- ======================== FULL CLEANUP FUNCTION ========================

local function fullCleanup()
	-- Disconnect all global connections
	for _, conn in pairs(connections) do
		pcall(function() conn:Disconnect() end)
	end

	-- Clean up all ESP instances
	for inst in pairs(espInstances) do
		cleanupESP(inst)
	end

	-- Disable infinite jump and restore
	if settings.infiniteJump then
		settings.infiniteJump = false
		ijRestoreJump()
		ijCleanupCharacter()
		if ijJumpRequestConn then ijJumpRequestConn:Disconnect() ijJumpRequestConn = nil end
		if ijCharacterAddedConn then ijCharacterAddedConn:Disconnect() ijCharacterAddedConn = nil end
	end

	-- Disable TP Walk
	settings.tpWalk = false

	-- Remove TP Tool
	removeTpTool()

	-- Disable prox prompt
	disableProxPrompt()

	-- Disable phasing
	if settings.powerPhasing then
		settings.powerPhasing = false
		for _, conn in ipairs(phasingConnections) do pcall(function() conn:Disconnect() end) end
		phasingConnections = {}
		isHoldingPhase = false
		isPhasing = false
		local joint = getRootJoint(LocalPlayer.Character)
		if joint and rootJointOriginalC0 then joint.C0 = rootJointOriginalC0 end
		resetCollision(LocalPlayer.Character)
		if phasingGui then phasingGui:Destroy() phasingGui = nil end
	end

	-- Disable invisibility
	if isInvisible then
		disableInvisibility()
	end
	-- Also destroy any leftover clone/platform
	if invisClone then pcall(function() invisClone:Destroy() end) invisClone = nil end
	if invisPlatform then pcall(function() invisPlatform:Destroy() end) invisPlatform = nil end

	-- Disable superspeed
	if settings.powerSpeed then
		settings.powerSpeed = false
		for _, conn in ipairs(speedConnections) do pcall(function() conn:Disconnect() end) end
		speedConnections = {}
		if slideHeartbeat then slideHeartbeat:Disconnect() slideHeartbeat = nil end
		if runningTrack then pcall(function() runningTrack:Destroy() end) runningTrack = nil end
		if slideTrack then pcall(function() slideTrack:Destroy() end) slideTrack = nil end
		if speedCurrentHumanoid then speedCurrentHumanoid.WalkSpeed = NORMAL_SPEED end
		if speedGui then speedGui:Destroy() speedGui = nil end
		isHoldingSpeed = false
		isSpeedSprinting = false
		isSliding = false
	end

	-- Disable flight
	if isFlying then stopFlight() end

	-- Disable teleport power
	for _, conn in ipairs(teleportConnections) do pcall(function() conn:Disconnect() end) end
	teleportConnections = {}

	-- Destroy main GUIs
	if activeNotif then pcall(function() activeNotif:Destroy() end) activeNotif = nil end
	gui:Destroy()
	script:Destroy()
end
