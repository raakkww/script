-- ESP Logic System
local espData = { survivors = {}, killers = {}, generators = {}, batteries = {} }

local function newHighlight(obj, fillColor, outlineColor)
    if not obj then return end
    local h = Instance.new("Highlight")
    h.FillColor = fillColor
    h.FillTransparency = 0.5
    h.OutlineColor = outlineColor or fillColor
    h.OutlineTransparency = 0
    h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    h.Adornee = obj
    h.Parent = obj
    return h
end

local function addESP(tbl, obj, fillColor, outlineColor)
    if obj and not tbl[obj] then tbl[obj] = newHighlight(obj, fillColor, outlineColor) end
end
local function removeESP(tbl, obj)
    if tbl[obj] then tbl[obj]:Destroy() tbl[obj] = nil end
end
local function clearESP(tbl)
    for obj, h in pairs(tbl) do if h then h:Destroy() end tbl[obj] = nil end
end

local function getSurvivorColor(char)
    if char:FindFirstChild("Medkit") or char:GetAttribute("Role") == "Medic" then
        return Color3.fromRGB(255, 255, 255) -- White
    elseif char:FindFirstChild("Axe") or char:GetAttribute("Role") == "Fighter" then
        return Color3.fromRGB(128, 0, 128) -- Purple
    else
        return Color3.fromRGB(0, 255, 0) -- Green
    end
end

VisualsTab:New("Toggle")({
    Title = "Highlight all survivors",
    Description = "Draws an outline around survivors. (Green = Normal, White = Medic, Purple = Fighter)",
    DefaultValue = false,
    Callback = function(Value)
        local aliveFolder = workspace:FindFirstChild("PLAYERS") and workspace.PLAYERS:FindFirstChild("ALIVE")
        if Value and aliveFolder then
            for _, v in ipairs(aliveFolder:GetChildren()) do 
                if v:IsA("Model") then addESP(espData.survivors, v, getSurvivorColor(v)) end 
            end
            espData.survivorAdd = aliveFolder.ChildAdded:Connect(function(v) 
                if v:IsA("Model") then addESP(espData.survivors, v, getSurvivorColor(v)) end 
            end)
            espData.survivorRemove = aliveFolder.ChildRemoved:Connect(function(v) removeESP(espData.survivors, v) end)
            
            espData.survivorUpdate = RunService.Heartbeat:Connect(function()
                for char, highlight in pairs(espData.survivors) do
                    if char and highlight then highlight.FillColor = getSurvivorColor(char) highlight.OutlineColor = getSurvivorColor(char) end
                end
            end)
        else
            if espData.survivorAdd then espData.survivorAdd:Disconnect() end
            if espData.survivorRemove then espData.survivorRemove:Disconnect() end
            if espData.survivorUpdate then espData.survivorUpdate:Disconnect() end
            clearESP(espData.survivors)
        end
    end,
})

                end
                
                if closest and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) then
                    root.CFrame = CFrame.new(root.Position, Vector3.new(closest.HumanoidRootPart.Position.X, root.Position.Y, closest.HumanoidRootPart.Position.Z))
                end
            end)
        else
            if silentAimConn then silentAimConn:Disconnect() silentAimConn = nil end
        end
    end,
})

local TeleportsTab = Window:AddTab("Teleports")
TeleportsTab:New("Title")({ Title = "Quick Travel" })

TeleportsTab:New("Button")({
    Title = "Teleport to FuseBox",
    Description = "Instantly teleports you to the map's fuse box.",
    Callback = function()
        local char = LocalPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        
        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") and (v.Name == "FuseBox" or v.Name == "Fuse Box") then
                root.CFrame = v.CFrame + Vector3.new(0, 3, 0)
                Window:Notify({ Title = "Teleported", Description = "Teleported to the FuseBox.", Duration = 2, Type = "Success" })
                return
            end
        end
        Window:Notify({ Title = "Failed", Description = "FuseBox not found on this map.", Duration = 2, Type = "Error" })
    end,
})

local batteryIndex = 1
TeleportsTab:New("Button")({
    Title = "Warp to Battery",
    Description = "Clicking this cycles your position to the next battery spawn on the map.",
    Callback = function()
        local function getBatteries()
            local bats = {}
            for _, v in ipairs(workspace:GetDescendants()) do
                if v:IsA("MeshPart") and v.Name == "Battery" then table.insert(bats, v) end
            end
            return bats
        end
        local batteries = getBatteries()
        if #batteries > 0 then
            local battery = batteries[batteryIndex]
            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                LocalPlayer.Character.HumanoidRootPart.CFrame = battery.CFrame + Vector3.new(0, 3, 0)
            end
            batteryIndex = (batteryIndex % #batteries) + 1
        end
    end,
})
