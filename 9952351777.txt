local function normalizeRank(name)
   if not name or name == "" then return "" end
   return name:lower():gsub("%s+", "")
end

local allRarities = {"All", "Normal", "Gold", "Platinum", "Emerald", "Diamond", "Rainbow"}

local rankList = {
   "Princess", "Queen", "Matriarch", "Empress", "Seraph",
   "Divine", "Celestial", "Goddess", "Valkyrie", "Archon",
   "Eternal", "Radiant", "Rising Star",
   "Ascendant", "Zenith", "Beyond"
}

local normalOnlyRanks = {
   ascendant = true,
   zenith = true,
   beyond = true,
}

SaveFile.PackRarities = {}
for _, rank in ipairs(rankList) do
   local key = normalizeRank(rank)
   SaveFile.PackRarities[key] = {
      all = false, normal = false, gold = false,
      platinum = false, emerald = false, diamond = false, rainbow = false,
   }
   if normalOnlyRanks[key] then
      SaveFile.PackRarities[key]["normal"] = true
   else
      SaveFile.PackRarities[key]["rainbow"] = true
   end
end

SaveFile.SelectedRanks = {}
for _, rank in ipairs(rankList) do
   SaveFile.SelectedRanks[normalizeRank(rank)] = false
end

local function getAPS()
   local myPlot = Workspace.Map.PlotModels:FindFirstChild(Player.Name)
   if not myPlot then return nil end
   local systems = myPlot:FindFirstChild("Systems")
   if not systems then return nil end
   return systems:FindFirstChild("ActivePackStation")
end

local function getPlayerPlot()
   local plotFolder = Workspace:WaitForChild("Map"):WaitForChild("PlotModels")
   local userPlot = plotFolder:FindFirstChild(Player.Name)
   if userPlot then
      return userPlot:FindFirstChild("Base") or userPlot:FindFirstChildWhichIsA("BasePart")
   end
   return nil
end

local function collectToken(tokenPart)
   if not (tokenPart and tokenPart.Parent) then return end
   local character = Player.Character or Player.CharacterAdded:Wait()
   local humanoid = character:WaitForChild("Humanoid")
   local hrp = character:WaitForChild("HumanoidRootPart")

   local attempts = 0
   while tokenPart and tokenPart.Parent and attempts < 10 do
      attempts += 1
      hrp.CFrame = CFrame.new(tokenPart.Position + Vector3.new(0, 5, 5))
      task.wait(0.3)
      humanoid:MoveTo(tokenPart.Position)
      local timeout = 3
      local elapsed = 0
      while tokenPart and tokenPart.Parent and elapsed < timeout do
         task.wait(0.1)
         elapsed += 0.1
      end
      if not (tokenPart and tokenPart.Parent) then
         print("Collected!")
         return
      end
   end
   print("Gave up on token")
end

local function collectAllTokens()
   local character = Player.Character or Player.CharacterAdded:Wait()
   local hrp = character:WaitForChild("HumanoidRootPart")
   local originalCFrame = hrp.CFrame
   local systems = Workspace:FindFirstChild("Systems")
   if not systems then return end

   -- Ground tokens
   local groundTokens = systems:FindFirstChild("GradeTokens")
   if groundTokens then
      for _, token in ipairs(groundTokens:GetChildren()) do
         local tokenPart = token:FindFirstChild("GradeTokenPart")
         if tokenPart and tokenPart.Parent then
            print("Collecting ground token: " .. token.Name)
            collectToken(tokenPart)
         end
      end
   end

   -- Obby tokens Easy/Medium/Hard
   local obbyTokens = systems:FindFirstChild("GradeObbyTokens")
   if obbyTokens then
      for _, diffName in ipairs({"Easy", "Medium", "Hard"}) do
         local diffFolder = obbyTokens:FindFirstChild(diffName)
         if diffFolder then
            for _, slot in ipairs(diffFolder:GetChildren()) do
               local tokenPart = slot:FindFirstChild("GradeTokenPart")
               if tokenPart and tokenPart.Parent then
                  print("Collecting " .. diffName .. " obby token: " .. slot.Name)
                  collectToken(tokenPart)
               end
            end
         end
      end
   end

   hrp.CFrame = originalCFrame
   print("All tokens done!")
   Rayfield:Notify({
      Title = "Tokens",
      Content = "All tokens collected!",
      Duration = 3,
      Image = 4483362458,
   })
end

AutoBuy:CreateToggle({

   Callback = function(Value)
      SaveFile.AutoBuyPacks = Value
      if SaveFile.AutoBuyPacks then
         task.spawn(function()
            local function findAndBuy(obj)
               for _, v in ipairs(obj:GetChildren()) do
                  if v.Name:find("ActivePack") then
                     local frame = v:FindFirstChild("PackPriceBillboardGui")
                        and v.PackPriceBillboardGui:FindFirstChild("Frame")
                     local nameLabel = frame and frame:FindFirstChild("NameLabel")
                     local variantLabel = frame and frame:FindFirstChild("VariantLabel")
                     local itemRank = normalizeRank(nameLabel and nameLabel.Text or "")
                     local itemRarity = normalizeRank(
                        (variantLabel and variantLabel.Text ~= "" and variantLabel.Text) or "normal"
                     )
                     local packNum = tonumber(obj.Name:match("%d+"))
                     if itemRank == "" then continue end
                     if not packNum then continue end
                     if not SaveFile.SelectedRanks[itemRank] then continue end
                     local packRarities = SaveFile.PackRarities[itemRank]
                     if not packRarities then continue end
                     if packRarities["all"] or packRarities[itemRarity] then
                        print("BUYING slot "..packNum.." | "..itemRank.." | "..itemRarity)
                        PackPurchase:FireServer(packNum)
                        task.wait(0.8)
                     end
                  end
                  findAndBuy(v)
               end
            end

            while SaveFile.AutoBuyPacks do
               pcall(function()
                  local aps = getAPS()
                  if not aps then return end
                  findAndBuy(aps)
               end)
               task.wait(1)
            end
         end)
      end
   end,
})

local function getInventoryFromUI()
   local hotbar = Player.PlayerGui:FindFirstChild("HotbarGui")
   local mainFrame = hotbar and hotbar:FindFirstChild("MainFrame")
   local items = {}
   if not mainFrame then return items end
   for _, obj in pairs(mainFrame:GetChildren()) do
      if obj:IsA("TextButton") and obj.Visible then
         local amountLabel = obj:FindFirstChild("AmountLabel")
         local amount = 1
         if amountLabel then
            amount = tonumber(amountLabel.Text:gsub("%D", "")) or 1
         end
         local splitName = string.split(obj.Name, "_")
         if splitName[1] and splitName[2] then
            local pType = "_" .. splitName[2]
            if pType == "_Normal" then pType = "Normal" end
            table.insert(items, {Rank = splitName[1], Type = pType, Quantity = amount})
         end
      end
   end
   return items
end

local function equipAndPlace(rank, pType, position)
   local character = Player.Character or Player.CharacterAdded:Wait()
   local humanoid = character:WaitForChild("Humanoid")
   local hrp = character:WaitForChild("HumanoidRootPart")
   hrp.CFrame = CFrame.new(position + Vector3.new(0, 5, 0))
   task.wait(0.1)
   humanoid:UnequipTools()
   EquipPackTool:FireServer(rank, pType)
   local tool = nil
   local start = tick()
   while not tool and (tick() - start) < 1.2 do
      tool = character:FindFirstChildOfClass("Tool") or Player.Backpack:FindFirstChildWhichIsA("Tool")
      task.wait(0.05)
   end
   if tool then
      if tool.Parent == Player.Backpack then humanoid:EquipTool(tool) end
      tool:Activate()
      local placementRemote = ReplicatedStorage.Remotes:FindFirstChild("PlaceItem")
      if placementRemote then placementRemote:FireServer(tool.Name, position) end
   end
end

AutoPlace:CreateToggle({
   Name = "Auto Place Packs",
   CurrentValue = false,
   Flag = "AutoPlaceUI",
   Callback = function(Value)
      SaveFile.AutoPlace = Value
      if SaveFile.AutoPlace then
         task.spawn(function()
            local plotBase = getPlayerPlot()
            if not plotBase then return end
            local size = plotBase.Size
            local margin = 4
            local minX, maxX = -(size.X/2)+margin, (size.X/2)-margin
            local minZ, maxZ = -(size.Z/2)+margin, (size.Z/2)-margin
            while SaveFile.AutoPlace do
               local visibleItems = getInventoryFromUI()
               if #visibleItems == 0 then task.wait(2) continue end
               for _, data in pairs(visibleItems) do
                  for i = 1, data.Quantity do
                     if not SaveFile.AutoPlace then break end
                     local targetPos = (plotBase.CFrame * CFrame.new(
                        math.random(minX, maxX),
                        (size.Y/2)+0.5,
                        math.random(minZ, maxZ)
                     )).Position
                     equipAndPlace(data.Rank, data.Type, targetPos)
                     task.wait(0.5)
                  end
                  if not SaveFile.AutoPlace then break end
               end
               task.wait(1)
            end
         end)
      end
   end,
})
