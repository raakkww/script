local itemsList = {}
local displayToName = {}
local tempNames = {}

for name, _ in pairs(itemData) do
    table.insert(tempNames, name)
end
table.sort(tempNames)

for _, name in ipairs(tempNames) do
    local rarity = itemData[name]
    local displayText = "[" .. rarity .. "] " .. name
    table.insert(itemsList, displayText)
    displayToName[displayText] = name
end

local mutationsList = {"Normal", "Golden", "Diamond", "Magma", "Disco", "Ruby", "Neon"}

local selectedDisplay = itemsList[1]
local selectedMutation = "Normal"

local function getZonePosition(rarityName)
    local targetPosition = Vector3.new(-29.859832763672, 4.5068635940552, -248.39669799805)
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("BasePart") and v.Name:lower():find(rarityName:lower()) then
            targetPosition = v.Position
            break
        end
    end
    return targetPosition
end

local function fireFarmEvent(itemName, mutation)
    local player = game.Players.LocalPlayer
    if not player or not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
    
    local storage = game:GetService("ReplicatedStorage")
    local remotes = storage:WaitForChild("ThrowLuckyBlockRemotes", 5)
    if not remotes then return end
    
    local ThrowStarted = remotes:WaitForChild("ThrowStarted", 5)
    local LuckyBlockLanded = remotes:WaitForChild("LuckyBlockLanded", 5)
    if not ThrowStarted or not LuckyBlockLanded then return end
    
    local rarity = itemData[itemName] or "Common"
    local blockName = blockMapping[rarity] or "Lucky Block"
    local calculatedPosition = getZonePosition(rarity)
    
    player.Character.HumanoidRootPart.CFrame = CFrame.new(16.4, 18.9, -499.9)
    task.wait(0.05)
    
    ThrowStarted:FireServer()
    LuckyBlockLanded:FireServer({
        LandingPosition = calculatedPosition,
        ItemName = itemName,
        Rarity = rarity,
        BlockName = blockName,
        LandingRarity = rarity,
        Mutation = mutation,
        Power = 7.6680453638405
    })
    
    task.wait(0.1)
    if player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
        player.Character.HumanoidRootPart.CFrame = CFrame.new(16.4, 18.9, -499.9)
    end
end

FarmTab:Button({

    Callback = function()
