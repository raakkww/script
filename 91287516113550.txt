-- Function to validate number input
local function validateNumber(input)
    local num = tonumber(input)
    if not num or num < 0 then
        return 0
    end
    return math.floor(num)
end

-- Coins button functionality
coinsButton.MouseButton1Click:Connect(function()
    local amount = validateNumber(coinsInput.Text)
    if amount > 99999 then
        amount = 99999
        coinsInput.Text = "99999"
    end
    local remote = game:GetService("ReplicatedStorage"):WaitForChild("GiveCoinsEvent")
    remote:FireServer(amount)
end)

-- Wood button functionality
woodButton.MouseButton1Click:Connect(function()
    local amount = validateNumber(woodInput.Text)
    game:GetService("ReplicatedStorage").Remoteevents.WoodAddValueEvent:FireServer(amount)
end)

-- Enter key functionality
coinsInput.FocusLost:Connect(function(enterPressed)
    if enterPressed then
        local amount = validateNumber(coinsInput.Text)
        if amount > 99999 then
            coinsInput.Text = "99999"
        elseif amount < 0 then
            coinsInput.Text = "0"
        end
    end
end)

woodInput.FocusLost:Connect(function(enterPressed)
    if enterPressed then
        local amount = validateNumber(woodInput.Text)
        if amount < 0 then
            woodInput.Text = "0"
        end
    end
end)
