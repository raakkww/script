local Tab = Window:CreateTab("GAMEK1D", 4483362458)

local Section = Tab:CreateSection("Anti Game over when you lose all heart at gamek1d (use it on last heart)")

_G.AntiGameOver = false -- สร้างตัวแปร Global ไว้ควบคุมการเปิด-ปิด Loop

local Toggle = Tab:CreateToggle({
   Name = "Anti Game over at last heart",
   CurrentValue = false,
   Flag = "AntiGameOverToggle", 
   Callback = function(Value)
       _G.AntiGameOver = Value -- อัปเดตสถานะตามการกดเปิด/ปิด Toggle
       
       if _G.AntiGameOver then
           -- ใช้ task.spawn เพื่อให้ระบบวนลูปทำงานในพื้นหลัง ไม่ทำให้ UI บนมือถือค้าง
           task.spawn(function()
               while _G.AntiGameOver do
                   -- ตรวจสอบก่อนว่ามี FunGui อยู่จริงไหม เพื่อป้องกันสคริปต์ Error บนมือถือ
                   local funGui = game:GetService("Players").LocalPlayer.PlayerGui:FindFirstChild("FunGui")
                   if funGui and funGui:FindFirstChild("SuccessEvent") then
                       funGui.SuccessEvent:FireServer(true)
                   end
                   
                   -- คอยส่งสัญญาณเช็กทุกๆ 0.5 วินาที (ปลอดภัยต่อเครื่องและไม่หนักเซิร์ฟเวอร์)
                   task.wait(0.5) 
               end
           end)
       end
   end,
})

local Tab = Window:CreateTab("Others", 4483362458)

_G.BlockRemoteOnly = false 

-- 🛠️ อัปเกรดระบบเพื่อ Delta 2.0: ใช้ checkcaller แยกแยะสัญญาณเกมกับสัญญาณสคริปต์
local oldNamecall
if not oldNamecall then
    oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
        local method = getnamecallmethod()
        
        -- ทำงานเมื่อเปิดปุ่ม และเป็นคำสั่ง FireServer
        if _G.BlockRemoteOnly and method == "FireServer" then
            -- 🔒 เช็กว่าถ้ารีโมทชื่อ RemoteEvent นี้ ถูกส่งมาจากสคริปต์ภายนอก (ไม่ใช่ตัวเกมหลักสั่งเอง)
            if self.Name == "RemoteEvent" and checkcaller() then
                -- และถ้ามันพยายามเข้าถึงหรือส่งข้อมูลที่เกี่ยวกับ ChustUI ให้ทำการบล็อกทันที
                if self.Parent and self.Parent.Name == "ChustUI" then
                    return nil 
                end
            end
        end
        
        return oldNamecall(self, ...)
    end)
end

local Toggle = Tab:CreateToggle({
   Name = "Anti Depriver",
   CurrentValue = false,
   Flag = "BlockChustUIRemoteFlag", 
   Callback = function(Value)
       _G.BlockRemoteOnly = Value 
   end,
})
