-- Funções auxiliares
local function CreateRoundedCorner(frame, radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius)
    corner.Parent = frame
    return corner
end

local function CreateStroke(frame, color, thickness)
    local stroke = Instance.new("UIStroke")
    stroke.Color = color
    stroke.Thickness = thickness
    stroke.Parent = frame
    return stroke
end

-- Sistema de Notificações (mantido original)
local function CreateNotificationContainer()
    if NotificationContainer then return NotificationContainer end
    NotificationContainer = Instance.new("ScreenGui")
    NotificationContainer.Name = "AquaNotifications"
    NotificationContainer.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    NotificationContainer.Parent = CoreGui
    return NotificationContainer
end

function AquaUI:Notify(config)
    local container = CreateNotificationContainer()
    
    local yOffset = 10
    for _, notif in ipairs(ActiveNotifications) do
        if notif and notif.Parent then
            yOffset = yOffset + notif.AbsoluteSize.Y + 8
        end
    end
    
    local notification = Instance.new("Frame")
    notification.Size = UDim2.new(0, 340, 0, 85)
    notification.Position = UDim2.new(1, 0, 0, yOffset)
    notification.BackgroundColor3 = Theme.Background
    notification.BackgroundTransparency = 0.05
    notification.BorderSizePixel = 0
    notification.Parent = container
    CreateRoundedCorner(notification, 16)
    CreateStroke(notification, Theme.Border, 1)
    
    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1, -55, 0, 32)
    title.Position = UDim2.new(0, 15, 0, 8)
    title.BackgroundTransparency = 1
    title.Text = config.Title or "Notificação"
    title.TextColor3 = Theme.Text
    title.TextSize = 16
    title.TextXAlignment = Enum.TextXAlignment.Left
    title.Font = Enum.Font.GothamBold
    title.Parent = notification
    
    local description = Instance.new("TextLabel")
    description.Size = UDim2.new(1, -55, 0, 38)
    description.Position = UDim2.new(0, 15, 0, 40)
    description.BackgroundTransparency = 1
    description.Text = config.Description or ""
    description.TextColor3 = Theme.TextMuted
    description.TextSize = 12
    description.TextWrapped = true
    description.Font = Enum.Font.Gotham
    description.Parent = notification
    
    local closeBtn = Instance.new("TextButton")
    closeBtn.Size = UDim2.new(0, 28, 0, 28)
    closeBtn.Position = UDim2.new(1, -38, 0, 8)
    closeBtn.BackgroundColor3 = Theme.Danger
    closeBtn.BackgroundTransparency = 0.4
    closeBtn.Text = "X"
    closeBtn.TextColor3 = Theme.Text
    closeBtn.TextSize = 14
    closeBtn.Font = Enum.Font.GothamBold
    closeBtn.BorderSizePixel = 0
    closeBtn.Parent = notification
    CreateRoundedCorner(closeBtn, 14)
    
    table.insert(ActiveNotifications, notification)
    
    notification.Position = UDim2.new(1, 0, 0, yOffset)
    local tweenIn = TweenService:Create(notification, TweenInfo.new(0.4, Enum.EasingStyle.Quad), {
        Position = UDim2.new(1, -350, 0, yOffset)
    })
    tweenIn:Play()
    
    local function ReorganizeNotifications()
        local newY = 10
        for _, notif in ipairs(ActiveNotifications) do
            if notif and notif.Parent then
                TweenService:Create(notif, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {
                    Position = UDim2.new(1, -350, 0, newY)
                }):Play()
                newY = newY + notif.AbsoluteSize.Y + 8
            end
        end
    end
    
    local function CloseNotification()
        for i, notif in ipairs(ActiveNotifications) do
            if notif == notification then
                table.remove(ActiveNotifications, i)
                break
            end
        end
        local tweenOut = TweenService:Create(notification, TweenInfo.new(0.3, Enum.EasingStyle.Quad), {
            Position = UDim2.new(1, 0, 0, notification.Position.Y.Offset)
        })
        tweenOut:Play()
        tweenOut.Completed:Connect(function()
            notification:Destroy()
            ReorganizeNotifications()
        end)
    end
    
    closeBtn.MouseButton1Click:Connect(CloseNotification)
    
    local duration = config.Duration or 4
    if duration > 0 then
        task.wait(duration)
        if notification and notification.Parent then
            CloseNotification()
        end
    end
    
    return notification
end

        dragCircle.Position = UDim2.new(0, position.X - 15, 0, position.Y - 15)
        dragCircle.Visible = true
        TweenService:Create(dragCircle, TweenInfo.new(0.1, Enum.EasingStyle.Quad), {

    local function hideCircle()
        TweenService:Create(dragCircle, TweenInfo.new(0.15, Enum.EasingStyle.Quad), {
            BackgroundTransparency = 1

        task.wait(0.15)
        dragCircle.Visible = false
    end

    local lastMoveTime = 0
    local moveCheckConnection = nil
    local dragging = false
    local dragStart = nil
    local startPos = nil

    self.TitleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if not self.IsMaximized then
                dragging = true
                dragStart = input.Position
                startPos = self.MainFrame.Position
                showCircle(dragStart)
                lastMoveTime = tick()
                if not moveCheckConnection then
                    moveCheckConnection = game:GetService("RunService").Heartbeat:Connect(function()
                        if dragging then
                            if tick() - lastMoveTime > 0.3 then
                                if dragCircle.Visible then hideCircle() end
                            end
                        else
                            if dragCircle.Visible then hideCircle() end
                            if moveCheckConnection then moveCheckConnection:Disconnect() end
                            moveCheckConnection = nil
                        end
                    end)
                end
            end
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            self.MainFrame.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
            lastMoveTime = tick()
            local mousePos = input.Position
            dragCircle.Position = UDim2.new(0, mousePos.X - 15, 0, mousePos.Y - 15)
            if not dragCircle.Visible then
                showCircle(mousePos)
            end
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            hideCircle()
            if moveCheckConnection then moveCheckConnection:Disconnect() end
            moveCheckConnection = nil
        end

function Window:CreateTab(name, icon)
    local tabButton = Instance.new("TextButton")
    tabButton.Size = UDim2.new(1, 0, 0, 45)
    tabButton.BackgroundColor3 = Theme.Background2
    tabButton.BackgroundTransparency = 0.25
    tabButton.Text = icon and (icon .. " " .. name) or name
    tabButton.TextColor3 = Theme.TextMuted
    tabButton.TextSize = 14
    tabButton.Font = Enum.Font.Gotham
    tabButton.BorderSizePixel = 0
    tabButton.Parent = self.TabContainer
    CreateRoundedCorner(tabButton, 8)
    
    tabButton.MouseEnter:Connect(function()
        if self.CurrentTab and self.CurrentTab.Button == tabButton then return end
        TweenService:Create(tabButton, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
            BackgroundColor3 = Theme.Background2,
            BackgroundTransparency = 0.1,
            TextColor3 = Theme.Text
        }):Play()
    end)
    tabButton.MouseLeave:Connect(function()
        if self.CurrentTab and self.CurrentTab.Button == tabButton then return end
        TweenService:Create(tabButton, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
            BackgroundColor3 = Theme.Background2,
            BackgroundTransparency = 0.3,
            TextColor3 = Theme.TextMuted
        }):Play()
    end)
    
    local tabContent = Instance.new("Frame")
    tabContent.Size = UDim2.new(1, 0, 1, 0)
    tabContent.Position = UDim2.new(0, 0, 0, 0)
    tabContent.BackgroundTransparency = 1
    tabContent.Visible = false
    tabContent.Parent = self.ContentContainer
    
    local tab = {
        Button = tabButton,
        Content = tabContent,
        Sections = {}
    }
    
    table.insert(self.Tabs, tab)
    
    if not self.CurrentTab then
        self.CurrentTab = tab
        tabContent.Visible = true
        tabButton.BackgroundColor3 = Theme.Primary
        tabButton.BackgroundTransparency = 0.1
        tabButton.TextColor3 = Theme.Text
    end
    
    tabButton.MouseButton1Click:Connect(function()
        for _, t in ipairs(self.Tabs) do
            t.Content.Visible = false
            TweenService:Create(t.Button, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
                BackgroundColor3 = Theme.Background2,
                BackgroundTransparency = 0.3,
                TextColor3 = Theme.TextMuted
            }):Play()
        end
        tabContent.Visible = true
        TweenService:Create(tabButton, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
            BackgroundColor3 = Theme.Primary,
            BackgroundTransparency = 0.1,
            TextColor3 = Theme.Text
        }):Play()
        self.CurrentTab = tab
    end)
    
    return tab
end

function Window:CreateSection(tab, title)
    local sectionContainer = Instance.new("Frame")
    sectionContainer.Size = UDim2.new(1, 0, 0, 0)  -- altura dinâmica
    sectionContainer.Position = UDim2.new(0, 0, 0, #tab.Sections * 0)  -- será ajustado
    sectionContainer.BackgroundTransparency = 1
    sectionContainer.Parent = tab.Content
    
    local sectionBg = Instance.new("Frame")
    sectionBg.Size = UDim2.new(1, -20, 1, 0)
    sectionBg.Position = UDim2.new(0, 10, 0, 0)
    sectionBg.BackgroundColor3 = Theme.Background2
    sectionBg.BackgroundTransparency = 0.4
    sectionBg.BorderSizePixel = 0
    sectionBg.Parent = sectionContainer
    CreateRoundedCorner(sectionBg, 10)
    CreateStroke(sectionBg, Theme.Border, 1)
    
    local sectionTitle = Instance.new("TextLabel")
    sectionTitle.Size = UDim2.new(1, -20, 0, 40)
    sectionTitle.Position = UDim2.new(0, 10, 0, 0)
    sectionTitle.BackgroundTransparency = 1
    sectionTitle.Text = title
    sectionTitle.TextColor3 = Theme.Text
    sectionTitle.TextSize = 16
    sectionTitle.TextXAlignment = Enum.TextXAlignment.Left
    sectionTitle.Font = Enum.Font.GothamBold
    sectionTitle.Parent = sectionContainer
    
    local separator = Instance.new("Frame")
    separator.Size = UDim2.new(1, -20, 0, 2)
    separator.Position = UDim2.new(0, 10, 0, 38)
    separator.BackgroundColor3 = Theme.Primary
    separator.BackgroundTransparency = 0.3
    separator.BorderSizePixel = 0
    separator.Parent = sectionContainer
    CreateRoundedCorner(separator, 2)
    
    local contentArea = Instance.new("Frame")
    contentArea.Size = UDim2.new(1, 0, 0, 0)
    contentArea.Position = UDim2.new(0, 0, 0, 48)
    contentArea.BackgroundTransparency = 1
    contentArea.Parent = sectionContainer
    
    local section = {
        Container = sectionContainer,
        ContentArea = contentArea,
        Elements = {}
    }
    
    table.insert(tab.Sections, section)
    
    local function updateHeight()
        local totalHeight = 0
        for i, element in ipairs(section.Elements) do
            local yPos = totalHeight
            element.Position = UDim2.new(0, 10, 0, yPos)
            totalHeight = totalHeight + element.AbsoluteSize.Y + 6  -- +6 espaçamento
        end
        contentArea.Size = UDim2.new(1, 0, 0, totalHeight)
        sectionContainer.Size = UDim2.new(1, 0, 0, totalHeight + 60)
        -- Reorganiza a posição vertical da section dentro da tab
        local yOffset = 10
        for _, s in ipairs(tab.Sections) do
            s.Container.Position = UDim2.new(0, 0, 0, yOffset)
            yOffset = yOffset + s.Container.AbsoluteSize.Y + 10
        end
        -- Atualiza CanvasSize do ContentContainer
        local totalSectionHeight = 0
        for _, s in ipairs(tab.Sections) do
            totalSectionHeight = totalSectionHeight + s.Container.AbsoluteSize.Y + 10
        end
        if tab.Content and tab.Content.Parent then
            local parent = tab.Content.Parent
            if parent and parent.CanvasSize then
                parent.CanvasSize = UDim2.new(0, 0, 0, totalSectionHeight + 20)
            end
        end
    end
    
    section.UpdateHeight = updateHeight
    return section
end

function Window:CreateToggle(section, config)

    toggleFrame.BackgroundTransparency = 0.3

    toggleBtn.TextColor3 = Theme.TextMuted
    toggleBtn.TextSize = 12
    toggleBtn.Font = Enum.Font.GothamBold
    toggleBtn.BorderSizePixel = 0
    toggleBtn.Parent = toggleFrame
    CreateRoundedCorner(toggleBtn, 8)
    CreateStroke(toggleBtn, Theme.Border, 0.5)
    
    local state = config.Default or false
    if state then
        toggleBtn.BackgroundColor3 = Theme.Primary
        toggleBtn.Text = "ON"
        toggleBtn.TextColor3 = Theme.Text
    end
    
    toggleBtn.MouseButton1Click:Connect(function()
        state = not state
        if state then
            TweenService:Create(toggleBtn, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
                BackgroundColor3 = Theme.Primary,
                BackgroundTransparency = 0.1,
                TextColor3 = Theme.Text
            }):Play()
            toggleBtn.Text = "ON"
        else
            TweenService:Create(toggleBtn, TweenInfo.new(0.2, Enum.EasingStyle.Quad), {
                BackgroundColor3 = Theme.Background2,
                BackgroundTransparency = 0.2,
                TextColor3 = Theme.TextMuted
            }):Play()
            toggleBtn.Text = "OFF"
        end
        if config.Callback then
            config.Callback(state)
        end
    end)
    
    table.insert(section.Elements, toggleFrame)
    if section.UpdateHeight then section.UpdateHeight() end
    
    local toggleObj = {
        Set = function(_, val)
            state = val
            if state then
                toggleBtn.BackgroundColor3 = Theme.Primary
                toggleBtn.Text = "ON"
                toggleBtn.TextColor3 = Theme.Text
            else
                toggleBtn.BackgroundColor3 = Theme.Background2
                toggleBtn.Text = "OFF"
                toggleBtn.TextColor3 = Theme.TextMuted
            end
            if config.Callback then config.Callback(state) end
        end,
        Get = function() return state end
    }
    return toggleObj
end

function Window:CreateSlider(section, config)
    local sliderFrame = Instance.new("Frame")
    sliderFrame.Size = UDim2.new(1, -20, 0, 70)
    sliderFrame.BackgroundColor3 = Theme.Background2
    sliderFrame.BackgroundTransparency = 0.2
    sliderFrame.BorderSizePixel = 0
    sliderFrame.Parent = section.ContentArea
    sliderFrame.ClipsDescendants = false
    CreateRoundedCorner(sliderFrame, 10)
    CreateStroke(sliderFrame, Theme.Border, 1)

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.6, -10, 0, 20)
    label.Position = UDim2.new(0, 10, 0, 5)
    label.BackgroundTransparency = 1
    label.Text = config.Title or config.Text or "Slider"
    label.TextColor3 = Theme.Text
    label.TextSize = 14
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.GothamSemibold
    label.Parent = sliderFrame

    local valueLabel = Instance.new("TextLabel")
    valueLabel.Size = UDim2.new(0, 45, 0, 20)
    valueLabel.Position = UDim2.new(1, -55, 0, 5)
    valueLabel.BackgroundTransparency = 1
    valueLabel.Text = tostring(config.Default or 0)
    valueLabel.TextColor3 = Theme.Primary
    valueLabel.TextSize = 13
    valueLabel.TextXAlignment = Enum.TextXAlignment.Right
    valueLabel.Font = Enum.Font.GothamBold
    valueLabel.Parent = sliderFrame

    local sliderBg = Instance.new("Frame")
    sliderBg.Size = UDim2.new(1, -20, 0, 5)
    sliderBg.Position = UDim2.new(0, 10, 0, 35)
    sliderBg.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
    sliderBg.BorderSizePixel = 0
    sliderBg.Parent = sliderFrame
    CreateRoundedCorner(sliderBg, 3)

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new(0, 0, 1, 0)
    fill.BackgroundColor3 = Theme.Primary
    fill.BorderSizePixel = 0
    fill.Parent = sliderBg
    CreateRoundedCorner(fill, 3)

    local handle = Instance.new("Frame")
    handle.Size = UDim2.new(0, 16, 0, 16)
    handle.Position = UDim2.new(0, -8, 0, -5.5)
    handle.BackgroundColor3 = Theme.Primary
    handle.BorderSizePixel = 0
    handle.Parent = sliderBg
    CreateRoundedCorner(handle, 8)

    local shadow = Instance.new("Frame")
    shadow.Size = UDim2.new(1, 2, 1, 2)
    shadow.Position = UDim2.new(0, -1, 0, -1)
    shadow.BackgroundColor3 = Color3.new(0, 0, 0)
    shadow.BackgroundTransparency = 0.4
    shadow.BorderSizePixel = 0
    shadow.Parent = handle
    CreateRoundedCorner(shadow, 9)

    local min = config.Min or 0
    local max = config.Max or 100
    local value = config.Default or min
    local dragging = false

    local function updateSlider(val)
        value = math.clamp(val, min, max)
        local percent = (value - min) / (max - min)
        fill.Size = UDim2.new(percent, 0, 1, 0)
        handle.Position = UDim2.new(percent, -8, 0, -5.5)
        valueLabel.Text = tostring(math.floor(value))
        if config.Callback then
            config.Callback(value)
        end
    end

    local function getValueFromPosition(mouseX)
        local barPos = sliderBg.AbsolutePosition.X
        local barWidth = sliderBg.AbsoluteSize.X
        if barWidth > 0 then
            local percent = math.clamp((mouseX - barPos) / barWidth, 0, 1)
            return min + (max - min) * percent
        end
        return value
    end

    local function onInputBegin(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            local newValue = getValueFromPosition(input.Position.X)
            updateSlider(newValue)
        end
    end

    local function onInputMove(input)
        if dragging then
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                local newValue = getValueFromPosition(input.Position.X)
                updateSlider(newValue)
            end
        end
    end

    local function onInputEnd(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end

    sliderBg.InputBegan:Connect(onInputBegin)
    fill.InputBegan:Connect(onInputBegin)
    handle.InputBegan:Connect(onInputBegin)
    UserInputService.InputChanged:Connect(onInputMove)
    UserInputService.InputEnded:Connect(onInputEnd)

    updateSlider(value)

    table.insert(section.Elements, sliderFrame)
    if section.UpdateHeight then section.UpdateHeight() end

    return {
        Set = function(_, val) updateSlider(val) end,
        Get = function() return value end
    }
end

function Window:CreateKeybind(section, config)
    local keybindFrame = Instance.new("Frame")
    keybindFrame.Size = UDim2.new(1, -20, 0, 50)
    keybindFrame.BackgroundColor3 = Theme.Background2
    keybindFrame.BackgroundTransparency = 0.3
    keybindFrame.BorderSizePixel = 0
    keybindFrame.Parent = section.ContentArea
    CreateRoundedCorner(keybindFrame, 8)
    CreateStroke(keybindFrame, Theme.Border, 1)
    
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.6, -10, 1, 0)
    label.Position = UDim2.new(0, 10, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = config.Title or config.Text or "Keybind"
    label.TextColor3 = Theme.Text
    label.TextSize = 14
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.Gotham
    label.Parent = keybindFrame
    
    local keyLabel = Instance.new("TextButton")
    keyLabel.Size = UDim2.new(0, 80, 0, 30)
    keyLabel.Position = UDim2.new(1, -90, 0, 10)
    keyLabel.BackgroundColor3 = Theme.Background2
    keyLabel.BackgroundTransparency = 0.2
    keyLabel.Text = config.Default or "None"
    keyLabel.TextColor3 = Theme.Text
    keyLabel.TextSize = 12
    keyLabel.Font = Enum.Font.Gotham
    keyLabel.BorderSizePixel = 0
    keyLabel.Parent = keybindFrame
    CreateRoundedCorner(keyLabel, 8)
    CreateStroke(keyLabel, Theme.Border, 0.5)
    
    local waiting = false
    local currentKey = config.Default or "None"
    
    local function setKey(key)
        currentKey = key
        keyLabel.Text = key
        if config.Callback then
            config.Callback(key)
        end
        waiting = false
    end
    
    keyLabel.MouseButton1Click:Connect(function()
        if waiting then return end
        waiting = true
        keyLabel.Text = "..."
        
        local connection
        connection = UserInputService.InputBegan:Connect(function(input, gameProcessed)
            if gameProcessed then return end
            if waiting then
                if input.KeyCode ~= Enum.KeyCode.Unknown then
                    local keyName = input.KeyCode.Name
                    if keyName == "Backspace" then
                        setKey("None")
                    else
                        setKey(keyName)
                    end
                elseif input.UserInputType == Enum.UserInputType.MouseButton1 then
                    setKey("Mouse1")
                elseif input.UserInputType == Enum.UserInputType.MouseButton2 then
                    setKey("Mouse2")
                elseif input.UserInputType == Enum.UserInputType.Touch then
                    setKey("Touch")
                end
                connection:Disconnect()
            end
        end)
        
        task.delay(5, function()
            if waiting then
                waiting = false
                keyLabel.Text = currentKey
                connection:Disconnect()
            end
        end)
    end)
    
    table.insert(section.Elements, keybindFrame)
    if section.UpdateHeight then section.UpdateHeight() end
    
    local keybindObj = {
        Set = function(_, key) setKey(key) end,
        Get = function() return currentKey end
    }
    return keybindObj
end

        if isRebuilding then return end
        isRebuilding = true
        
        -- Limpeza completa: remove tudo exceto layout e padding
        local toDestroy = {}
        for _, child in ipairs(listScroller:GetChildren()) do
            if child ~= listLayout and child ~= listPadding then
                table.insert(toDestroy, child)
            end
        end
        for _, child in ipairs(toDestroy) do
            child:Destroy()
        end
        
        -- Cria os novos itens
        for i, option in ipairs(currentOptions) do
            local item = Instance.new("TextButton")
            item.Size = UDim2.new(1, -10, 0, 30)
            item.BackgroundColor3 = Theme.Background2
            item.BackgroundTransparency = 0.5
            item.Text = option
            item.TextColor3 = Theme.Text
            item.TextSize = 12
            item.Font = Enum.Font.Gotham
            item.BorderSizePixel = 0
            item.Parent = listScroller
            CreateRoundedCorner(item, 6)
            
            item.MouseEnter:Connect(function()
                TweenService:Create(item, TweenInfo.new(0.1, Enum.EasingStyle.Quad), {
                    BackgroundColor3 = Theme.Primary,
                    BackgroundTransparency = 0.2
                }):Play()
            end)
            
            item.MouseLeave:Connect(function()
                TweenService:Create(item, TweenInfo.new(0.1, Enum.EasingStyle.Quad), {
                    BackgroundColor3 = Theme.Background2,
                    BackgroundTransparency = 0.5
                }):Play()
            end)
            
            item.MouseButton1Click:Connect(function()
                currentValue = option
                mainButton.Text = option
                listContainer.Visible = false
                isOpen = false
                arrowIcon.Text = "▼"
                if config.Callback then
                    config.Callback(option)
                end
            end)
        end
        
        -- Força o ScrollingFrame a recalcular o canvas
        local autoState = listScroller.AutomaticCanvasSize
        listScroller.AutomaticCanvasSize = Enum.AutomaticSize.None
        listScroller.AutomaticCanvasSize = autoState
        
        task.defer(function()
            listScroller.CanvasSize = UDim2.new(0, 0, 0, 0)
        end)
        
        updateListPosition()
        isRebuilding = false
    end
    
    mainButton.MouseButton1Click:Connect(function()
        if isOpen then
            listContainer.Visible = false
            isOpen = false
            arrowIcon.Text = "▼"
        else
            isOpen = true
            rebuildList()
            updateListPosition()
            listContainer.Visible = true
            arrowIcon.Text = "▲"
        end
    end)
    
    table.insert(section.Elements, dropdownFrame)
    if section.UpdateHeight then section.UpdateHeight() end
    
    local dropdownObj = {
        Set = function(_, val)
            currentValue = val
            mainButton.Text = val
            if config.Callback then config.Callback(val) end
        end,
        Get = function()
            return currentValue
        end,
        Update = function(_, newOptions)
            -- Cópia e comparação para evitar recriação desnecessária
            local newTable = {}
            for i, v in ipairs(newOptions) do
                table.insert(newTable, v)
            end
            
            if #newTable == #currentOptions then
                local changed = false
                for i = 1, #newTable do
                    if newTable[i] ~= currentOptions[i] then
                        changed = true
                        break
                    end
                end
                if not changed then
                    return
                end
            end
            
            currentOptions = newTable
            if isOpen then
                rebuildList()
                updateListPosition()
            end
        end,
        GetOptions = function()
            return currentOptions
        end,
        Open = function()
            if not isOpen then
                isOpen = true
                rebuildList()
                updateListPosition()
                listContainer.Visible = true
                arrowIcon.Text = "▲"
            end
        end,
        Close = function()
            if isOpen then
                listContainer.Visible = false
                isOpen = false
                arrowIcon.Text = "▼"
            end
        end,
        Destroy = function()  -- ⬅️ Método para limpeza completa
            if listContainer then
                listContainer:Destroy()
            end
            if dropdownFrame then
                dropdownFrame:Destroy()
            end
        end
    }
    
    return dropdownObj
end

function Window:CreateTextInput(section, config)
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -20, 0, 50)
    frame.BackgroundColor3 = Theme.Background2
    frame.BackgroundTransparency = 0.3
    frame.BorderSizePixel = 0
    frame.Parent = section.ContentArea
    CreateRoundedCorner(frame, 8)
    CreateStroke(frame, Theme.Border, 1)

    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(0.6, -10, 1, 0)
    label.Position = UDim2.new(0, 10, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = config.Title or "Texto"
    label.TextColor3 = Theme.Text
    label.TextSize = 14
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.Gotham
    label.Parent = frame

    local displayBtn = Instance.new("TextButton")
    displayBtn.Size = UDim2.new(0, 80, 0, 30)
    displayBtn.Position = UDim2.new(1, -90, 0, 10)
    displayBtn.BackgroundColor3 = Theme.Background2
    displayBtn.BackgroundTransparency = 0.2
    displayBtn.Text = config.Default or (config.Placeholder or "Digite...")
    displayBtn.TextColor3 = config.Default and Theme.Text or Theme.TextMuted
    displayBtn.TextSize = 12
    displayBtn.Font = Enum.Font.Gotham
    displayBtn.BorderSizePixel = 0
    displayBtn.Parent = frame
    CreateRoundedCorner(displayBtn, 8)
    CreateStroke(displayBtn, Theme.Border, 0.5)

    local currentText = config.Default or ""
    local isEditing = false
    local textBox = nil

    local function updateDisplay()
        if currentText == "" then
            displayBtn.Text = config.Placeholder or "Digite..."
            displayBtn.TextColor3 = Theme.TextMuted
        else
            displayBtn.Text = currentText
            displayBtn.TextColor3 = Theme.Text
        end
    end

    local function finishEditing(newText)
        if newText ~= nil then
            currentText = newText
            updateDisplay()
            if config.Callback then
                config.Callback(currentText)
            end
        end
        if textBox and textBox.Parent then
            textBox:Destroy()
        end
        displayBtn.Visible = true
        isEditing = false
    end

    local function startEditing()
        if isEditing then return end
        isEditing = true
        displayBtn.Visible = false

        textBox = Instance.new("TextBox")
        textBox.Size = displayBtn.Size
        textBox.Position = displayBtn.Position
        textBox.BackgroundColor3 = Theme.Background2
        textBox.BackgroundTransparency = 0.2
        textBox.Text = currentText
        textBox.TextColor3 = Theme.Text
        textBox.TextSize = 12
        textBox.Font = Enum.Font.Gotham
        textBox.PlaceholderText = config.Placeholder or "Digite..."
        textBox.ClearTextOnFocus = false
        textBox.Parent = frame
        CreateRoundedCorner(textBox, 8)
        CreateStroke(textBox, Theme.Border, 0.5)

        task.wait(0.05)
        textBox:CaptureFocus()

        textBox.FocusLost:Connect(function(enterPressed)
            if enterPressed then
                finishEditing(textBox.Text)
            else
                finishEditing(currentText)
            end
        end)
    end

    displayBtn.MouseButton1Click:Connect(startEditing)

    table.insert(section.Elements, frame)
    if section.UpdateHeight then section.UpdateHeight() end

    updateDisplay()

    return {
        Get = function() return currentText end,
        Set = function(_, text)
            currentText = text
            updateDisplay()
            if config.Callback then config.Callback(currentText) end
        end,
        Clear = function()
            currentText = ""
            updateDisplay()
            if config.Callback then config.Callback("") end
        end
    }
end

    if NotificationContainer then
        NotificationContainer:Destroy()
        NotificationContainer = nil
    end
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not gameProcessed and input.KeyCode == Enum.KeyCode.Insert then
        AquaUI:ToggleUI()
    end
end)

return AquaUI
