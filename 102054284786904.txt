-- Helper: Get character humanoid
local function getHumanoid()
    local char = LocalPlayer.Character
    if char then
        return char:FindFirstChildOfClass("Humanoid")
    end
    return nil
end

-- Helper: Get HumanoidRootPart
local function getHRP()
    local char = LocalPlayer.Character
    if char then
        return char:FindFirstChild("HumanoidRootPart")
    end
    return nil
end

-- Helper: Get all NPCs in workspace
local function getNPCs()
    local npcs = {}
    local npcWorkspace = Workspace:FindFirstChild("WorkspaceScriptable")
    if npcWorkspace then
        local storage = npcWorkspace:FindFirstChild("Storage")
        if storage then
            local normalStorage = storage:FindFirstChild("NormalStorage")
            if normalStorage then
                local npcFolder = normalStorage:FindFirstChild("NPCWorkspace")
                if npcFolder then
                    for _, npc in pairs(npcFolder:GetChildren()) do
                        if npc:IsA("Model") and npc:FindFirstChild("Humanoid") then
                            table.insert(npcs, npc)
                        end
                    end
                end
            end
        end
    end
    return npcs
end

-- Helper: Check if NPC has contraband
local function hasContraband(npc)
    local props = npc:FindFirstChild("Properties")
    if props then
        local randVars = props:FindFirstChild("RandomVariables")
        if randVars then
            local real = randVars:FindFirstChild("ContrabandReal")
            if real and real.Value == true then
                return "real"
            end
            local fake = randVars:FindFirstChild("ContrabandFake")
            if fake and fake.Value == true then
                return "fake"
            end
            local passport = randVars:FindFirstChild("FakePassport")
            if passport and passport.Value == true then
                return "passport"
            end
        end
    end
    return nil
end

-- Clean up ESP
local function clearESP()
    for _, obj in pairs(espObjects) do
        if obj and obj.Parent then
            obj:Destroy()
        end
    end
    espObjects = {}
end

-- Speed Hack
local speedValue = GlobalSettings.PlayerRunSpeed
local function setSpeed(enabled)
    toggles.Speed = enabled
    local hum = getHumanoid()
    if hum then
        if enabled then
            hum.WalkSpeed = speedValue
        else
            hum.WalkSpeed = GlobalSettings.PlayerWalkSpeed
        end
    end
end

-- Infinite Sprint
local function setInfSprint(enabled)
    toggles.InfSprint = enabled
    if enabled then
        -- Use RenderStepped to run AFTER the game's sprint drain code every frame
        connections.InfSprint = RunService.RenderStepped:Connect(function()
            local hum = getHumanoid()
            if not hum then return end

            -- Force WalkSpeed to run speed (or custom speed if speed hack is on)
            if toggles.Speed then
                hum.WalkSpeed = speedValue
            else
                hum.WalkSpeed = GlobalSettings.PlayerRunSpeed
            end

            -- Keep sprint bar visually full
            local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
            if playerGui then
                local controlUI = playerGui:FindFirstChild("ControlUI")
                if controlUI then
                    local sprintBar = controlUI:FindFirstChild("SprintBarFrame")
                    if sprintBar then
                        local bar = sprintBar:FindFirstChild("Bar")
                        if bar then
                            bar.Size = UDim2.new(1, 0, 1, 0)
                        end
                        sprintBar.Visible = false
                    end
                end
            end
        end)
    else
        if connections.InfSprint then
            connections.InfSprint:Disconnect()
            connections.InfSprint = nil
        end
        -- Reset to normal walk speed if speed hack is off
        if not toggles.Speed then
            local hum = getHumanoid()
            if hum then
                hum.WalkSpeed = GlobalSettings.PlayerWalkSpeed
            end
        end
    end
end

-- NPC ESP

    if not toggles.NPCESP then return end

    for _, npc in pairs(getNPCs()) do

            highlight.Name = "HubESP"
            highlight.FillTransparency = 0.7
            highlight.OutlineTransparency = 0

                highlight.FillColor = Color3.fromRGB(255, 50, 50)
                highlight.OutlineColor = Color3.fromRGB(255, 80, 80)
            elseif contraband == "fake" then
                highlight.FillColor = Color3.fromRGB(255, 180, 30)
                highlight.OutlineColor = Color3.fromRGB(255, 200, 50)
            elseif contraband == "passport" then
                highlight.FillColor = Color3.fromRGB(255, 120, 0)
                highlight.OutlineColor = Color3.fromRGB(255, 140, 30)

                highlight.FillColor = Color3.fromRGB(50, 200, 80)
                highlight.OutlineColor = Color3.fromRGB(80, 255, 120)
            end

            highlight.Adornee = npc
            highlight.Parent = npc
            table.insert(espObjects, highlight)

            -- Name + status billboard

            bb.Name = "HubBB"
            bb.Size = UDim2.new(0, 160, 0, 40)
            bb.StudsOffset = Vector3.new(0, 3.5, 0)
            bb.AlwaysOnTop = true
            bb.Parent = hrp
            table.insert(espObjects, bb)

            local nameLabel = Instance.new("TextLabel")
            nameLabel.Size = UDim2.new(1, 0, 0.5, 0)
            nameLabel.BackgroundTransparency = 1
            nameLabel.TextColor3 = Color3.fromRGB(255, 255, 255)

            statusLabel.Position = UDim2.new(0, 0, 0.5, 0)
            statusLabel.BackgroundTransparency = 1
            statusLabel.TextStrokeTransparency = 0.5
            statusLabel.TextSize = 10
            statusLabel.Font = Enum.Font.GothamSemibold
            statusLabel.Parent = bb

            if contraband == "real" then
                nameLabel.Text = "⚠ CONTRABAND"
                statusLabel.Text = "REAL - ARREST"
                statusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
            elseif contraband == "fake" then
                nameLabel.Text = "⚡ FAKE ITEM"
                statusLabel.Text = "FALSE ALARM"
                statusLabel.TextColor3 = Color3.fromRGB(255, 200, 50)
            elseif contraband == "passport" then
                nameLabel.Text = "📄 FAKE PASSPORT"
                statusLabel.Text = "ARREST"
                statusLabel.TextColor3 = Color3.fromRGB(255, 140, 30)
            else
                nameLabel.Text = "✓ CLEAN"
                statusLabel.Text = "SAFE"
                statusLabel.TextColor3 = Color3.fromRGB(80, 255, 120)
            end

            -- Distance label
            local distLabel = Instance.new("TextLabel")
            distLabel.Size = UDim2.new(1, 0, 0, 14)
            distLabel.Position = UDim2.new(0, 0, 1, 2)
            distLabel.BackgroundTransparency = 1
            distLabel.TextColor3 = Color3.fromRGB(180, 170, 220)
            distLabel.TextStrokeTransparency = 0.5
            distLabel.TextSize = 9
            distLabel.Font = Enum.Font.Gotham
            distLabel.Parent = bb

            -- Update distance
            task.spawn(function()
                while bb.Parent and toggles.NPCESP do
                    local myHRP = getHRP()
                    if myHRP and hrp.Parent then
                        local dist = (myHRP.Position - hrp.Position).Magnitude
                        distLabel.Text = string.format("[%.0f studs]", dist)
                    end
                    task.wait(0.5)
                end
            end)
        end
    end
end

-- Noclip
local function setNoclip(enabled)
    toggles.Noclip = enabled
    if enabled then
        connections.Noclip = RunService.Stepped:Connect(function()
            local char = LocalPlayer.Character
            if char then
                for _, part in pairs(char:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.CanCollide = false
                    end
                end
            end
        end)
    else
        if connections.Noclip then
            connections.Noclip:Disconnect()
            connections.Noclip = nil
        end
    end
end

-- Teleport to nearest NPC with contraband
local function tpToContraband()
    local hrp = getHRP()
    if not hrp then return end

    local closest = nil
    local closestDist = math.huge

    for _, npc in pairs(getNPCs()) do
        local contraband = hasContraband(npc)
        if contraband == "real" or contraband == "passport" then
            local npcHRP = npc:FindFirstChild("HumanoidRootPart")
            if npcHRP then
                local dist = (hrp.Position - npcHRP.Position).Magnitude
                if dist < closestDist then
                    closestDist = dist
                    closest = npcHRP
                end
            end
        end
    end

    if closest then
        hrp.CFrame = closest.CFrame * CFrame.new(0, 0, 3)
    end
end

-- Teleport to a specific location
local function tpToSpawn()
    local hrp = getHRP()
    if not hrp then return end

    local spawn = Workspace:FindFirstChild("Map")
    if spawn then
        local spawnLoc = spawn:FindFirstChildOfClass("SpawnLocation", true)
        if spawnLoc then
            hrp.CFrame = spawnLoc.CFrame + Vector3.new(0, 3, 0)
        end
    end

    -- Fallback: find any SpawnLocation in workspace
    if hrp then
        for _, desc in pairs(Workspace:GetDescendants()) do
            if desc:IsA("SpawnLocation") then
                hrp.CFrame = desc.CFrame + Vector3.new(0, 3, 0)
                break
            end
        end
    end
end

local dragging = false
local dragInput
local dragStart
local startPos

TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

TitleBar.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
