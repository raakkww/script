-- Services
local UserInputService = game:GetService("UserInputService")
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer

-- Positions
local START_POS = Vector3.new(-114, 6, -1267)
local END_POS = Vector3.new(-111, 5, -53)

-- State
local isRunning = false
local loopTask = nil

-- GUI
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "GamingSpiritFix"
ScreenGui.Parent = LocalPlayer:WaitForChild("PlayerGui")
ScreenGui.ResetOnSpawn = false

local MainFrame = Instance.new("Frame")
MainFrame.Parent = ScreenGui
MainFrame.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
MainFrame.Position = UDim2.new(0.3, 0, 0.3, 0)
MainFrame.Size = UDim2.new(0, 320, 0, 200)
MainFrame.Active = true
Instance.new("UICorner", MainFrame).CornerRadius = UDim.new(0, 12)

-- Title
local Title = Instance.new("TextLabel")
Title.Parent = MainFrame
Title.BackgroundTransparency = 1
Title.Position = UDim2.new(0.05, 0, 0.05, 0)
Title.Size = UDim2.new(0.9, 0, 0, 35)
Title.Font = Enum.Font.FredokaOne
Title.Text = ""
Title.TextColor3 = Color3.fromRGB(255, 255, 255)
Title.TextSize = 18

-- Container
local ToggleFrame = Instance.new("Frame")
ToggleFrame.Parent = MainFrame
ToggleFrame.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
ToggleFrame.Position = UDim2.new(0.05, 0, 0.3, 0)
ToggleFrame.Size = UDim2.new(0.9, 0, 0, 60)
Instance.new("UICorner", ToggleFrame).CornerRadius = UDim.new(0, 10)

-- Toggle Button
local ToggleButton = Instance.new("TextButton")
ToggleButton.Parent = ToggleFrame
ToggleButton.BackgroundColor3 = Color3.fromRGB(45, 45, 45)
ToggleButton.Position = UDim2.new(0.78, 0, 0.15, 0)
ToggleButton.Size = UDim2.new(0, 42, 0, 42)
ToggleButton.Text = ""
Instance.new("UICorner", ToggleButton).CornerRadius = UDim.new(0, 8)

-- Drag System
local dragging = false
local dragInput
local dragStart
local startPos

local function updateDrag(input)
    local delta = input.Position - dragStart
    MainFrame.Position = UDim2.new(
        startPos.X.Scale,
        startPos.X.Offset + delta.X,
        startPos.Y.Scale,
        startPos.Y.Offset + delta.Y
    )
end

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
    or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement
    or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        updateDrag(input)
    end
end)

-- Main Loop Function
local function startAutoFarm()
    if loopTask then
        task.cancel(loopTask)
    end

    loopTask = task.spawn(function()
        while isRunning do
            local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
            local root = character:WaitForChild("HumanoidRootPart", 5)

            if root then
                -- Teleport to start
                root.CFrame = CFrame.new(START_POS)
                task.wait(0.1)

                if not isRunning then break end

                root.CFrame = CFrame.new(START_POS)

                -- Wait loop
                for i = 1, 100 do
                    if not isRunning then break end
                    task.wait(0.1)
                end

                if not isRunning then break end

                -- Teleport to end
                root.CFrame = CFrame.new(END_POS)
                task.wait(1)
            else
                task.wait(1)
            end
        end
    end)
end

-- Toggle Logic
ToggleButton.MouseButton1Click:Connect(function()
    isRunning = not isRunning

    if isRunning then
        ToggleButton.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
        startAutoFarm()
    else
        ToggleButton.BackgroundColor3 = Color3.fromRGB(45, 45, 45)

        if loopTask then
            task.cancel(loopTask)
            loopTask = nil
        end
    end
end)

-- Re-run after respawn
LocalPlayer.CharacterAdded:Connect(function()
    if isRunning then
        task.wait(1)
        startAutoFarm()
    end
end)
