local function QUSGZZ_fake_script()
	local script = Instance.new('LocalScript', Autofarm)

	local button = script.Parent
	local frame = button.Parent
	local player = game.Players.LocalPlayer
	
	local farming = false
	local toggledZones = {}
	
	local ON_COLOR = Color3.fromRGB(0, 200, 0)
	local OFF_COLOR = Color3.fromRGB(200, 0, 0)
	
	local function getChar()
		local char = player.Character or player.CharacterAdded:Wait()
		local hrp = char:WaitForChild("HumanoidRootPart")
		return char, hrp
	end
	
	for _, ui in ipairs(frame:GetChildren()) do
		if ui:IsA("TextButton") and ui.Name:match("Zone%d+") then
			toggledZones[ui.Name] = false
			ui.BackgroundColor3 = OFF_COLOR
	
			ui.MouseButton1Click:Connect(function()
				toggledZones[ui.Name] = not toggledZones[ui.Name]
				ui.BackgroundColor3 = toggledZones[ui.Name] and ON_COLOR or OFF_COLOR
			end)
		end
	end
	
	local function getSelectedZones()
		local list = {}
	
		for zoneName, enabled in pairs(toggledZones) do
			if enabled then
				table.insert(list, zoneName)
			end
		end
	
		table.sort(list, function(a, b)
			return tonumber(a:match("%d+")) < tonumber(b:match("%d+"))
		end)
	
		return list
	end
	
	local function farm()
		if farming then return end
		farming = true
	
		local _, hrp = getChar()
		local originalCFrame = hrp.CFrame
	
		local zonesFolder = workspace:WaitForChild("Zones")
		local oresFolder = zonesFolder:WaitForChild("Ores")
	
		while farming do
			local selectedZones = getSelectedZones()
	
			if #selectedZones == 0 then
				break
			end
	
			for _, zoneName in ipairs(selectedZones) do
	
				if not toggledZones[zoneName] then
					continue
				end
	
				local zonePart = oresFolder:FindFirstChild(zoneName)
	
				if zonePart then
					local pointsFolder = zonePart:FindFirstChild("GeneratedOrePoints")
	
					if pointsFolder then
						for i = 1, 13 do
	
							if not toggledZones[zoneName] then
								break
							end
	
							local pointName = string.format("OrePoint_%03d", i)
							local point = pointsFolder:FindFirstChild(pointName)
	
							if point and point:IsA("BasePart") then
								hrp.CFrame = CFrame.new(point.Position)
								task.wait(0.8)
							end
						end
					end
				end
			end
	
			task.wait(0.2)
		end
	
		hrp.CFrame = originalCFrame
		farming = false
	end
	
	button.MouseButton1Click:Connect(function()
		farm()
	end)
end
coroutine.wrap(QUSGZZ_fake_script)()
