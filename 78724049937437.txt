-- StarterGui.admin.ToggleController
local function C_3e()
local script = G2L["3e"];
	local TweenService = game:GetService("TweenService")
	local Players = game:GetService("Players")
	local RunService = game:GetService("RunService")
	local UIS = game:GetService("UserInputService")
	local lp = Players.LocalPlayer
	local gui = script.Parent
	local mainFrame = gui:WaitForChild("MainFrame")
	
	gui.ResetOnSpawn = false
	
	local ON_COLOR = Color3.fromRGB(64, 200, 110)   
	local OFF_COLOR = Color3.fromRGB(70, 70, 78)     
	local KNOB_OFF = UDim2.new(0, 2, 0.5, 0)
	local KNOB_ON = UDim2.new(1, -24, 0.5, 0)
	local TWEEN_INFO = TweenInfo.new(0.16, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
	
	local states: { [string]: boolean } = {}
	
	_G.PhState = _G.PhState or {
		conns = {},
		drawings = {},
		frames = {},
		flyBV = nil,
		flyBG = nil,
		coinFarmOn = false
	}
	local P = _G.PhState
	
	local function clearConns(name)
		local c = P.conns[name]
		if c then
			if type(c) == "thread" then
				pcall(function() task.cancel(c) end)
			else
				pcall(function() c:Disconnect() end)
			end
			P.conns[name] = nil
		end
	end
	
	local function getESPFolder()
		local f = gui:FindFirstChild("ESPFolder")
		if not f then
			f = Instance.new("Folder")
			f.Name = "ESPFolder"
			f.Parent = gui
		end
		return f
	end
	
	local function getFrame(key, isTracer)
		if not P.frames[key] then
			local f = Instance.new("Frame")
			f.Name = key
			if isTracer then
				f.AnchorPoint = Vector2.new(0.5, 0.5)
				f.BackgroundColor3 = Color3.fromRGB(255, 80, 100)
				f.BackgroundTransparency = 0.2
				f.BorderSizePixel = 0
			else
				f.BackgroundTransparency = 1
				local stroke = Instance.new("UIStroke")
				stroke.Color = Color3.fromRGB(255, 50, 70)
				stroke.Thickness = 1.5
				stroke.Parent = f
			end
			f.Parent = getESPFolder()
			P.frames[key] = f
		end
		return P.frames[key]
	end
	
	local function getSkeletonLine(key)
		if not P.frames[key] then
			local f = Instance.new("Frame")
			f.Name = key
			f.AnchorPoint = Vector2.new(0.5, 0.5)
			f.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
			f.BorderSizePixel = 0
			f.Parent = getESPFolder()
			P.frames[key] = f
		end
		return P.frames[key]
	end
	
	local function getRole(p)
		local tc = p.TeamColor
		if not tc then return 'PLAYER' end
		if tc.Name == 'Bright red' or tc.Name == 'Really red' then return 'SEEKER'
		elseif tc.Name == 'Bright blue' or tc.Name == 'Really blue' then return 'HIDER' end
		return 'PLAYER'
	end
	
	local function updateVisuals()
		clearConns("ESPLoop")
	
		if not (states["Box esp"] or states["Role ESP"] or states["Tracers ESP"] or states["Skeleton ESP"]) then
			local f = gui:FindFirstChild("ESPFolder")
			if f then f:ClearAllChildren() end
			P.frames = {}
			if P.drawings["Role"] then
				for _, d in pairs(P.drawings["Role"]) do pcall(function() d:Destroy() end) end
				P.drawings["Role"] = {}
			end
			return
		end
	
		P.conns["ESPLoop"] = RunService.RenderStepped:Connect(function()
			local cam = workspace.CurrentCamera
			if not cam then return end
	
			for _, p in ipairs(Players:GetPlayers()) do
				local char = p.Character
				local hrp = char and char:FindFirstChild("HumanoidRootPart")
				local hum = char and char:FindFirstChild("Humanoid")
	
				if p ~= lp and char and hrp and hum and hum.Health > 0 then
					local head = char:FindFirstChild("Head")
					local pos, onScreen = cam:WorldToViewportPoint(hrp.Position)
					local uid = p.UserId
	
					if states["Box esp"] then
						local box = getFrame("Box_"..uid, false)
						if onScreen and head then
							local headPos = cam:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
							local legPos = cam:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
							local height = math.abs(headPos.Y - legPos.Y)
							local width = height * 0.6
							box.Size = UDim2.new(0, width, 0, height)
							box.Position = UDim2.new(0, pos.X - width/2, 0, headPos.Y)
							box.Visible = true
						else
							box.Visible = false
						end
					else
						if P.frames["Box_"..uid] then P.frames["Box_"..uid]:Destroy() P.frames["Box_"..uid] = nil end
					end
	
					if states["Tracers ESP"] then
						local line = getFrame("Tracer_"..uid, true)
						if onScreen then
							local p1 = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y)
							local p2 = Vector2.new(pos.X, pos.Y)
							local distance = (p1 - p2).Magnitude
							line.Size = UDim2.new(0, distance, 0, 1.5)
							line.Position = UDim2.new(0, (p1.X + p2.X) / 2, 0, (p1.Y + p2.Y) / 2)
							line.Rotation = math.deg(math.atan2(p2.Y - p1.Y, p2.X - p1.X))
							line.Visible = true
						else
							line.Visible = false
						end
					else
						if P.frames["Tracer_"..uid] then P.frames["Tracer_"..uid]:Destroy() P.frames["Tracer_"..uid] = nil end
					end
	
					if states["Role ESP"] then
						P.drawings["Role"] = P.drawings["Role"] or {}
						local bb = P.drawings["Role"][uid]
						if not bb or not bb.Parent then
							bb = Instance.new("BillboardGui")
							bb.Size = UDim2.new(0, 100, 0, 20) bb.StudsOffset = Vector3.new(0, 3.5, 0)
							bb.AlwaysOnTop = true
							local txt = Instance.new("TextLabel", bb)
							txt.Size = UDim2.new(1, 0, 1, 0) txt.BackgroundTransparency = 1
							txt.TextSize = 12 txt.Font = Enum.Font.GothamBold
							txt.TextStrokeTransparency = 0.2
							P.drawings["Role"][uid] = bb
						end
						bb.Parent = char.PrimaryPart or hrp
						local txt = bb:FindFirstChildOfClass("TextLabel")
						local role = getRole(p)
						txt.Text = "[" .. role .. "]"
						txt.TextColor3 = role == "SEEKER" and Color3.fromRGB(255, 60, 60) or Color3.fromRGB(60, 180, 255)
					else
						if P.drawings["Role"] and P.drawings["Role"][uid] then pcall(function() P.drawings["Role"][uid]:Destroy() end) P.drawings["Role"][uid] = nil end
					end
	
					if states["Skeleton ESP"] then
						local joints = {
							{"Head", "UpperTorso"}, {"UpperTorso", "LowerTorso"},
							{"UpperTorso", "LeftUpperArm"}, {"LeftUpperArm", "LeftLowerArm"}, {"LeftLowerArm", "LeftHand"},
							{"UpperTorso", "RightUpperArm"}, {"RightUpperArm", "RightLowerArm"}, {"RightLowerArm", "RightHand"},
							{"LowerTorso", "LeftUpperLeg"}, {"LeftUpperLeg", "LeftLowerLeg"}, {"LeftLowerLeg", "LeftFoot"},
							{"LowerTorso", "RightUpperLeg"}, {"RightUpperLeg", "RightLowerLeg"}, {"RightLowerLeg", "RightFoot"}
						}
						if onScreen then
							for i, pair in ipairs(joints) do
								local p1, p2 = char:FindFirstChild(pair[1]), char:FindFirstChild(pair[2])
								if p1 and p2 then
									local pos1, vis1 = cam:WorldToViewportPoint(p1.Position)
									local pos2, vis2 = cam:WorldToViewportPoint(p2.Position)
									if vis1 or vis2 then
										local line = getSkeletonLine("Skel_"..uid.."_"..i)
										local vec1, vec2 = Vector2.new(pos1.X, pos1.Y), Vector2.new(pos2.X, pos2.Y)
										local distance = (vec1 - vec2).Magnitude
										line.Size = UDim2.new(0, distance, 0, 1.5)
										line.Position = UDim2.new(0, (vec1.X + vec2.X) / 2, 0, (vec1.Y + vec2.Y) / 2)
										line.Rotation = math.deg(math.atan2(vec2.Y - vec1.Y, vec2.X - vec1.X))
										line.Visible = true
									else
										if P.frames["Skel_"..uid.."_"..i] then P.frames["Skel_"..uid.."_"..i].Visible = false end
									end
								end
							end
						else
							for i=1,14 do
								if P.frames["Skel_"..uid.."_"..i] then P.frames["Skel_"..uid.."_"..i].Visible = false end
							end
						end
					else
						for i=1,14 do
							if P.frames["Skel_"..uid.."_"..i] then P.frames["Skel_"..uid.."_"..i]:Destroy() P.frames["Skel_"..uid.."_"..i] = nil end
						end
					end
	
				else
					if p.UserId then
						if P.frames["Box_"..p.UserId] then P.frames["Box_"..p.UserId]:Destroy() P.frames["Box_"..p.UserId] = nil end
						if P.frames["Tracer_"..p.UserId] then P.frames["Tracer_"..p.UserId]:Destroy() P.frames["Tracer_"..p.UserId] = nil end
						for i=1,14 do
							if P.frames["Skel_"..p.UserId.."_"..i] then P.frames["Skel_"..p.UserId.."_"..i]:Destroy() P.frames["Skel_"..p.UserId.."_"..i] = nil end
						end
						if P.drawings["Role"] and P.drawings["Role"][p.UserId] then pcall(function() P.drawings["Role"][p.UserId]:Destroy() end) P.drawings["Role"][p.UserId] = nil end
					end
				end
			end
		end)
	end
	
	local function onToggle(feature: string, enabled: boolean)
		states[feature] = enabled
	
		if feature == "Box esp" or feature == "Role ESP" or feature == "Tracers ESP" or feature == "Skeleton ESP" then
			updateVisuals()
		end
	
		if feature == "Invis" then
			clearConns("InvisLoop")
			if enabled then
				P.conns["InvisLoop"] = RunService.RenderStepped:Connect(function()
					local char = lp.Character
					if char then
						for _, v in ipairs(char:GetDescendants()) do
							if v:IsA("BasePart") or v:IsA("Decal") then
								if v.Transparency ~= 1 and v.Name ~= "HumanoidRootPart" then
									v:SetAttribute("OldTrans", v.Transparency)
									v.Transparency = 1
								end
							end
						end
					end
				end)
			else
				local char = lp.Character
				if char then
					for _, v in ipairs(char:GetDescendants()) do
						if v:IsA("BasePart") or v:IsA("Decal") then
							if v:GetAttribute("OldTrans") then
								v.Transparency = v:GetAttribute("OldTrans")
								v:SetAttribute("OldTrans", nil)
							end
						end
					end
				end
			end
	
		elseif feature == "Speed" then
			clearConns("Speed")
			if enabled then
				local function applySpeed(c) task.wait(0.1) local h = c:FindFirstChildOfClass("Humanoid") if h then h.WalkSpeed = 70 end end
				if lp.Character then applySpeed(lp.Character) end
				P.conns["Speed"] = lp.CharacterAdded:Connect(applySpeed)
			else
				local h = lp.Character and lp.Character:FindFirstChildOfClass("Humanoid") if h then h.WalkSpeed = 16 end
			end
	
		elseif feature == "Fly" then
			clearConns("Fly")
			if P.flyBV then P.flyBV:Destroy() P.flyBV = nil end
			if P.flyBG then P.flyBG:Destroy() P.flyBG = nil end
	
			local char = lp.Character
			local hum = char and char:FindFirstChildOfClass("Humanoid")
			local hrp = char and char:FindFirstChild("HumanoidRootPart")
	
			if hum then hum.PlatformStand = false end
			if not enabled or not hrp or not hum then return end
	
			hum.PlatformStand = true
			P.flyBV = Instance.new("BodyVelocity", hrp) P.flyBV.MaxForce = Vector3.new(1e5, 1e5, 1e5) P.flyBV.Velocity = Vector3.zero
			P.flyBG = Instance.new("BodyGyro", hrp) P.flyBG.MaxTorque = Vector3.new(1e5, 1e5, 1e5) P.flyBG.P = 5000 P.flyBG.CFrame = hrp.CFrame
	
			local spd = 50
			P.conns["Fly"] = RunService.RenderStepped:Connect(function()
				local cam = workspace.CurrentCamera
				if not cam then return end
				local fwd = cam.CFrame.LookVector
				local rgt = cam.CFrame.RightVector
				local v = Vector3.zero
				if UIS:IsKeyDown(Enum.KeyCode.W) then v = v + fwd * spd end
				if UIS:IsKeyDown(Enum.KeyCode.S) then v = v - fwd * spd end
				if UIS:IsKeyDown(Enum.KeyCode.A) then v = v - rgt * spd end
				if UIS:IsKeyDown(Enum.KeyCode.D) then v = v + rgt * spd end
				if UIS:IsKeyDown(Enum.KeyCode.Space) then v = v + Vector3.new(0, spd, 0) end
				if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.LeftShift) then v = v - Vector3.new(0, spd, 0) end
				P.flyBV.Velocity = v
				if v.Magnitude > 1 then P.flyBG.CFrame = CFrame.lookAt(hrp.Position, hrp.Position + Vector3.new(v.X, 0, v.Z)) end
			end)
	
		elseif feature == "AutoFarm" then
			P.coinFarmOn = enabled
			clearConns("AutoFarm")
			if not enabled then return end
	
			P.conns["AutoFarm"] = task.spawn(function()
				while P.coinFarmOn do
					local char = lp.Character
					local hrp = char and char:FindFirstChild("HumanoidRootPart")
					local hum = char and char:FindFirstChild("Humanoid")
					local bucket = workspace:FindFirstChild("SpawnedCoins")
	
					if hrp and hum and hum.Health > 0 and bucket then
						for _, coin in ipairs(bucket:GetChildren()) do
							if not P.coinFarmOn then break end
							local pos = nil
							if coin:IsA("BasePart") then pos = coin.Position
							elseif coin:IsA("Model") and coin.PrimaryPart then pos = coin.PrimaryPart.Position end
	
							if pos then
								hrp.CFrame = CFrame.new(pos + Vector3.new(0, 2.5, 0))
								task.wait(0.08)
							end
						end
					end
					task.wait(0.4)
				end
			end)
		end
	end
	
	local function applyVisual(toggle: TextButton, enabled: boolean, animate: boolean)
		local knob = toggle:FindFirstChild("Knob")
		local goalTrack = { BackgroundColor3 = enabled and ON_COLOR or OFF_COLOR }
		local goalKnob = { Position = enabled and KNOB_ON or KNOB_OFF }
		if animate then
			TweenService:Create(toggle, TWEEN_INFO, goalTrack):Play()
			if knob then
				TweenService:Create(knob, TWEEN_INFO, goalKnob):Play()
			end
		else
			toggle.BackgroundColor3 = goalTrack.BackgroundColor3
			if knob then
				knob.Position = goalKnob.Position
			end
		end
	end
	
	local function setup(toggle: TextButton)
		local feature = toggle:GetAttribute("Feature")
		if type(feature) ~= "string" then return end
	
		local enabled = toggle:GetAttribute("Toggled") == true
		states[feature] = enabled
		applyVisual(toggle, enabled, false)
	
		toggle.Activated:Connect(function()
			enabled = not enabled
			toggle:SetAttribute("Toggled", enabled)
			applyVisual(toggle, enabled, true)
			onToggle(feature, enabled)
		end)
	end
	
	for _, inst in ipairs(mainFrame:GetDescendants()) do
		if inst:IsA("TextButton") and inst.Name == "Toggle" then
			setup(inst)
		end
	end
	
	local tabBar = mainFrame:WaitForChild("Tab Bar")
	local btnVisuals = tabBar:WaitForChild("Visuals Button")
	local btnPlayer = tabBar:WaitForChild("Player Button")
	local btnAutoFarm = nil
	
	local pageVisuals = tabBar:WaitForChild("Visuals")
	local pagePlayer = tabBar:WaitForChild("Player Tab")
	local pageAutoFarm = nil
	
	for _, child in ipairs(tabBar:GetChildren()) do
		if child.Name == "Auto Farm" then
			if child:IsA("ImageButton") then
				btnAutoFarm = child
			elseif child:IsA("Frame") then
				pageAutoFarm = child
			end
		end
	end
	
	local tabs = {
		[btnVisuals] = pageVisuals,
		[btnPlayer] = pagePlayer,
		[btnAutoFarm] = pageAutoFarm
	}
	
	local function switchTab(activeBtn)
		for btn, page in pairs(tabs) do
			if btn and page then
				if btn == activeBtn then
					page.Visible = true
					btn.ImageTransparency = 0
				else
					page.Visible = false
					btn.ImageTransparency = 0.4
				end
			end
		end
	end
	
	for btn, _ in pairs(tabs) do
		if btn then
			btn.Activated:Connect(function()
				switchTab(btn)
			end)
		end
	end
	switchTab(btnVisuals)
	
	local dragToggle = nil
	local dragInput = nil
	local dragStart = nil
	local startPos = nil
	
	local function updateInput(input)
		local Delta = input.Position - dragStart
		local Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + Delta.X, startPos.Y.Scale, startPos.Y.Offset + Delta.Y)
		TweenService:Create(mainFrame, TweenInfo.new(0.25), {Position = Position}):Play()
	end
	
	mainFrame.InputBegan:Connect(function(input)
		if (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) and UIS:GetFocusedTextBox() == nil then
			dragToggle = true
			dragStart = input.Position
			startPos = mainFrame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragToggle = false
				end
			end)
		end
	end)
	
	mainFrame.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	
	UIS.InputChanged:Connect(function(input)
		if input == dragInput and dragToggle then
			updateInput(input)
		end
	end)
	
	UIS.InputBegan:Connect(function(input, gameProcessed)
		if not gameProcessed and input.KeyCode == Enum.KeyCode.RightShift then
			gui.Enabled = not gui.Enabled
		end
	end)
end;
