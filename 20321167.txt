local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer

if game.CoreGui:FindFirstChild("PTFSV2") then game.CoreGui.PTFSV2:Destroy() end

local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "PTFSV2"
ScreenGui.Parent = game.CoreGui

-- Variáveis de Configuração Interna
local SpeedEnabled = false
local JetSpeedValue = 3000 -- Velocidade padrão inicial
local IsMinimized = false

-- =============================================================================
-- INTERFACE GRÁFICA (UI MOBILE EVOLUÍDA)
-- =============================================================================

-- Janela Principal
local MainFrame = Instance.new("Frame", ScreenGui)
MainFrame.Size = UDim2.new(0, 280, 0, 190)
MainFrame.Position = UDim2.new(0.5, -140, 0.3, 0)
MainFrame.BackgroundColor3 = Color3.fromRGB(15, 15, 20)
MainFrame.BorderSizePixel = 0
MainFrame.ClipsDescendants = true -- Essencial para o efeito de minimizar
Instance.new("UICorner", MainFrame).CornerRadius = UDim.new(0, 12)

-- Barra de Cima (Header / Arrastável)
local Header = Instance.new("Frame", MainFrame)
Header.Size = UDim2.new(1, 0, 0, 45)
Header.BackgroundColor3 = Color3.fromRGB(25, 25, 30)
Instance.new("UICorner", Header).CornerRadius = UDim.new(0, 12)

local Title = Instance.new("TextLabel", Header)
Title.Size = UDim2.new(1, -90, 1, 0)
Title.Text = "PTFS"
Title.TextColor3 = Color3.fromRGB(0, 255, 255)
Title.Font = Enum.Font.GothamBold
Title.BackgroundTransparency = 1
Title.TextXAlignment = Enum.TextXAlignment.Left

-- Botão de Minimizar ("−")
local MinimizeBtn = Instance.new("TextButton", Header)
MinimizeBtn.Size = UDim2.new(0, 40, 1, 0)
MinimizeBtn.Position = UDim2.new(1, -85, 0, 0)
MinimizeBtn.BackgroundTransparency = 1
MinimizeBtn.Text = "−"
MinimizeBtn.TextColor3 = Color3.fromRGB(200, 200, 200)
MinimizeBtn.TextSize = 20
MinimizeBtn.Font = Enum.Font.GothamBold

-- Botão de Fechar ("X")
local CloseBtn = Instance.new("TextButton", Header)
CloseBtn.Size = UDim2.new(0, 45, 1, 0)
CloseBtn.Position = UDim2.new(1, -45, 0, 0)
CloseBtn.BackgroundTransparency = 1
CloseBtn.Text = "X"
CloseBtn.TextColor3 = Color3.fromRGB(255, 50, 50)
CloseBtn.TextSize = 16
CloseBtn.Font = Enum.Font.GothamBold

-- Container do Conteúdo (Soneca quando minimizado)
local ContentFrame = Instance.new("Frame", MainFrame)
ContentFrame.Size = UDim2.new(1, 0, 1, -45)
ContentFrame.Position = UDim2.new(0, 0, 0, 45)
ContentFrame.BackgroundTransparency = 1

-- Botão ON/OFF (Grande e fácil de tocar)
local ToggleBtn = Instance.new("TextButton", ContentFrame)
ToggleBtn.Size = UDim2.new(0, 240, 0, 45)
ToggleBtn.Position = UDim2.new(0, 20, 0, 15)
ToggleBtn.BackgroundColor3 = Color3.fromRGB(255, 50, 75)
ToggleBtn.Text = "SISTEMA: DESLIGADO"
ToggleBtn.TextColor3 = Color3.new(1, 1, 1)
ToggleBtn.Font = Enum.Font.GothamBold
Instance.new("UICorner", ToggleBtn).CornerRadius = UDim.new(0, 8)

-- Caixa de Entrada de Texto para a Velocidade (Input Numérico)
local SpeedInput = Instance.new("TextBox", ContentFrame)
SpeedInput.Size = UDim2.new(0, 240, 0, 45)
SpeedInput.Position = UDim2.new(0, 20, 0, 75)
SpeedInput.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
SpeedInput.Text = tostring(JetSpeedValue) -- Mostra a velocidade atual
SpeedInput.TextColor3 = Color3.fromRGB(0, 255, 255)
SpeedInput.PlaceholderText = "Digite a velocidade (Ex: 5000)"
SpeedInput.PlaceholderColor3 = Color3.fromRGB(100, 100, 100)
SpeedInput.TextSize = 16
SpeedInput.Font = Enum.Font.GothamBold
SpeedInput.ClearTextOnFocus = false -- Não apaga o número atual quando você clica
Instance.new("UICorner", SpeedInput).CornerRadius = UDim.new(0, 8)

local UIStroke = Instance.new("UIStroke", SpeedInput)
UIStroke.Color = Color3.fromRGB(50, 50, 60)
UIStroke.Thickness = 1.5

-- =============================================================================
-- LOGICA DOS BOTÕES E ACESSIBILIDADE
-- =============================================================================

-- Sistema Touch de Arrastar a Janela
local dragging, dragStart, startPos
Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement) then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
UserInputService.InputEnded:Connect(function() dragging = false end)

-- Sistema de Minimizar ("−" / "+")
MinimizeBtn.MouseButton1Click:Connect(function()
    IsMinimized = not IsMinimized
    if IsMinimized then
        MainFrame:TweenSize(UDim2.new(0, 280, 0, 45), "Out", "Quad", 0.2, true)
        ContentFrame.Visible = false
        MinimizeBtn.Text = "+"
    else
        MainFrame:TweenSize(UDim2.new(0, 280, 0, 190), "Out", "Quad", 0.2, true)
        ContentFrame.Visible = true
        MinimizeBtn.Text = "−"
    end
end)

ToggleBtn.MouseButton1Click:Connect(function()
    SpeedEnabled = not SpeedEnabled
    ToggleBtn.BackgroundColor3 = SpeedEnabled and Color3.fromRGB(0, 200, 100) or Color3.fromRGB(255, 50, 75)
    ToggleBtn.Text = SpeedEnabled and "SISTEMA: ATIVADO" or "SISTEMA: DESLIGADO"
end)

-- Captura do Número Digitado (Executa quando você clica fora ou dá "Enter/Concluir" no teclado do celular)
SpeedInput.FocusLost:Connect(function(enterPressed)
    local enteredNumber = tonumber(SpeedInput.Text)
    
    if enteredNumber then
        -- Limita o número digitado entre 50 e 10.000 para proteção
        JetSpeedValue = math.clamp(enteredNumber, 50, 10000)
        SpeedInput.Text = tostring(JetSpeedValue) -- Corrige o texto se passou do limite
    else
        -- Se digitou letras ou algo inválido, volta para o número que estava antes
        SpeedInput.Text = tostring(JetSpeedValue)
    end
end)

CloseBtn.MouseButton1Click:Connect(function() ScreenGui:Destroy() end)


RunService.Heartbeat:Connect(function()
    if not SpeedEnabled then return end
    
    local char = LocalPlayer.Character
    local humanoid = char and char:FindFirstChildOfClass("Humanoid")
    
    if humanoid and humanoid.SeatPart then
        local seat = humanoid.SeatPart
        local airplaneModel = seat:FindFirstAncestorOfClass("Model")
        
        if airplaneModel then
            local root = airplaneModel.PrimaryPart or seat
            
            if root.AssemblyLinearVelocity.Magnitude > 1 then
                local forwardDirection = root.CFrame.LookVector
                
                root.CFrame = root.CFrame + (forwardDirection * (JetSpeedValue * 0.01))
                root.AssemblyLinearVelocity = forwardDirection * JetSpeedValue
                
                for _, object in pairs(root:GetChildren()) do
                    if object:IsA("BodyVelocity") or object:IsA("LinearVelocity") then
                        object:Destroy()
                    end
                end
            end
        end
    end
end)
