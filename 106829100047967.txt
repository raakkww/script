-- Create GUI elements
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "DuelScriptGUI"
ScreenGui.Parent = game.Players.LocalPlayer:WaitForChild("PlayerGui")

-- Main Frame
local MainFrame = Instance.new("Frame")
MainFrame.Size = UDim2.new(0, 300, 0, 400)
MainFrame.Position = UDim2.new(0.5, -150, 0.5, -200)
MainFrame.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
MainFrame.BackgroundTransparency = 0.1
MainFrame.BorderSizePixel = 0
MainFrame.Active = true
MainFrame.Draggable = true
MainFrame.Parent = ScreenGui

-- Title
local TitleLabel = Instance.new("TextLabel")
TitleLabel.Size = UDim2.new(1, 0, 0, 30)
TitleLabel.Position = UDim2.new(0, 0, 0, 0)
TitleLabel.Text = "Duel Script Controller"
TitleLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
TitleLabel.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
TitleLabel.Font = Enum.Font.SourceSansBold
TitleLabel.TextSize = 18
TitleLabel.Parent = MainFrame

-- Close Button
local CloseButton = Instance.new("TextButton")
CloseButton.Size = UDim2.new(0, 30, 1, 0)
CloseButton.Position = UDim2.new(1, -30, 0, 0)
CloseButton.Text = "X"
CloseButton.TextColor3 = Color3.fromRGB(255, 255, 255)
CloseButton.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
CloseButton.Font = Enum.Font.SourceSansBold
CloseButton.TextSize = 14
CloseButton.Parent = TitleLabel
CloseButton.MouseButton1Click:Connect(function()
    ScreenGui:Destroy()
end)

-- Status Label
local StatusLabel = Instance.new("TextLabel")
StatusLabel.Size = UDim2.new(0.9, 0, 0, 40)
StatusLabel.Position = UDim2.new(0.05, 0, 0, 40)
StatusLabel.Text = "Status: Idle"
StatusLabel.TextColor3 = Color3.fromRGB(200, 200, 200)
StatusLabel.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
StatusLabel.Font = Enum.Font.SourceSans
StatusLabel.TextSize = 14
StatusLabel.Parent = MainFrame

-- Button Frame
local ButtonFrame = Instance.new("Frame")
ButtonFrame.Size = UDim2.new(0.9, 0, 0, 250)
ButtonFrame.Position = UDim2.new(0.05, 0, 0, 90)
ButtonFrame.BackgroundTransparency = 1
ButtonFrame.Parent = MainFrame

-- Create buttons in a grid
local buttons = {}
local buttonSize = 60
local spacing = 10
local columns = 4

for i = 1, 19 do
    local row = math.floor((i-1) / columns)
    local col = (i-1) % columns
    
    local button = Instance.new("TextButton")
    button.Size = UDim2.new(0, buttonSize, 0, buttonSize)
    button.Position = UDim2.new(0, col * (buttonSize + spacing), 0, row * (buttonSize + spacing))
    button.Text = tostring(i)
    button.TextColor3 = Color3.fromRGB(255, 255, 255)
    button.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
    button.Font = Enum.Font.SourceSansBold
    button.TextSize = 16
    button.Parent = ButtonFrame
    
    button.MouseButton1Click:Connect(function()
        -- Fire DuelEnd with the selected number
        local args = {true, i}
        game:GetService("ReplicatedStorage"):WaitForChild("RemoteEvents"):WaitForChild("DuelEnd"):FireServer(unpack(args))
        
        -- Visual feedback
        button.BackgroundColor3 = Color3.fromRGB(100, 200, 100)
        StatusLabel.Text = "Status: Fired DuelEnd with value " .. i
        wait(0.1)
        button.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
    end)
    
    buttons[i] = button
end

-- Control Panel Frame
local ControlFrame = Instance.new("Frame")
ControlFrame.Size = UDim2.new(0.9, 0, 0, 60)
ControlFrame.Position = UDim2.new(0.05, 0, 0, 340)
ControlFrame.BackgroundTransparency = 1
ControlFrame.Parent = MainFrame

-- StepTaken Toggle Button
local StepButton = Instance.new("TextButton")
StepButton.Size = UDim2.new(0, 120, 0, 40)
StepButton.Position = UDim2.new(0, 0, 0, 0)
StepButton.Text = "Start StepTaken"
StepButton.TextColor3 = Color3.fromRGB(255, 255, 255)
StepButton.BackgroundColor3 = Color3.fromRGB(50, 150, 50)
StepButton.Font = Enum.Font.SourceSansBold
StepButton.TextSize = 14
StepButton.Parent = ControlFrame

local isStepRunning = false
local stepConnection = nil

StepButton.MouseButton1Click:Connect(function()
    if not isStepRunning then
        -- Start the StepTaken loop
        isStepRunning = true
        StepButton.Text = "Stop StepTaken"
        StepButton.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
        StatusLabel.Text = "Status: StepTaken running"
        
        stepConnection = game:GetService("RunService").Stepped:Connect(function()
            game:GetService("ReplicatedStorage"):WaitForChild("RemoteEvents"):WaitForChild("StepTaken"):FireServer()
        end)
    else
        -- Stop the StepTaken loop
        isStepRunning = false
        StepButton.Text = "Start StepTaken"
        StepButton.BackgroundColor3 = Color3.fromRGB(50, 150, 50)
        StatusLabel.Text = "Status: StepTaken stopped"
        
        if stepConnection then
            stepConnection:Disconnect()
            stepConnection = nil
        end
    end
end)

-- Fire All Button
local FireAllButton = Instance.new("TextButton")
FireAllButton.Size = UDim2.new(0, 120, 0, 40)
FireAllButton.Position = UDim2.new(0, 130, 0, 0)
FireAllButton.Text = "Fire All (1-19)"
FireAllButton.TextColor3 = Color3.fromRGB(255, 255, 255)
FireAllButton.BackgroundColor3 = Color3.fromRGB(150, 100, 50)
FireAllButton.Font = Enum.Font.SourceSansBold
FireAllButton.TextSize = 12
FireAllButton.Parent = ControlFrame

FireAllButton.MouseButton1Click:Connect(function()
    StatusLabel.Text = "Status: Firing all DuelEnd events..."
    for i = 1, 19 do
        local args = {true, i}
        game:GetService("ReplicatedStorage"):WaitForChild("RemoteEvents"):WaitForChild("DuelEnd"):FireServer(unpack(args))
        
        -- Visual feedback on buttons
        if buttons[i] then
            buttons[i].BackgroundColor3 = Color3.fromRGB(100, 200, 100)
            wait(0.05)
            buttons[i].BackgroundColor3 = Color3.fromRGB(70, 70, 70)
        end
        wait(0.05)
    end
    StatusLabel.Text = "Status: Finished firing all events"
end)

-- Minimize/Maximize Button
local MinimizeButton = Instance.new("TextButton")
MinimizeButton.Size = UDim2.new(0, 30, 1, 0)
MinimizeButton.Position = UDim2.new(1, -60, 0, 0)
MinimizeButton.Text = "_"
MinimizeButton.TextColor3 = Color3.fromRGB(255, 255, 255)
MinimizeButton.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
MinimizeButton.Font = Enum.Font.SourceSansBold
MinimizeButton.TextSize = 14
MinimizeButton.Parent = TitleLabel

local isMinimized = false
local originalHeight = MainFrame.Size.Y.Scale
local contentHeight = 0.7

MinimizeButton.MouseButton1Click:Connect(function()
    if not isMinimized then
        MainFrame.Size = UDim2.new(0, 300, 0, 50)
        ButtonFrame.Visible = false
        ControlFrame.Visible = false
        StatusLabel.Visible = false
        isMinimized = true
        MinimizeButton.Text = "□"
    else
        MainFrame.Size = UDim2.new(0, 300, 0, 400)
        ButtonFrame.Visible = true
        ControlFrame.Visible = true
        StatusLabel.Visible = true
        isMinimized = false
        MinimizeButton.Text = "_"
    end
end)

-- Console output for debugging
print("Duel Script GUI Loaded Successfully!")
print("Features:")
print("- 19 individual DuelEnd buttons (values 1-19)")
print("- StepTaken toggle (auto-fire on game stepped)")
print("- Fire All button to trigger all DuelEnd events")
print("- Minimize button to reduce GUI size")
print("- Draggable window")

-- Clean up on player death or leave
game.Players.LocalPlayer.CharacterAdded:Connect(function()
    if stepConnection then
        stepConnection:Disconnect()
        stepConnection = nil
        isStepRunning = false
    end
end)
