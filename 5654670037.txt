-- ð STATUS CHECKER
local function GetStatus(model)
    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum then return "DEAD", AYARLAR.Colors.Dead end
    
    local arrested = model:FindFirstChild("Handcuffed") or model:GetAttribute("Handcuffed") or 
                     model:FindFirstChild("Arrested") or model:GetAttribute("Arrested") or
                     model:FindFirstChild("Cuffed") or (model:FindFirstChild("Values") and model.Values:FindFirstChild("Cuffed"))
    
    if hum.Health <= 0 then return "DEAD", AYARLAR.Colors.Dead
    elseif arrested then return "ARRESTED", AYARLAR.Colors.Arrested
    else return "ALIVE", AYARLAR.Colors.Hostile end
end

local function ApplyESP(obj, name, color, isEntity)
    if isEntity then
        local head = obj:FindFirstChild("Head") or obj:FindFirstChild("HumanoidRootPart")
        if not head then return end
        
        local status, sColor = GetStatus(obj)
        local finalColor = (name == "PLAYER") and AYARLAR.Colors.Player or sColor
        
        -- Hitbox Expansion (Sadece CanlÄ± NPC'ler iÃ§in)
        if name == "NPC" and status == "ALIVE" then
            head.Size = Vector3.new(AYARLAR.HitboxSize, AYARLAR.HitboxSize, AYARLAR.HitboxSize)
            head.Transparency = 0.7 -- KafayÄ± hafif Åeffaf yapar ki Ã¶nÃ¼nÃ¼ gÃ¶r
            head.CanCollide = false
        elseif status == "DEAD" or status == "ARRESTED" then
            head.Size = Vector3.new(1, 1, 1) -- Normale dÃ¶ndÃ¼r
            head.Transparency = 0
        end

        -- Billboard Tag
        local tag = head:FindFirstChild("OmniTag")
        if not tag then
            tag = Instance.new("BillboardGui", head); tag.Name = "OmniTag"
            tag.AlwaysOnTop = true; tag.Size = UDim2.new(0, 100, 0, 40); tag.StudsOffset = Vector3.new(0, 3, 0)
            local tl = Instance.new("TextLabel", tag); tl.Name = "L"; tl.Size = UDim2.new(1,0,1,0); tl.BackgroundTransparency = 1; tl.TextSize = 10; tl.Font = Enum.Font.Code; tl.TextStrokeTransparency = 0
        end
        tag.L.Text = string.format("%s\n[%s]", name, status)
        tag.L.TextColor3 = finalColor
        
        -- Highlight
        local hl = obj:FindFirstChild("OmniHL") or Instance.new("Highlight", obj)
        hl.Name = "OmniHL"; hl.FillColor = finalColor; hl.FillTransparency = 0.75; hl.DepthMode = "AlwaysOnTop"
    else
        -- Obje ESP
        if not obj:FindFirstChild("OmniHL") then
            local hl = Instance.new("Highlight", obj); hl.Name = "OmniHL"
            hl.FillColor = AYARLAR.Colors.Object; hl.FillTransparency = 0.6; hl.DepthMode = "AlwaysOnTop"
        end
    end
end

-- ð¯ AIMBOT
local function GetTarget()
    local closest, dist = nil, AYARLAR.FOV
    local mouse = UserInputService:GetMouseLocation()
    
    for _, v in pairs(Workspace:GetDescendants()) do
        if v:IsA("Model") and v:FindFirstChild("Humanoid") and v ~= LocalPlayer.Character then
            local status, _ = GetStatus(v)
            if status == "ALIVE" and not Players:GetPlayerFromCharacter(v) then
                local head = v:FindFirstChild("Head")
                if head then
                    local pos, vis = Camera:WorldToViewportPoint(head.Position)
                    if vis then
                        local mag = (Vector2.new(pos.X, pos.Y) - mouse).Magnitude
                        if mag < dist then dist = mag; closest = head end
                    end
                end
            end
        end
    end
    return closest
end

-- ð« ACS MODS
local function ModACS()
    local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Tool")
    if tool and tool:FindFirstChild("ACS_Settings") then
        local success, v2 = pcall(require, tool.ACS_Settings)
        if success and type(v2) == "table" then
            v2.MinSpread = 0; v2.MaxSpread = 0
            if v2.camRecoil then for i,v in pairs(v2.camRecoil) do v2.camRecoil[i] = {0,0} end end
            if v2.gunRecoil then for i,v in pairs(v2.gunRecoil) do v2.gunRecoil[i] = {0,0} end end
        end
    end
end
