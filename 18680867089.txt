-- === УЛУЧШЕННАЯ ЛОГИКА ПОИСКА БАЗЫ ===
local function FindMyBase()
    -- Если игрок сам нажал "SET BASE", используем его сохранение
    if SavedBaseCF then return SavedBaseCF end
    
    -- Авто-поиск по всем возможным папкам
    local searchIn = {workspace, workspace:FindFirstChild("Tycoons"), workspace:FindFirstChild("Plots"), workspace:FindFirstChild("Bases")}
    
    for _, folder in pairs(searchIn) do
        if folder then
            for _, base in pairs(folder:GetChildren()) do
                -- Проверка: имя базы совпадает с ником ИЛИ внутри есть объект Owner с ником
                local owner = base:FindFirstChild("Owner") or base:FindFirstChild("Player") or base:FindFirstChild("OwnerName")
                local isOwner = (base.Name == Player.Name) or (owner and (tostring(owner.Value) == Player.Name or owner.Value == Player))
                
                if isOwner then
                    -- Ищем точку для ТП (Спавн, Сборщик или просто центр)
                    local spawn = base:FindFirstChild("Spawn") or base:FindFirstChild("Essentials") and base.Essentials:FindFirstChild("Spawn") or base:FindFirstChild("Collector") or base:FindFirstChildWhichIsA("BasePart", true)
                    return spawn and spawn.CFrame
                end
            end
        end
    end
    return nil
end

local TeleportPage = CreatePage("Teleport")

local function OpenTab(tabName)
    for _, page in pairs(Pages:GetChildren()) do
        if page:IsA("ScrollingFrame") then page.Visible = false end
    end

end

-- === ТЕЛЕПОРТЫ ===
local function AddTP(name, color, defaultCoords)
    local btn = Instance.new("TextButton", TeleportPage)
    btn.Size = UDim2.new(1, -5, 0, 50)
    btn.BackgroundColor3 = color
    btn.Text = name
    btn.TextColor3 = Color3.new(1, 1, 1)
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 17
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)
    
    btn.MouseButton1Click:Connect(function()
        local hrp = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            if name == "BASE" then
                local baseCF = FindMyBase()
                hrp.CFrame = baseCF and (baseCF + Vector3.new(0, 5, 0)) or CFrame.new(defaultCoords)
            else
                hrp.CFrame = CFrame.new(defaultCoords + Vector3.new(0, 3, 0))
            end
            btn.Text = "done!"
            task.wait(0.5)
            btn.Text = name
        end
    end)
end
