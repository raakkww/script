-- Autofarm Logic
local Enabled = false

Toggle.MouseButton1Click:Connect(function()
	Enabled = not Enabled

	if Enabled then
		Toggle.Text = "🪙 Autofarm: ON"
		Toggle.BackgroundColor3 = Color3.fromRGB(50,170,90)

		task.spawn(function()
			while Enabled do
				local Character = Player.Character
				local Root = Character and Character:FindFirstChild("HumanoidRootPart")

				local CoinsFolder = workspace:FindFirstChild("Coins")

				if Root and CoinsFolder then
					local CoinParts = {}

					for _, Object in ipairs(CoinsFolder:GetChildren()) do
						if Object:IsA("BasePart") then
							table.insert(CoinParts, Object)
						end
					end

					if #CoinParts > 0 then
						local RandomCoin = CoinParts[math.random(1, #CoinParts)]

						Root.CFrame = RandomCoin.CFrame + Vector3.new(0, 3, 0)
					end
				end

				task.wait(0.25)
			end
		end)
	else
		Toggle.Text = "🪙 Autofarm: OFF"
		Toggle.BackgroundColor3 = Color3.fromRGB(55,55,70)
	end
end)

-- Minimize

end)

-- Dragging (PC + Mobile)
local Dragging = false
local DragStart
local StartPos

local function Update(Input)
	local Delta = Input.Position - DragStart

	Main.Position = UDim2.new(
		StartPos.X.Scale,
		StartPos.X.Offset + Delta.X,
		StartPos.Y.Scale,
		StartPos.Y.Offset + Delta.Y
	)

	Shadow.Position = UDim2.new(
		StartPos.X.Scale,
		StartPos.X.Offset + Delta.X + 5,
		StartPos.Y.Scale,
		StartPos.Y.Offset + Delta.Y + 5
	)
end

TopBar.InputBegan:Connect(function(Input)
	if Input.UserInputType == Enum.UserInputType.MouseButton1
	or Input.UserInputType == Enum.UserInputType.Touch then

		Dragging = true
		DragStart = Input.Position
		StartPos = Main.Position

		Input.Changed:Connect(function()
			if Input.UserInputState == Enum.UserInputState.End then
				Dragging = false
			end
		end)
	end
end)

UIS.InputChanged:Connect(function(Input)
	if Dragging and (
		Input.UserInputType == Enum.UserInputType.MouseMovement
		or Input.UserInputType == Enum.UserInputType.Touch
	) then
		Update(Input)
	end
end)
