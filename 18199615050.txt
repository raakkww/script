local dragging = false

toggleBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = toggleBtn.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        toggleBtn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

local isOpen = true

local function CheckInventory(ItemName)
    for _, obj in ipairs(player.PlayerGui.Hotbar.Slots:GetChildren()) do
        if obj:IsA("Frame") and string.find(string.lower(obj.Name), "invslot") and obj.ItemName.Text == ItemName then
            return true, tonumber(obj.Name:match("%d+"))
        end
    end
    return false, nil
end

local function FindItem(ItemName)
    for _, v in pairs(workspace.Items:GetChildren()) do
        if v:IsA("Model") and v:GetAttribute("ItemName") == ItemName then
            return true, v
        end
    end
    return false, nil
end

local function ActiveItem()
    local char = player.Character
    if char then
        for _, v in pairs(char:GetChildren()) do
            if (v:IsA("Model") or tonumber(v.Name)) and v:GetAttribute("Enabled") ~= true and v:FindFirstChild("Handle") then
                Events.ToggleItemState:FireServer(v)
                return true, v
            end
        end
    end
    return false, nil
end

local function EquipItem(SlotNum)
    Events.RequestItemEquip:FireServer("InvSlot" .. tostring(SlotNum))
end

local function PickupItem(Model)
    Events.RequestItemPickup:FireServer(Model)
end

local function DropItem(SlotNum)
    Events.RequestItemDrop:FireServer("InvSlot" .. tostring(SlotNum))
end

local function TpOutside()
    pcall(function()
        local pegboard = workspace:WaitForChild("Map"):WaitForChild("Rooms"):WaitForChild("Base Camp"):WaitForChild("Pegboard")
        local union = pegboard:FindFirstChild("Union")
        local char = player.Character or player.CharacterAdded:Wait()
        if union and char and char:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.CFrame = union.CFrame + Vector3.new(0, 3, 0)
        end
    end)
end

local function FireSpiritBox()
    local questions = {
        "Are you far away?", "Are you near?", "Where are you?", "What do you want?",
        "When did you cross over?", "Are you in the room with me?", "Do you want us to leave?",
        "When did you pass away?", "What is your goal?", "Why are you here?",
        "How long ago did you die?", "Is there a ghost here?"
    }
    Events.AskSpiritBoxFromUI:FireServer(questions[math.random(1, #questions)])
end

-- Evidence checkers
local function CheckHandprints()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and (obj.Name == "Handprint1" or obj.Name == "Footprint1") then return true end
    end
    return false
end

local function CheckGhostOrb()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name == "GhostOrb" then return true end
    end
    return false
end

local function CheckEMF()
    local level = 0
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Folder") and obj.Name == "Indicators" then
            for _, v in pairs(obj:GetChildren()) do
                if v:IsA("BasePart") and v.Material == Enum.Material.Neon then
                    local lvl = tonumber(v.Name)
                    if lvl and lvl > level then level = lvl end
                end
            end
        end
    end
    return level
end

local function CheckWither()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") and obj.Name == "Petals" and obj.Color == Color3.new(0,0,0) then return true end
    end
    return false
end

local function CheckGhostWriting()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Decal") then
            local model = obj:FindFirstAncestorWhichIsA("Model")
            if model and model:GetAttribute("ItemName") == "Spirit Book" and obj.Texture ~= "" then return true end
        end
    end
    return false
end

local function CheckSpiritBox()
    local ok, text = pcall(function()
        return player.PlayerGui.Subtitles.Holder.TextLabel.Text
    end)
    return ok and #text:gsub("%s+", "") >= 3
end

-- ESP functions
local function CreateGhostESP()
    local ghost = GetGhost()
    if not ghost then return end
    if ghostBillboard then ghostBillboard:Destroy() end
    if ghostHighlight then ghostHighlight:Destroy() end

    ghostBillboard = Instance.new("BillboardGui")
    ghostBillboard.Name = "HeliosGhostBill"
    ghostBillboard.Parent = game:GetService("CoreGui")
    ghostBillboard.Adornee = ghost
    ghostBillboard.Size = UDim2.new(0, 60, 0, 60)
    ghostBillboard.AlwaysOnTop = true
    local label = Instance.new("TextLabel", ghostBillboard)
    label.Size = UDim2.new(1,0,1,0)
    label.BackgroundTransparency = 1
    label.Text = "Ghost"
    label.TextColor3 = Color3.fromRGB(255,0,0)
    label.TextScaled = true

    ghostHighlight = Instance.new("Highlight")
    ghostHighlight.Parent = ghost
    ghostHighlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    ghostHighlight.OutlineColor = Color3.fromRGB(255,0,0)
    ghostHighlight.FillTransparency = 0.5
end

local function UpdateItemEsp()
    for _, v in pairs(itemEspList) do pcall(function() v:Destroy() end) end
    itemEspList = {}
    if not ItemEspToggle then return end
    for _, v in pairs(workspace.Items:GetChildren()) do
        if v:IsA("Model") and v:GetAttribute("ItemName") then
            local bill = Instance.new("BillboardGui")
            bill.Parent = game:GetService("CoreGui")
            bill.Adornee = v
            bill.Size = UDim2.new(0, 100, 0, 40)
            bill.AlwaysOnTop = true
            local lbl = Instance.new("TextLabel", bill)
            lbl.Size = UDim2.new(1,0,1,0)
            lbl.BackgroundTransparency = 1
            lbl.Text = v:GetAttribute("ItemName")
            lbl.TextColor3 = Color3.fromRGB(255,170,0)
            lbl.TextScaled = true
            table.insert(itemEspList, bill)
        end
    end
end

            lbl.TextScaled = true

local function UpdatePlayersEsp()
    for _, plr in pairs(PS:GetPlayers()) do
        if plr.Character then
            local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                local bill = hrp:FindFirstChild("HeliosPlrBil")
                local high = hrp:FindFirstChild("HeliosPlrEsp")
                if PlayersEspToggle then
                    if not bill then
                        bill = Instance.new("BillboardGui")
                        bill.Name = "HeliosPlrBil"
                        bill.Parent = hrp
                        bill.Size = UDim2.new(0, 100, 0, 40)
                        bill.AlwaysOnTop = true
                        local lbl = Instance.new("TextLabel", bill)
                        lbl.Size = UDim2.new(1,0,1,0)
                        lbl.BackgroundTransparency = 1
                        lbl.Text = plr.DisplayName
                        lbl.TextColor3 = Color3.fromRGB(0,170,255)
                        lbl.TextScaled = true
                    end
                    if not high then
                        high = Instance.new("Highlight")
                        high.Name = "HeliosPlrEsp"
                        high.Parent = hrp
                        high.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        high.OutlineColor = Color3.fromRGB(0,170,255)
                        high.FillTransparency = 1
                    end
                    bill.Enabled = true
                    high.Enabled = true
                else
                    if bill then bill:Destroy() end
                    if high then high:Destroy() end
                end
            end
        end
    end
end

local function UpdateFullBright()
    if FullBrightToggle then
        Lighting.Ambient = Color3.new(1,1,1)
        Lighting.OutdoorAmbient = Color3.new(1,1,1)
        Lighting.Brightness = 3
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 100000
        for _, obj in ipairs(Lighting:GetChildren()) do
            if obj:IsA("FogEffect") or obj:IsA("BloomEffect") or obj:IsA("BlurEffect") then
                obj.Enabled = false
            end
        end
    else
        Lighting.Ambient = Color3.new(0,0,0)
        Lighting.OutdoorAmbient = Color3.new(0,0,0)
        Lighting.Brightness = 2
        Lighting.GlobalShadows = true
        Lighting.FogEnd = 1000
        for _, obj in ipairs(Lighting:GetChildren()) do
            if obj:IsA("FogEffect") or obj:IsA("BloomEffect") or obj:IsA("BlurEffect") then
                obj.Enabled = true
            end
        end
    end
end

-- Place Stuff Near Ghost
demonologyTab:Button({
    Title = "Place stuffs near ghost",
    Justify = "Center",
    Color = Color3.fromRGB(100,100,200),
    Callback = function()
        local ghost = GetGhost()
        if not ghost then
            WindUI:Notify({ Title = "Place Stuff", Content = "Ghost not found", Duration = 2, Type = "error" })
            return
        end
        if ghost:GetAttribute("Hunting") then
            WindUI:Notify({ Title = "Place Stuff", Content = "Ghost is hunting!", Duration = 2, Type = "error" })
            return
        end
        local char = player.Character
        if not char then return end

        char:PivotTo(ghost:GetPivot())
        task.wait(0.1)

        -- Drop anything currently held
        for i = 1, 3 do
            EquipItem(1)
            task.wait(0.2)
            DropItem(1)
            task.wait(0.2)
        end

        -- Collect all available items
        local allItems = {}
        for _, v in pairs(workspace.Items:GetChildren()) do
            if v:IsA("Model") and v:GetAttribute("ItemName") then
                local has, _ = CheckInventory(v:GetAttribute("ItemName"))
                if not has then
                    table.insert(allItems, v)
                end
            end
        end

        -- Pick up and drop in batches of 3
        local batch = 0
        for _, v in ipairs(allItems) do
            PickupItem(v)
            task.wait(0.35)
            batch = batch + 1
            if batch >= 3 then
                task.wait(0.2)
                for i = 1, 3 do
                    EquipItem(1)
                    task.wait(0.2)
                    DropItem(1)
                    task.wait(0.2)
                end
                batch = 0
            end
        end

        -- Drop any remaining
        if batch > 0 then
            for i = 1, batch do
                EquipItem(1)
                task.wait(0.2)
                DropItem(1)
                task.wait(0.2)
            end
        end

        TpOutside()
        WindUI:Notify({ Title = "Place Stuff", Content = "All items placed near ghost", Duration = 2 })
    end,
})

                itemType = "BurntCross"
                player.Character.HumanoidRootPart.CFrame = v:GetPivot() * CFrame.new(0,1,0)
                break
            end
        end
        if not item then
            for _, v in pairs(workspace.Interactables:GetChildren()) do
                if v:GetAttribute("PhotoRewardAvailable") and not AlreadyTaken(v) then
                    SetTaken(v)
                    item = v
                    itemType = "Interaction"
                    player.Character.HumanoidRootPart.CFrame = v:GetPivot() * CFrame.new(0,1,0)
                    break
                end
            end
        end
        if item then
            pcall(function()
                Events.TakePhotoWithCamera:FireServer(
                    workspace.CurrentCamera.CFrame,
                    { Stars = 3, Type = itemType, Object = item, Reward = (itemType == "BurntCross" and 12 or 8) }
                )
            end)
            WindUI:Notify({ Title = "3 Stars", Content = "Photo taken!", Duration = 2 })
        else
            WindUI:Notify({ Title = "3 Stars", Content = "No object found", Duration = 2, Type = "error" })
        end
        TpOutside()
    end,
})
