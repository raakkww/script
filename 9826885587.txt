-- Populate dropdown with islands
local function populateIslands()
    for _, child in ipairs(GUI.ScrollFrame:GetChildren()) do

            child:Destroy()
        end
    end

    if not scene then

        return
    end
    
    local islandFolder = scene:FindFirstChild("Island")
    if not islandFolder then
        GUI.StatusLabel.Text = "❌ Island folder not found!"
        return
    end
    
    local count = 0
    for _, island in ipairs(islandFolder:GetChildren()) do

            count = count + 1
            
            local itemButton = Instance.new("TextButton")

            itemButton.BorderSizePixel = 0

            itemButton.TextColor3 = Color3.fromRGB(255, 255, 255)
            itemButton.TextSize = 14
            itemButton.TextXAlignment = Enum.TextXAlignment.Left
            itemButton.TextYAlignment = Enum.TextYAlignment.Center
            itemButton.Font = Enum.Font.GothamBold
            itemButton.ZIndex = 11
            itemButton.Parent = GUI.ScrollFrame
            
            local itemCorner = Instance.new("UICorner")
            itemCorner.CornerRadius = UDim.new(0, 4)
            itemCorner.Parent = itemButton
            
            itemButton.MouseEnter:Connect(function()
                itemButton.BackgroundTransparency = 0.3
                itemButton.BackgroundColor3 = Color3.fromRGB(80, 80, 100)
            end)
            
            itemButton.MouseLeave:Connect(function()
                itemButton.BackgroundTransparency = 0.9
                itemButton.BackgroundColor3 = Color3.fromRGB(60, 60, 70)
            end)
            
            itemButton.MouseButton1Click:Connect(function()
                selectedIsland = island.Name
                GUI.DropdownButton.Text = "📍 " .. island.Name
                GUI.DropdownList.Visible = false
                GUI.DropdownArrow.Text = "▼"
                dropdownOpen = false
                GUI.DropdownList.Size = UDim2.new(1, 0, 0, 0)
                GUI.StatusLabel.Text = "✅ Selected: " .. island.Name
            end)
        end
    end
    
    if count > 0 then
        GUI.StatusLabel.Text = "✅ Found " .. count .. " islands"
        local firstIsland = islandFolder:GetChildren()[1]
        if firstIsland and firstIsland:IsA("Model") then
            selectedIsland = firstIsland.Name
            GUI.DropdownButton.Text = "📍 " .. firstIsland.Name
        end
    else
        GUI.StatusLabel.Text = "❌ No islands found!"
    end
end

task.wait(1)

-- Teleport function
local function teleportToIsland(islandName)
    if isTeleporting then return end
    if not islandName then
        GUI.StatusLabel.Text = "⚠️ No island selected!"
        return
    end
    
    isTeleporting = true
    GUI.StatusLabel.Text = "🔄 Teleporting to " .. islandName .. "..."
    
    local scene = workspace:FindFirstChild("Scene")
    if not scene then
        GUI.StatusLabel.Text = "❌ Scene not found!"
        isTeleporting = false
        return
    end
    
    local islandFolder = scene:FindFirstChild("Island")
    if not islandFolder then
        GUI.StatusLabel.Text = "❌ Island folder not found!"
        isTeleporting = false
        return
    end
    
    local targetIsland = islandFolder:FindFirstChild(islandName)
    if not targetIsland then
        GUI.StatusLabel.Text = "❌ Island not found: " .. islandName
        isTeleporting = false
        return
    end
    
    local destination = nil
    for _, child in ipairs(targetIsland:GetDescendants()) do
        if child:IsA("BasePart") then
            destination = child
            break
        end
    end
    
    if not destination then
        GUI.StatusLabel.Text = "❌ No parts found in island!"
        isTeleporting = false
        return
    end
    
    local teleportPosition = destination.Position + Vector3.new(0, 5, 0)
    
    local success, err = pcall(function()
        rootPart.CFrame = CFrame.new(teleportPosition)
    end)
    
    if success then
        GUI.StatusLabel.Text = "✅ Teleported to " .. islandName
    else
        GUI.StatusLabel.Text = "❌ Teleport failed!"
    end
    
    task.wait(0.5)
    isTeleporting = false
end

GUI.TeleportButton.MouseButton1Click:Connect(function()
    teleportToIsland(selectedIsland)
end)

local function findNearestCreature()
    local nearest = nil
    local nearestDist = math.huge
    local charPos = rootPart.Position
    
    local creatureCache = workspace:FindFirstChild("RuntimeCache")
    if creatureCache then
        creatureCache = creatureCache:FindFirstChild("RuntimeCacheServer")
        if creatureCache then
            creatureCache = creatureCache:FindFirstChild("CreatureModelCache")
        end
    end
    
    if not creatureCache then
        GUI.StatusLabel.Text = "❌ CreatureModelCache not found!"
        return nil
    end
    
    for _, outerModel in ipairs(creatureCache:GetChildren()) do
        if outerModel:IsA("Model") then
            if string.match(outerModel.Name, "^%d%d%d%d$") then
                for _, innerModel in ipairs(outerModel:GetChildren()) do
                    if innerModel:IsA("Model") then
                        local creatureHumanoid = innerModel:FindFirstChild("Humanoid")
                        if creatureHumanoid and creatureHumanoid:IsA("Humanoid") and creatureHumanoid.Health > 0 then
                            local targetPart = innerModel:FindFirstChild("HumanoidRootPart")
                            if not targetPart then
                                for _, child in ipairs(innerModel:GetChildren()) do
                                    if child:IsA("BasePart") then
                                        targetPart = child
                                        break
                                    end
                                end
                            end
                            
                            if targetPart and targetPart:IsA("BasePart") then
                                local dist = (targetPart.Position - charPos).Magnitude
                                if dist < 1 then continue end -- Skip self
                                
                                if dist < nearestDist and dist < 200 then
                                    nearestDist = dist
                                    nearest = {
                                        OuterModel = outerModel,
                                        InnerModel = innerModel,
                                        Humanoid = creatureHumanoid,
                                        Part = targetPart,
                                        Distance = dist,
                                        ID = outerModel.Name
                                    }
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    
    if nearest then
        GUI.StatusLabel.Text = "🎯 " .. nearest.ID .. " (" .. math.floor(nearest.Distance) .. " studs)"
    else
        GUI.StatusLabel.Text = "🔍 No creatures found!"
    end
    
    return nearest
end

-- ============================================
-- PRESS E KEY
-- ============================================

local function pressE()
    if not isFarming then return end
    
    local currentTime = tick()
    if currentTime - lastEPressTime < E_PRESS_INTERVAL then return end
    
    lastEPressTime = currentTime
    isEPressed = true
    
    updateCatchButtonStatus(true)
    GUI.StatusLabel.Text = "⌨️ Catch pressed!"
    
    if VirtualInputManager then
        pcall(function()
            VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.E, false, game)
            task.wait(0.05)
            VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.E, false, game)
        end)
    end
    
    pcall(function()
        ContextActionService:FireKeyPress(Enum.KeyCode.E)
        task.wait(0.05)
        ContextActionService:FireKeyRelease(Enum.KeyCode.E)
    end)
    
    pcall(function()
        local keyDownEvent = {
            KeyCode = Enum.KeyCode.E,
            UserInputType = Enum.UserInputType.Keyboard,
            Position = Vector2.new(0, 0),
            Delta = Vector2.new(0, 0)
        }
        UserInputService:SetKeyDown(keyDownEvent)
        task.wait(0.05)
        UserInputService:SetKeyUp(keyDownEvent)
    end)
    
    task.spawn(function()
        task.wait(0.5)
        isEPressed = false
        if isFarming then
            updateCatchButtonStatus(false)
        end
    end)
end

-- ============================================

local function startEPressLoop()
    if ePressLoopConnection then
        ePressLoopConnection:Disconnect()
        ePressLoopConnection = nil
    end
    
    ePressLoopConnection = RunService.Heartbeat:Connect(function()
        if isFarming then
            pressE()
        end
    end)
end

local function stopEPressLoop()
    if ePressLoopConnection then
        ePressLoopConnection:Disconnect()
        ePressLoopConnection = nil
    end
end

-- ============================================
-- MOVEMENT - WALKS THROUGH CREATURES
-- ============================================

local function moveToCreature(creatureData)
    if not creatureData or not creatureData.Part then
        return false
    end
    
    local targetPos = creatureData.Part.Position
    local currentPos = rootPart.Position
    local distance = (targetPos - currentPos).Magnitude
    
    isMovingToTarget = true
    currentTarget = creatureData
    walkingIntoCreature = false
    targetPosition = targetPos
    moveStartTime = tick()
    
    GUI.StatusLabel.Text = "🚶 Moving to " .. creatureData.ID
    
    -- Calculate position WELL PAST the creature (8 studs beyond)
    local directionToTarget = (targetPos - currentPos).Unit
    local walkThroughPos = targetPos + (directionToTarget * 8)
    
    humanoid:MoveTo(walkThroughPos)
    
    if moveCheckConnection then
        moveCheckConnection:Disconnect()
        moveCheckConnection = nil
    end
    
    moveCheckConnection = RunService.Heartbeat:Connect(function()
        if not isFarming or not isMovingToTarget then
            if moveCheckConnection then
                moveCheckConnection:Disconnect()
                moveCheckConnection = nil
            end
            return
        end
        
        if not creatureData or not creatureData.Part or not creatureData.Part.Parent then
            isMovingToTarget = false
            if moveCheckConnection then
                moveCheckConnection:Disconnect()
                moveCheckConnection = nil
            end
            return
        end
        
        if creatureData.Humanoid and creatureData.Humanoid.Health <= 0 then
            isMovingToTarget = false
            if moveCheckConnection then
                moveCheckConnection:Disconnect()
                moveCheckConnection = nil
            end
            local nextTarget = findNearestCreature()
            if nextTarget then
                moveToCreature(nextTarget)
            else
                isMovingToTarget = false
                GUI.StatusLabel.Text = "🔍 No more creatures found"
            end
            return
        end
        
        local currentPos = rootPart.Position
        local targetPos = creatureData.Part.Position
        local dist = (targetPos - currentPos).Magnitude
        
        -- Keep moving toward the walk-through position
        local direction = (targetPos - currentPos).Unit
        local newWalkThrough = targetPos + (direction * 8)
        humanoid:MoveTo(newWalkThrough)
        
        -- If we're very close to the creature, press E and keep walking
        if dist < 2.5 then
            walkingIntoCreature = true
            GUI.StatusLabel.Text = "💥 Fighting " .. creatureData.ID
            
            if VirtualInputManager then
                pcall(function()
                    VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.W, false, game)
                    task.wait(0.1)
                    VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.W, false, game)
                end)
            end
            
            pressE()
            
            local directionPast = (targetPos - currentPos).Unit
            local pastPos = targetPos + (directionPast * 10)
            humanoid:MoveTo(pastPos)
            
            task.wait(0.3)
            isMovingToTarget = false
            
            if moveCheckConnection then
                moveCheckConnection:Disconnect()
                moveCheckConnection = nil
            end
            
            local nextTarget = findNearestCreature()
            if nextTarget then
                moveToCreature(nextTarget)
            else
                isMovingToTarget = false
                GUI.StatusLabel.Text = "🔍 No more creatures found"
            end
            return
        end
        
        if dist < 8 then
            if VirtualInputManager then
                pcall(function()
                    VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.W, false, game)
                    task.wait(0.1)
                    VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.W, false, game)
                end)
            end
        end
    end)
    
    return true
end

-- ============================================
-- FARMING SYSTEM
-- ============================================

local function startFarming()
    if isFarming then return end
    
    if not character or not character.Parent then
        GUI.StatusLabel.Text = "❌ Character not found!"
        return
    end
    
    isFarming = true
    isMovingToTarget = false
    lastEPressTime = tick()
    walkingIntoCreature = false
    
    startEPressLoop()
    
    GUI.FarmingButton.BackgroundColor3 = Color3.fromRGB(0, 200, 50)
    GUI.FarmingButton.Text = "⚔️ Farming: ON"
    GUI.StatusLabel.Text = "🔍 Searching for creatures..."
    updateCatchButtonStatus(false)
    
    print("⚔️ FARMING STARTED - Walking THROUGH creatures")
    
    local initialTarget = findNearestCreature()
    if initialTarget then
        moveToCreature(initialTarget)
    else
        GUI.StatusLabel.Text = "⚠️ No creatures found!"
    end
    
    if farmingConnection then
        farmingConnection:Disconnect()
    end
    
    farmingConnection = RunService.Heartbeat:Connect(function()
        if not isFarming then return end
        
        if not character or not character.Parent then
            stopFarming()
            return
        end
        
        if not isMovingToTarget then
            if currentTarget and currentTarget.Part and currentTarget.Part.Parent then
                local dist = (currentTarget.Part.Position - rootPart.Position).Magnitude
                if dist > 30 then
                    local newTarget = findNearestCreature()
                    if newTarget then
                        moveToCreature(newTarget)
                    end
                elseif dist < 3 then
                    pressE()
                end
            else
                local newTarget = findNearestCreature()
                if newTarget then
                    moveToCreature(newTarget)
                end
            end
        end
    end)
end

local function stopFarming()
    isFarming = false
    isMovingToTarget = false
    currentTarget = nil
    walkingIntoCreature = false
    targetPosition = nil
    
    stopEPressLoop()
    
    if moveCheckConnection then
        moveCheckConnection:Disconnect()
        moveCheckConnection = nil
    end
    
    GUI.FarmingButton.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
    GUI.FarmingButton.Text = "⚔️ Farming: OFF"
    GUI.StatusLabel.Text = "⏹️ Farming stopped - E presses stopped"
    updateCatchButtonStatus(false)
    
    humanoid:MoveTo(rootPart.Position)
    
    if farmingConnection then
        farmingConnection:Disconnect()
        farmingConnection = nil
    end
    
    print("⏹️ FARMING STOPPED - E presses stopped")
end

-- Farming button

        stopFarming()
    else
        startFarming()
    end
end)
