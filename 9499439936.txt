-- 🔧 TP
local function tp(pos)
    local char = player.Character
    if char and char:FindFirstChild("HumanoidRootPart") then
        char.HumanoidRootPart.CFrame = CFrame.new(pos)
    end
end

local autoDiamond = false

local diamondPositions = {
    Vector3.new(47.55,621.20,-48.67),
    Vector3.new(-47.84,261.20,-36.70),
    Vector3.new(1.36,130.20,0.45),
    Vector3.new(-48.19,458.50,47.73)

        task.wait(0.2)
        if autoDiamond then

-- 🪙 AUTO COIN JUMP (TP 1x + PULO)
-- =====================
local autoJumpCoins = {false,false,false,false}
local jumped = {false,false,false,false}

local jumpPositions = {
    Vector3.new(0.03,228.35,-33.35),
    Vector3.new(80.46,316.35,-64.31),
    Vector3.new(-16.58,449.35,-34.77),
    Vector3.new(-33.77,560.35,47.51)
}

task.spawn(function()
    while true do
        task.wait(0.2)
        for i,pos in ipairs(jumpPositions) do
            if autoJumpCoins[i] then

                if not jumped[i] then
                    tp(pos)
                    jumped[i] = true
                end

                if player.Character then
                    local hum = player.Character:FindFirstChildOfClass("Humanoid")
                    if hum and hum.FloorMaterial ~= Enum.Material.Air then
                        hum.Jump = true
                    end
                end

            else
                jumped[i] = false
            end
        end
    end
end)

for i = 1,4 do
    Main:CreateToggle({
        Name = "Auto Coin Jump "..i,
        CurrentValue = false,
        Callback = function(v)
            autoJumpCoins[i] = v
        end
    })
end
