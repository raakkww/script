-- === HELPER FUNCTIONS ===

local function getCar()
    local char = LocalPlayer.Character
    if not char then return nil, nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return nil, nil end
    local seat = hum.SeatPart
    if not seat then return nil, nil end
    return seat.Parent, seat
end

local function cleanBodyMovers(seat)
    if not seat then return end
    for _, child in pairs(seat:GetChildren()) do
        if child:IsA("BodyGyro") or child:IsA("BodyVelocity") then
            child:Destroy()
        end
    end
end

-- === ANTI-AFK ===
local antiAfkConn = nil
local function startAntiAfk()
    antiAfkConn = LocalPlayer.Idled:Connect(function()
        VirtualUser:CaptureController()
        VirtualUser:ClickButton2(Vector2.new(0, 0))
    end)
end

local function stopAntiAfk()
    if antiAfkConn then
        antiAfkConn:Disconnect()
        antiAfkConn = nil
    end
end

-- === MAIN LOGIC ===
local flySpeed = 500
local ascendSpeed = 250 -- slower for controlled ascent

local function runAutoFarm()
    task.spawn(function()
        while _G.AutoFarmEnabled do
            local car, seat = getCar()
            
            -- If not in a car, wait
            if not car or not seat then
                StatusText.Text = "STATUS: WAITING FOR CAR..."
                StatusText.TextColor3 = Color3.fromRGB(255, 200, 0)
                BarFill.Size = UDim2.new(0.2, 0, 1, 0)
                BarFill.BackgroundColor3 = Color3.fromRGB(255, 200, 0)
                task.wait(1)
                continue
            end
            
            -- Clean old body movers
            cleanBodyMovers(seat)
            
            -- Unanchor entire car
            for _, part in pairs(car:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.Anchored = false
                end
            end
            
            -- Create movers
            local bg = Instance.new("BodyGyro", seat)
            local bv = Instance.new("BodyVelocity", seat)
            bg.P = 9e4
            bg.D = 2000
            bg.maxTorque = Vector3.new(9e9, 9e9, 9e9)
            bv.maxForce = Vector3.new(9e9, 9e9, 9e9)
            
            local hasAscended = false
            
            -- Flight loop while in the SAME car
            while _G.AutoFarmEnabled do
                local currentCar, currentSeat = getCar()
                if currentSeat ~= seat then
                    break -- Left the car or changed cars
                end
                
                if not hasAscended then
                    -- PHASE 1: ASCEND (only once per car)
                    StatusText.Text = "STATUS: ASCENDING..."
                    StatusText.TextColor3 = Color3.fromRGB(100, 255, 255)
                    BarFill.BackgroundColor3 = Color3.fromRGB(0, 200, 255)
                    BarFill.Size = UDim2.new(0.3, 0, 1, 0)
                    
                    bg.CFrame = CFrame.new(seat.Position, seat.Position + Vector3.new(0, 1, 0))
                    bv.Velocity = Vector3.new(0, ascendSpeed, 0)
                    task.wait(2.5)
                    hasAscended = true
                else
                    -- PHASE 2: LEFT
                    if not _G.AutoFarmEnabled then break end
                    local checkCar, checkSeat = getCar()
                    if checkSeat ~= seat then break end
                    
                    StatusText.Text = "STATUS: LEFT →"
                    StatusText.TextColor3 = Color3.fromRGB(255, 255, 100)
                    BarFill.BackgroundColor3 = Color3.fromRGB(255, 200, 0)
                    BarFill.Size = UDim2.new(0.5, 0, 1, 0)
                    
                    bg.CFrame = CFrame.new(seat.Position, seat.Position + Vector3.new(-1, 0, 0))
                    bv.Velocity = Vector3.new(-flySpeed, 0, 0)
                    task.wait(1.5)
                    
                    -- PHASE 3: RIGHT
                    if not _G.AutoFarmEnabled then break end
                    checkCar, checkSeat = getCar()
                    if checkSeat ~= seat then break end
                    
                    StatusText.Text = "STATUS: ← RIGHT"
                    StatusText.TextColor3 = Color3.fromRGB(100, 255, 100)
                    BarFill.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
                    BarFill.Size = UDim2.new(1, 0, 1, 0)
                    
                    bg.CFrame = CFrame.new(seat.Position, seat.Position + Vector3.new(1, 0, 0))
                    bv.Velocity = Vector3.new(flySpeed, 0, 0)
                    task.wait(1.5)
                end
            end
            
            -- Cleanup on car exit or disable
            cleanBodyMovers(seat)
            if car then
                for _, part in pairs(car:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.Anchored = false
                    end
                end
            end
            
            task.wait(0.5)
        end
        
        -- Final status
        StatusText.Text = "STATUS: OFF"
        StatusText.TextColor3 = Color3.fromRGB(255, 255, 255)
        BarFill.Size = UDim2.new(0, 0, 1, 0)
    end)
end

-- === BUTTONS ===

ToggleBtn.MouseButton1Click:Connect(function()
    _G.AutoFarmEnabled = not _G.AutoFarmEnabled
    
    if _G.AutoFarmEnabled then
        ToggleBtn.Text = "AUTOFARM: ON"
        ToggleBtn.TextColor3 = Color3.fromRGB(100, 255, 100)
        startAntiAfk()
        runAutoFarm()
    else
        ToggleBtn.Text = "AUTOFARM: OFF"
        ToggleBtn.TextColor3 = Color3.fromRGB(255, 100, 100)
        stopAntiAfk()
    end
end)

    minimized = not minimized
    if minimized then
        MainFrame:TweenSize(UDim2.new(0, 320, 0, 40), "Out", "Quad", 0.3, true)
        for _, child in pairs(MainFrame:GetChildren()) do
            if child ~= Title and child ~= MinBtn and child ~= CloseBtn and child:IsA("GuiObject") then
                child.Visible = false
            end
        end
    else
        MainFrame:TweenSize(UDim2.new(0, 320, 0, 260), "Out", "Quad", 0.3, true)
        for _, child in pairs(MainFrame:GetChildren()) do
            if child:IsA("GuiObject") then
                child.Visible = true
            end
        end
    end
end)
